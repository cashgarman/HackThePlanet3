 695    // Internal member supporting workstation or graph viewport behavior.
 696    private GraphScene BuildRackScene()
 697    {
 698      // Create the document and register the empty-cell node type.
 699      var document = new GraphDocument("rig", "Rig rack");
 700      var schema = new GraphSchema();
 701      schema.Register(new GraphNodeTypeDefinition("empty", "Empty", "default", Palette.Hex("243A40"), .2f));
 702
 703      // Register one node type per module branch, colored by branch.
 704      foreach (var branch in State.Modules.Select(m => m.branch).Distinct())
 705        schema.Register(new GraphNodeTypeDefinition(branch, branch, "default", Palette.Branch(branch), .6f));
 706
 707      // Add an empty slot node for every chassis cell.
 708      for (var row = 0; row < State.Chassis.rows; row++)
 709        for (var col = 0; col < State.Chassis.columns; col++)
 710        {
 711          // Slot nodes are keyed by their row-major index.
 712          var index = row * State.Chassis.columns + col;
 713          document.UpsertNode(new GraphNodeRecord("slot-" + index, "empty", ""));
 714        }
 715
 716      // Add a node for each installed part.
 717      foreach (var part in State.InstalledParts)
 718        document.UpsertNode(new GraphNodeRecord("part-" + part.instanceId, State.Module(part.modelId).branch,
 719                            State.Module(part.modelId).title));
 720
 721      // Build the scene from the document and its schema.
 722      var scene = new GraphScene(document, schema);
 723
 724      // Pin each cell node to its grid position.
 725      for (var row = 0; row < State.Chassis.rows; row++)
 726        for (var col = 0; col < State.Chassis.columns; col++)
 727        {
 728          var node = scene.Nodes["slot-" + (row * State.Chassis.columns + col)];
 729          node.Position = new Vector3(col * 4, -row * 4, 0);
 730          node.Pinned = true;
 731        }
 732
 733      // Center each installed part node over its footprint and pin it.
 734      foreach (var part in State.InstalledParts)
 735      {
 736        var size = State.PartSize(part);
 737        var node = scene.Nodes["part-" + part.instanceId];
 738        node.Position = new Vector3((part.column + (size.x - 1) * .5f) * 4,
 739                      -(part.row + (size.y - 1) * .5f) * 4, 0);
 740        node.Pinned = true;
 741      }
 742
