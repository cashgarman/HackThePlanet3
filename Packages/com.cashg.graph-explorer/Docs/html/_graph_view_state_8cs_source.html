57 public GraphNodeViewState GetOrCreate(string id)
58 {
59 GraphRecordCopy.Required(id, nameof(id));
60
61 // Seed new view positions from the node ID so the initial layout is repeatable.
62 if (!nodes.TryGetValue(id, out var state))
63 nodes.Add(id, state = new GraphNodeViewState { Position = InitialPosition(id) });

69 public void Synchronize(GraphDocument document)
70 {
71 if (document == null || document.DatasetId != DatasetId)
72 throw new ArgumentException("View and document must belong to the same dataset.", nameof(document));
73
74 // Remove state and selections that no longer refer to document nodes.
75 foreach (var id in nodes.Keys.Where(id => !document.Nodes.ContainsKey(id)).ToArray()) nodes.Remove(id);
76 FoldedNodeIds.RemoveWhere(id => !document.Nodes.ContainsKey(id));
77 if (!document.Nodes.ContainsKey(SelectedId ?? "")) SelectedId = "";
78 if (!document.Nodes.ContainsKey(FocusId ?? "")) FocusId = "";
79 }

84 public static Vector3 InitialPosition(string id)
85 {
86 // Derive coordinates from SHA-256 bytes so independent views and process runs share the same seed layout.
87 using (var hash = SHA256.Create())
88 {
89 var bytes = hash.ComputeHash(Encoding.UTF8.GetBytes(GraphRecordCopy.Required(id, nameof(id))));
90 return new Vector3((BitConverter.ToUInt16(bytes, 0) / 65535f - .5f) * 65,
91 (BitConverter.ToUInt16(bytes, 2) / 65535f - .5f) * 40,
92 (BitConverter.ToUInt16(bytes, 4) / 65535f - .5f) * 45);
93 }
94 }
