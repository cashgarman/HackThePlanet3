75 public IEnumerator LoadEvent(GraphDocument document, string id, CancellationToken token,
76 Action<GraphSnapshot> completed, Action<Exception> failed)
77 {
78 if (token.IsCancellationRequested)
79 {
80 failed(new OperationCanceledException());
81 yield break;
82 }
83
84 string context = null;
85 Exception error = null;
86
87 // Provider discovery can fail while the document still contains a usable portable cache.
88 try
89 {
90 context = Provider?.CacheContext;
91 }
92 catch (Exception failure)
93 {
94 error = failure;
95 }
96
97 // Prefer the active provider's cache context to avoid loading an identically named event from another source.
98 if (context != null && document.Timeline.Workspace.TryGet(context, id, out var cached))
99 {
100 completed(cached);
101 yield break;
102 }
103
104 if (Provider == null || error != null)
105 {
106 // Without a provider context, accept a portable match only when its identity is unambiguous.
107 var portable = document.Timeline.Workspace.CachedEvents.Where(item => item.Event.Id == id).ToArray();
108 if (portable.Length == 1)
109 {
110 completed(portable[0].Snapshot);
111 yield break;
112 }
113
114 failed(error ?? new InvalidOperationException("No timeline event provider is attached."));
115 yield break;
116 }
117
118 yield return Provider.LoadEvent(id, token, completed, failed);
119 }

125 public void RestoreCachedEvents(GraphTimelineWorkspace workspace, string context = null)
126 {
127 // Collapse duplicate event IDs across saved contexts before replacing the descriptor list.
128 Replace(workspace.CachedEvents
129 .Where(item => context == null || item.Context == context)
130 .GroupBy(item => item.Event.Id, StringComparer.Ordinal)
131 .Select(group => group.Last().Event));
132 }

136 public void Replace(IEnumerable<GraphTimelineEvent> incoming)
137 {
138 var copy = (incoming ?? Array.Empty<GraphTimelineEvent>()).ToArray();
139
140 // Validate the replacement before notifying playback or discarding confirmed selection.
141 if (copy.Any(item => item == null) ||
142 copy.Select(item => item.Id).Distinct(StringComparer.Ordinal).Count() != copy.Length)
143 throw new ArgumentException("Duplicate or null timeline event.", nameof(incoming));
144 // OrderBy is stable: equal timestamps preserve the host's discovery order.
145 SelectionStarting?.Invoke();
146 Events = Array.AsReadOnly(copy.OrderBy(item => item.TimestampUtc).ToArray());
147 SelectedId = PendingId = Error = null;
148 Revision++;
149 Changed?.Invoke();
150 }
151
155 public void RequestSelection(string id, bool stopPlayback = true)
156 {
157 var item = Events.FirstOrDefault(value => value.Id == id) ??
158 throw new ArgumentException("Unknown event.", nameof(id));
159 if (stopPlayback) SelectionStarting?.Invoke();
160
161 // Keep the confirmed selection until the host finishes loading this pending request.
162 PendingId = id;
163 Error = null;
164 Changed?.Invoke();
165 SelectionRequested?.Invoke(item);
166 }

170 public void ConfirmSelection(string id)
171 {
172 if (!Events.Any(item => item.Id == id)) throw new ArgumentException("Unknown event.", nameof(id));
173 SelectedId = id;
174 PendingId = Error = null;

176 }

181 public void FailSelection(string id, string error)
182 {
183 if (PendingId != id) return;
184
185 // Ignore stale failures; a current failure clears only the pending request.
186 PendingId = null;
187 Error = error ?? "Could not load event.";
188 Changed?.Invoke();
189 }
