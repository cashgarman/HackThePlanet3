48 public static Button HeaderButton(string text, string icon, Action action)
49 {
50 // Render the icon and caption as noninteractive children so the button owns their pointer events.
51 var button = new Button(action) { text = "", tooltip = text };
52 button.AddToClassList("header-button");
53 var glyph = new VisualElement { pickingMode = PickingMode.Ignore };
54 glyph.AddToClassList("header-icon");
55 glyph.generateVisualContent += ctx => HeaderIcons.Paint(ctx.painter2D, icon, glyph.contentRect);
56 glyph.RegisterCallback<GeometryChangedEvent>(_ => glyph.MarkDirtyRepaint());
57 var caption = Label(text, "header-button-label");
58 caption.pickingMode = PickingMode.Ignore;
59 button.Add(glyph);
60 button.Add(caption);
61 return button;
62 }

180 public static void PaintCheck(Painter2D paint, Rect rect)
181 {
182 if (rect.width < 4f || rect.height < 4f) return;
183 paint.strokeColor = new Color(1f, 0.973f, 0.941f);
184 paint.lineWidth = 2f;
185 paint.lineCap = LineCap.Round;
186 paint.lineJoin = LineJoin.Round;
187 paint.BeginPath();
188 paint.MoveTo(new Vector2(rect.xMin + 3f, rect.yMin + rect.height * 0.52f));
189 paint.LineTo(new Vector2(rect.xMin + rect.width * 0.42f, rect.yMax - 3.5f));
190 paint.LineTo(new Vector2(rect.xMax - 2.5f, rect.yMin + 3.5f));
191 paint.Stroke();
192 }
