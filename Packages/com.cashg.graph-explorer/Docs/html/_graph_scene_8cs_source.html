88 public GraphScene(GraphDocument document, GraphSchema schema, GraphViewState view = null)
89 {
90 Document = document ?? throw new ArgumentNullException(nameof(document));
91 Schema = schema ?? throw new ArgumentNullException(nameof(schema));
92 View = view ?? new GraphViewState(document.DatasetId);
93 if (View.DatasetId != document.DatasetId)
94 throw new ArgumentException("View belongs to another dataset.", nameof(view));
95 Nodes = new ReadOnlyDictionary<string, IGraphNode>(nodes);
96 Edges = new ReadOnlyDictionary<string, IGraphEdge>(edges);
97 Synchronize();
98 }

101 public GraphDocument Document { get; }
103 public GraphViewState View { get; }
105 public GraphSchema Schema { get; }
107 public IReadOnlyDictionary<string, IGraphNode> Nodes { get; }
109 public IReadOnlyDictionary<string, IGraphEdge> Edges { get; }
111 public int Revision => Document.TopologyRevision;
112
114 public void Synchronize()
115 {
116 // Prune removed facts while retaining wrappers and view state for surviving identities.
117 View.Synchronize(Document);
118 foreach (var id in nodes.Keys.Where(id => !Document.Nodes.ContainsKey(id)).ToArray()) nodes.Remove(id);
119 foreach (var id in edges.Keys.Where(id => !Document.Edges.ContainsKey(id)).ToArray()) edges.Remove(id);
120
121 // Add wrappers only for new facts; existing wrappers read updated records directly from the document.
122 foreach (var id in Document.Nodes.Keys)
123 if (!nodes.ContainsKey(id))
124 nodes.Add(id, new Node(this, id));
125 foreach (var id in Document.Edges.Keys)
126 if (!edges.ContainsKey(id))
127 edges.Add(id, new Edge(this, id));
128 }
