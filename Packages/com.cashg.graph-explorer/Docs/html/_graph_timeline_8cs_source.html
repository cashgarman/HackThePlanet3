15 public GraphPositionFrame(DateTimeOffset timestampUtc, IEnumerable<KeyValuePair<string, Vector3>> positions)
16 {
17 TimestampUtc = timestampUtc.ToUniversalTime();
18
19 // Copy only valid, finite position samples and reject duplicate node IDs before exposing the frame.
20 var copy = new SortedDictionary<string, Vector3>(StringComparer.Ordinal);
21 if (positions != null)
22 foreach (var pair in positions)
23 {
24 if (string.IsNullOrWhiteSpace(pair.Key))
25 throw new ArgumentException("Position sample node IDs cannot be empty.", nameof(positions));
26 if (!Finite(pair.Value.x) || !Finite(pair.Value.y) || !Finite(pair.Value.z))
27 throw new ArgumentOutOfRangeException(nameof(positions), "Node positions must be finite.");
28 copy.Add(pair.Key, pair.Value);
29 }
30
31 Positions = new ReadOnlyDictionary<string, Vector3>(copy);
32 }

218 internal void Append(GraphTimelineChange change, DateTimeOffset? timestampUtc = null)
219 {
220 if (!Enabled || !HasBaseline || change == null) return;
221
222 // Discard entries beyond the cursor and keep new timestamps monotonic across both tracks.
223 if (Cursor < entries.Count - 1) entries.RemoveRange(Cursor + 1, entries.Count - Cursor - 1);
224 var timestamp = (timestampUtc ?? DateTimeOffset.UtcNow).ToUniversalTime();
225 if (timestamp < LastRecordedTimeUtc) timestamp = LastRecordedTimeUtc;
226 entries.Add(new GraphTimelineEntry(entries.Count + 1L, timestamp, change));
227 Cursor = entries.Count - 1;
228 playheadTimeUtc = timestamp;
229 hasPlayhead = true;
230 }

235 public void RecordPositionFrame(IEnumerable<KeyValuePair<string, Vector3>> positions,
236 DateTimeOffset? timestampUtc = null)
237 {
238 if (!Enabled || !PositionRecordingAvailable || !PositionRecordingEnabled || !HasBaseline ||
239 IsHistorical) return;
240
241 // Append motion only in the live view and advance its playhead with the new sample.
242 var wasHistorical = IsHistorical;
243 var timestamp = (timestampUtc ?? DateTimeOffset.UtcNow).ToUniversalTime();
244 if (timestamp < LastRecordedTimeUtc) timestamp = LastRecordedTimeUtc;
245 positionFrames.Add(new GraphPositionFrame(timestamp, positions));
246 if (!wasHistorical)
247 {
248 playheadTimeUtc = timestamp;
249 hasPlayhead = true;
250 }

252 Changed?.Invoke();
253 }

258 public int EntryIndexAt(DateTimeOffset timeUtc)
259 {
260 // Binary-search for the final semantic entry at or before the requested timestamp.
261 var ticks = timeUtc.ToUniversalTime().UtcDateTime.Ticks;
262 int low = 0, high = entries.Count - 1, result = -1;
263 while (low <= high)
264 {
265 var middle = low + (high - low) / 2;
266 if (entries[middle].TimestampUtc.UtcDateTime.Ticks <= ticks)
267 {
268 result = middle;
269 low = middle + 1;
270 }
271 else
272 {
273 high = middle - 1;
274 }
275 }
276
277 return result;
278 }

283 public IReadOnlyDictionary<string, Vector3> PositionsAt(DateTimeOffset timeUtc)
284 {
285 var result = new SortedDictionary<string, Vector3>(StringComparer.Ordinal);
286 if (positionFrames.Count == 0) return new ReadOnlyDictionary<string, Vector3>(result);
287
288 // Find the bracketing samples, using an endpoint frame outside the recorded range.
289 var ticks = timeUtc.ToUniversalTime().UtcDateTime.Ticks;
290 var right = 0;
291 while (right < positionFrames.Count &&
292 positionFrames[right].TimestampUtc.UtcDateTime.Ticks < ticks) right++;
293 var before = right > 0 ? positionFrames[right - 1] : null;
294 var after = right < positionFrames.Count ? positionFrames[right] : null;
295 if (before == null) before = after;
296 if (after == null) after = before;
297 if (before == null) return new ReadOnlyDictionary<string, Vector3>(result);
298 var span = after.TimestampUtc.UtcDateTime.Ticks - before.TimestampUtc.UtcDateTime.Ticks;
299 var amount = span <= 0
300 ? 0
301 : Mathf.Clamp01((float)((ticks - before.TimestampUtc.UtcDateTime.Ticks) / (double)span));
302 foreach (var pair in before.Positions) result[pair.Key] = pair.Value;
303
304 // Interpolate shared nodes and retain positions that appear in only one frame.
305 foreach (var pair in after.Positions)
306 result[pair.Key] = before.Positions.TryGetValue(pair.Key, out var start)
307 ? Vector3.Lerp(start, pair.Value, amount)
308 : pair.Value;
309 return new ReadOnlyDictionary<string, Vector3>(result);
310 }

352 internal void Restore(GraphSnapshot baseline, DateTimeOffset baselineTimeUtc,
353 IEnumerable<GraphTimelineEntry> history, int cursor,
354 IEnumerable<GraphPositionFrame> positionHistory = null,
355 bool positionRecordingEnabled = false, float sampleIntervalSeconds = 1f,
356 DateTimeOffset? currentTimeUtc = null)
357 {
358 Baseline = baseline;
359 BaselineTimeUtc = baselineTimeUtc.ToUniversalTime();
360
361 // Restore each track in its own ordering before rebuilding the bounded cursor and playhead.
362 entries.Clear();
363 if (history != null) entries.AddRange(history.OrderBy(item => item.Sequence));
364 positionFrames.Clear();
365 if (positionHistory != null) positionFrames.AddRange(positionHistory.OrderBy(item => item.TimestampUtc));
366 this.positionRecordingEnabled = positionRecordingEnabled;
367 positionSampleIntervalSeconds = Mathf.Clamp(sampleIntervalSeconds, .1f, 60f);
368 Cursor = Math.Max(-1, Math.Min(cursor, entries.Count - 1));
369 playheadTimeUtc =
370 (currentTimeUtc ?? (Cursor == entries.Count - 1 ? LatestTimeUtc :
371 Cursor < 0 ? BaselineTimeUtc : entries[Cursor].TimestampUtc)).ToUniversalTime();
372 hasPlayhead = true;
373 }
