15 private static void Drain(IEnumerator routine)
16 {
17 // Execute nested provider iterators and dispose each scope so coordinator finally blocks run.
18 using (routine as IDisposable)
19 {
20 while (routine.MoveNext())
21 if (routine.Current is IEnumerator nested)
22 Drain(nested);
23 }
24 }

48 [Test]
49 public void SubmissionPersistsBeforeSendingAndBlocksOnStorageFailure()
50 {
51 // Assert the pending request is durable before the provider receives it.
52 var provider = new Provider();
53 var queue = new GraphQueueCoordinator(provider);
54 Exception error = null;
55 Drain(queue.Submit(Review, () => false, CancellationToken.None, _ => Assert.Fail("sent"),
56 failure => error = failure));
57 Assert.That(error, Is.Not.Null);
58 Assert.That(provider.Calls, Is.Empty);

60 {
61 provider.Calls.Add("save");
62 return true;
