9 private static GraphScene Scene()

11 // Supply equal-length routes and parallel edges so stable tie ordering can be observed.
12 var doc = new GraphDocument("paths", "Paths");
13 foreach (var id in new[] { "a", "b", "c", "d", "z" }) doc.UpsertNode(new GraphNodeRecord(id, "node", id));
14 doc.UpsertEdge(new GraphEdgeRecord("ab-z", "a", "b", "link"));
15 doc.UpsertEdge(new GraphEdgeRecord("ab-a", "a", "b", "link"));
16 doc.UpsertEdge(new GraphEdgeRecord("ac", "a", "c", "link"));
17 doc.UpsertEdge(new GraphEdgeRecord("bd", "b", "d", "link"));
18 doc.UpsertEdge(new GraphEdgeRecord("cd", "c", "d", "link"));
19 var schema = new GraphSchema();
20 schema.Register(new GraphNodeTypeDefinition("node", "Node"));
21 return new GraphScene(doc, schema);

100 scene.Synchronize();
101 Assert.That(GraphPathfinder.Find(scene, "a", "d").Jumps, Is.EqualTo(1));
102 var state = new GraphPathState(scene, edge => edge.Relation == "link");
103 state.Show("a", "d");
104 Assert.That(state.Result.Jumps, Is.EqualTo(2));
105 foreach (var node in scene.Nodes.Values) node.Visible = false;
106
107 // Display filters do not affect semantic traversal, and rebinding retains the relation predicate.
108 state.Synchronize(true);
109 Assert.That(state.Result.Jumps, Is.EqualTo(2), "UI visibility must not change semantic traversal.");
110 var replacement = Scene();
111 replacement.Document.RemoveNode("d");
112 replacement.Synchronize();
113 state.Bind(replacement);
114 Assert.That(state.IsActive, Is.True);
115 Assert.That(state.Result.IsReachable, Is.False);
116 Assert.That(state.StartId, Is.EqualTo("a"));
117 Assert.That(state.EndId, Is.EqualTo("d"));
