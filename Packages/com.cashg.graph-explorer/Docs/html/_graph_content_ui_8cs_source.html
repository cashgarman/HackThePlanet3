18 public static void AddProperty(VisualElement parent, string key, string value, Action<Foldout> decorate,
19 Action<VisualElement, string> bindTip)
20 {
21 var original = value;
22 var timestampKey = key.IndexOf("date", StringComparison.OrdinalIgnoreCase) >= 0 ||
23 key.IndexOf("time", StringComparison.OrdinalIgnoreCase) >= 0 ||
24 key == "discovered" || key == "created" || key == "updated" ||
25 key.EndsWith("_at", StringComparison.OrdinalIgnoreCase);
26 // An offset-free backend timestamp is a wall-clock value; do not invent a timezone.
27 if (timestampKey && DateTime.TryParse(value, CultureInfo.InvariantCulture, DateTimeStyles.RoundtripKind,
28 out var timestamp))
29 value = timestamp.ToString("MMM d, yyyy · HH:mm:ss", CultureInfo.InvariantCulture) +
30 (timestamp.Kind == DateTimeKind.Utc
31 ? " UTC"
32 :
33 timestamp.Kind == DateTimeKind.Local
34 ?
35 timestamp.ToString(" zzz", CultureInfo.InvariantCulture)
36 : "");
37 var caption = key.Replace('_', ' ');
38 if (caption.Length > 0) caption = char.ToUpperInvariant(caption[0]) + caption.Substring(1);
39
40 // Move multiline or long values into a disclosure so the property list remains scannable.
41 if (value.Length > 160 || value.Contains('\n'))
42 {
43 var disclosure = new Foldout { text = caption, value = false };
44 disclosure.AddToClassList("inspector-disclosure");
45 decorate(disclosure);
46 disclosure.Add(Label(value, "property-value"));
47 parent.Add(disclosure);
48 return;
49 }
50
51 var row = Element("inspector-property");
52 row.Add(Label(caption, "inspector-property-key"));
53 var content = Label(value, "inspector-property-value");
54
55 // Preserve the raw timestamp as hover text whenever its displayed value was reformatted.
56 if (value != original) bindTip(content, original);
57 row.Add(content);
58 parent.Add(row);
59 }
