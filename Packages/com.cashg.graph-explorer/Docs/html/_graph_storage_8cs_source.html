70 public bool Write(string name, int supportedVersion, string json)
71 {
72 // Validate the new payload before inspecting or replacing any existing file.
73 if (Version(json) != supportedVersion)
74 throw new ArgumentException("Document version does not match the writer.", nameof(json));
75 var path = PathFor(name);
76 if (File.Exists(path))
77 try
78 {
79 if (Version(File.ReadAllText(path, Encoding.UTF8)) != supportedVersion) return false;
80 }
81 catch (JsonException)
82 {
83 return false;
84 }
85
86 Directory.CreateDirectory(directory);
87 var temporary = path + "." + Guid.NewGuid().ToString("N") + ".tmp";
88 try
89 {
90
91 // Stage the complete file before replacing the destination.
92 File.WriteAllText(temporary, json, new UTF8Encoding(false));
93 if (File.Exists(path)) File.Replace(temporary, path, path + ".bak");
94 else File.Move(temporary, path);
95 return true;
96 }
97 finally
98 {
99 if (File.Exists(temporary)) File.Delete(temporary);
100 }
101 }
