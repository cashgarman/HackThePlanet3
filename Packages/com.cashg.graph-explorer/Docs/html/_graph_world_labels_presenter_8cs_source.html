155 public bool TryNodeLabelRect(GraphNodeLabelSettings s, string id, Vector2 anchor, float nodeRadius, float width,
156 float height, bool featured, float left, float right, float top, float bottom,
157 out Rect rect, out int rowUsed)
158 {
159 var rows = featured ? Mathf.Clamp(s.StackRows, 1, 4) : 1;
160
161 // Try the prior stacking row first to keep featured labels from jumping between free positions.
162 var start = 0;
163 if (featured && options.labelOffsets.TryGetValue(id, out var previous) && (uint)previous < (uint)rows)
164 start = previous;
165 var baseline = anchor.y - nodeRadius - Mathf.Max(0f, s.AboveNodeGap);
166 for (var n = 0; n < rows; n++)
167 {
168 var row = (start + n) % rows;
169 var x = anchor.x - width * .5f;
170 var y = baseline - height - row * (height + s.LabelSpacing);
171 if (featured)
172 {
173 // Keep selected and task labels inside the safe area even near screen boundaries.
174 x = Mathf.Clamp(x, left, Mathf.Max(left, right - width));
175 y = Mathf.Clamp(y, top, Mathf.Max(top, bottom - height));
176 }
177
178 rect = new Rect(x, y, width, height);
179 if (rect.x < left - .5f || rect.xMax > right + .5f || rect.y < top - .5f ||
180 rect.yMax > bottom + .5f) continue;
181 if (NodeLabelBlocked(rect, s.LabelSpacing)) continue;
182 rowUsed = row;
183 return true;
184 }
185
186 rect = default;
187 rowUsed = -1;
188 return false;
189 }

195 public float NodeRadiusPixels(TNode node, Vector2 anchor)
196 {
197 // Project the node's upper radius into panel space so the label clears its visible model.
198 if (options.graphRenderer == null) return 0f;
199 var top = options.Camera()
200 .WorldToScreenPoint(node.Position +
201 options.Camera().transform.up * options.graphRenderer.Radius(node));
202 if (top.z <= 0f) return 0f;
203 var topPanel =
204 RuntimePanelUtils.ScreenToPanel(options.root.panel, new Vector2(top.x, Screen.height - top.y));
205 return Mathf.Abs(anchor.y - topPanel.y);
206 }

212 public int AcquireLabelSlot(TNode node, HashSet<int> usedSlots)
213 {
214 // Prefer the node's old slot, then recycle any unused slot before growing the bounded pool.
215 if (options.nodeLabelSlots.TryGetValue(node.Id, out var slot) && (uint)slot < (uint)options.labels.Count &&
216 !usedSlots.Contains(slot)) return slot;
217 for (var i = 0; i < options.labels.Count; i++)
218 if (!usedSlots.Contains(i))
219 return i;
220 if (options.labels.Count >= options.NodeLabelCap || options.worldLabels == null) return -1;
221 options.CreateLabel();
222 return options.labels.Count - 1;
223 }

227 public void QueueLabel(TNode node)
228 {
229 if (node == null || !options.labelQueued.Add(node.Id)) return;
230 options.labelPlacementOrder.Add(node);

234 public void UpdateWorldLabels()
235 {
236 // Reuse pooled labels while recomputing which nodes fit the current camera view.
237 using var profile = GraphDiagnostics.NodeLabels.Auto();
238 options.labelPlacementOrder.Clear();
239 options.labelQueued.Clear();
240 QueueLabel(options.Selected());
241 QueueLabel(options.Hovered());
242
243 // Prioritize tasks, then last frame's labels, before admitting new ordinary candidates.
244 foreach (var id in options.taskVisuals.Keys)
245 if (options.Nodes().TryGetValue(id, out var taskNode))
246 QueueLabel(taskNode);
247 for (var pass = 0; pass < 2; pass++)
248 foreach (var node in options.labelCandidates)
249 {
250 if (node == options.Selected() || node == options.Hovered() ||
251 options.taskVisuals.ContainsKey(node.Id)) continue;
252 if (options.lastFrameLabelNodes.Contains(node.Id) != (pass == 0)) continue;
253 QueueLabel(node);
254 }
255
256 options.occupiedLabels.Clear();
257 var usedSlots = new HashSet<int>();
258 var placedNodes = new HashSet<string>();
259 var panelWidth = options.root.resolvedStyle.width;
260 var panelHeight = options.root.resolvedStyle.height;
261 if (float.IsNaN(panelWidth) || panelWidth < 1f) panelWidth = Screen.width;
262 if (float.IsNaN(panelHeight) || panelHeight < 1f) panelHeight = Screen.height;
263 var labelSettings = options.Settings();
264 var mouse = Mouse.current;
265 var mouseScreen = mouse != null ? mouse.position.ReadValue() : Vector2.zero;
266 var pointerOverGraph = false;
267 if (labelSettings.MouseProximityReveal && mouse != null && Application.isFocused &&
268 options.root.panel != null)
269 {
270 // Reveal on pointer proximity only inside graph space that is not blocked by shell controls.
271 var mousePanel =
272 RuntimePanelUtils.ScreenToPanel(options.root.panel,
273 new Vector2(mouseScreen.x, Screen.height - mouseScreen.y));
274 pointerOverGraph = options.Camera().pixelRect.Contains(mouseScreen)
275 && options.SafeArea().Contains(mousePanel)
276 && !GraphSceneInput.IsUiBlockingGraph(options.root, mouseScreen);
277 }
278
279 var labelPadding = Mathf.Max(0f, labelSettings.Padding);
280 var typeLine = labelSettings.TypeFontSize + 4f;
281 var nameLine = labelSettings.NameFontSize + 4f;
282 var metaLine = labelSettings.MetaFontSize + 4f;
283 var safeArea = options.SafeArea();
284 float leftEdge = safeArea.xMin,
285 rightEdge = safeArea.xMax,
286 topEdge = safeArea.yMin,
287 bottomEdge = safeArea.yMax;
288 if (rightEdge - leftEdge < labelSettings.MinWidth)
289 {
290 leftEdge = 8f;
291 rightEdge = Mathf.Max(leftEdge + labelSettings.MinWidth, panelWidth - 8f);
292 }
293
294 var cameraPosition = options.Camera().transform.position;
295 var cameraRotation = options.Camera().transform.rotation;
296 var cameraMoved = !labelCameraSampled
297 || (cameraPosition - labelCameraPosition).sqrMagnitude > 1e-8f
298 || Quaternion.Angle(cameraRotation, labelCameraRotation) > .02f;
299
300 // Camera motion bypasses smoothing so labels stay attached while the user navigates.
301 if (cameraMoved)
302 {
303 labelCameraPosition = cameraPosition;
304 labelCameraRotation = cameraRotation;
305 labelCameraSampled = true;
306 }
307
308 var smooth = 1f - Mathf.Exp(-14f * Time.unscaledDeltaTime);
309 foreach (var node in options.labelPlacementOrder)
310 {
311 var focused = node == options.Selected() || node == options.Hovered();
312 if (!node.Visible && !focused) continue;
313 var hasTask = options.taskVisuals.TryGetValue(node.Id, out var taskStatus);
314 var featured = focused || hasTask;
315 var screen = options.Camera().WorldToScreenPoint(node.Position);
316 if (screen.z <= 0f) continue;
317 if (screen.x < -48f || screen.x > Screen.width + 48f || screen.y < -48f ||
318 screen.y > Screen.height + 48f) continue;
319 var distance = Vector3.Distance(node.Position, cameraPosition);
320 var opacity = options.Opacity(distance, focused);
321 if (labelSettings.MouseProximityReveal)
322 {
323 // Fade each node's reveal independently before combining it with distance-based opacity.
324 var target = labelSettings.ProximityOpacity(
325 Vector2.Distance(new Vector2(screen.x, screen.y), mouseScreen), pointerOverGraph,
326 node == options.Selected());
327 revealOpacities.TryGetValue(node.Id, out var reveal);
328 reveal = labelSettings.RevealFadeSeconds <= 0f
329 ? target
330 : Mathf.MoveTowards(reveal, target, Time.unscaledDeltaTime / labelSettings.RevealFadeSeconds);
331 if (reveal > 0f) revealOpacities[node.Id] = reveal;
332 else revealOpacities.Remove(node.Id);
333 opacity *= reveal;
334 }
335
336 if (opacity <= 0f) continue;
337 var detailed = focused || distance <= Mathf.Min(options.DetailDistance(), options.MaxDistance());
338 var anchor =
339 RuntimePanelUtils.ScreenToPanel(options.root.panel,
340 new Vector2(screen.x, Screen.height - screen.y));
341 Vector2 smoothPoint;
342 if (cameraMoved || !options.labelSmoothPosition.TryGetValue(node.Id, out smoothPoint))
343 smoothPoint = anchor;
344 else smoothPoint = Vector2.Lerp(smoothPoint, anchor, smooth);
345 if ((smoothPoint - anchor).sqrMagnitude > 120f * 120f) smoothPoint = anchor;
346 var typeName = options.TypeCaption != null
347 ? options.TypeCaption(node, detailed, featured)
348 : detailed
349 ? options.TypeLabel(node).ToUpperInvariant()
350 : "";
351 options.Compose(node, hasTask ? taskStatus : null, out var nameText, out var metaText,
352 out var metaColor);
353 var jobSteps = detailed && options.HasSteps(node, hasTask);
354 if (jobSteps)
355 {
356 options.FillSteps(node, options.jobStepStates);
357 metaText = node.Pinned ? "PINNED" : "";
358 }
359
360 if (!detailed)
361 {
362 typeName = "";
363 metaText = "";
364 }
365
366 var nameFontSize = options.NameFontSize != null
367 ? options.NameFontSize(node, detailed, featured, labelSettings)
368 : labelSettings.NameFontSize;
369
370 // Measure wrapped text, metadata, and optional step chips before reserving collision space.
371 var extraWidth = options.ExtraLabelWidth?.Invoke(node, detailed, featured) ?? 0f;
372 var width = NodeLabelBlockWidth(labelSettings, typeName, nameText, metaText, nameFontSize, extraWidth);
373 if (jobSteps)
374 width = Mathf.Max(
375 width, Mathf.Min(labelSettings.MaxWidth, options.NodeStepRowWidth + labelPadding * 2f));
376 var column = rightEdge - leftEdge;
377 if (width > column) width = Mathf.Max(48f, column);
378 var textWidth = width - labelPadding * 2f - extraWidth;
379 var nameLineAtSize = nameFontSize + 4f;
380 var height = labelPadding * 2f + (typeName.Length > 0 ? typeLine : 0f)
381 + NodeNameLineCount(labelSettings, nameText, textWidth, nameFontSize) *
382 nameLineAtSize
383 + (metaText.Length > 0 ? metaLine : 0f) +
384 (jobSteps ? options.NodeStepRowHeight : 0f);
385 var nodeRadius = NodeRadiusPixels(node, smoothPoint);
386 var placed = TryNodeLabelRect(labelSettings, node.Id, smoothPoint, nodeRadius, width, height, featured,
387 leftEdge, rightEdge, topEdge, bottomEdge, out var rect, out var rowUsed);
388 if (!placed) continue;
389 var slot = AcquireLabelSlot(node, usedSlots);
390 if (slot < 0) break;
391
392 // Commit placement only after both a collision-free rectangle and a pooled label are available.
393 options.labelSmoothPosition[node.Id] = smoothPoint;
394 options.labelOffsets[node.Id] = rowUsed;
395 options.nodeLabelSlots[node.Id] = slot;
396 usedSlots.Add(slot);
397 placedNodes.Add(node.Id);
398 options.occupiedLabels.Add(rect);
399 var label = options.labels[slot];
400 label.Root.style.display = DisplayStyle.Flex;
401 label.Root.style.width = width;
402 label.Root.style.opacity = opacity;
403 ApplyLabelPanel(label.Root, labelSettings);
404 label.Root.style.left = Mathf.Round(rect.x);
405 label.Root.style.top = Mathf.Round(rect.y);
406 label.TypeHeader.style.display = typeName.Length > 0 ? DisplayStyle.Flex : DisplayStyle.None;
407 label.TypeHeader.style.fontSize = labelSettings.TypeFontSize;
408 if (label.TypeHeader.text != typeName) label.TypeHeader.text = typeName;
409 label.TypeHeader.style.color = options.TypeColor(node);
410 label.Name.style.fontSize = nameFontSize;
411 if (label.Name.text != nameText) label.Name.text = nameText;
412 label.Name.style.color =
413 node.Matches || focused ? new Color(.84f, .89f, .96f) : new Color(.39f, .44f, .51f);
414 var showMeta = metaText.Length > 0;
415 label.Meta.style.display = showMeta ? DisplayStyle.Flex : DisplayStyle.None;
416 if (showMeta)
417 {
418 label.Meta.style.fontSize = labelSettings.MetaFontSize;
419 if (label.Meta.text != metaText) label.Meta.text = metaText;
420 label.Meta.style.color = metaColor;
421 }
422
423 label.Steps.style.display = jobSteps ? DisplayStyle.Flex : DisplayStyle.None;
424 if (jobSteps) options.PaintSteps(label, options.jobStepStates);
425 options.FinalizeLabel?.Invoke(node, label, detailed, featured);
426 }
427
428 // Hide unused pool entries and discard smoothing, reveal, and row state for labels no longer placed.
429 for (var i = 0; i < options.labels.Count; i++)
430 if (!usedSlots.Contains(i))
431 options.labels[i].Root.style.display = DisplayStyle.None;
432 options.lastFrameLabelNodes.Clear();
433 foreach (var id in placedNodes) options.lastFrameLabelNodes.Add(id);
434 options.labelPrune.Clear();
435 foreach (var id in revealOpacities.Keys)
436 if (!options.labelQueued.Contains(id))
437 options.labelPrune.Add(id);
438 foreach (var id in options.labelPrune) revealOpacities.Remove(id);
439 if (!labelSettings.MouseProximityReveal) revealOpacities.Clear();
440 options.labelPrune.Clear();
441 foreach (var id in options.labelSmoothPosition.Keys)
442 if (!placedNodes.Contains(id))
443 options.labelPrune.Add(id);
444 for (var i = 0; i < options.labelPrune.Count; i++)
445 {
446 options.labelSmoothPosition.Remove(options.labelPrune[i]);
447 options.labelOffsets.Remove(options.labelPrune[i]);
448 options.nodeLabelSlots.Remove(options.labelPrune[i]);
449 }
450 }
451 }
