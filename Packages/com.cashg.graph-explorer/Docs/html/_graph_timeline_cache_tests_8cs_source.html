30 private static void Drive(IEnumerator routine)
31 {
32 // Drive nested load iterators while guaranteeing disposal of unfinished work after a failure.
33 var stack = new Stack<IEnumerator>();
34 stack.Push(routine);
35 try
36 {
37 while (stack.Count > 0)
38 {
39 var next = stack.Peek();
40 if (!next.MoveNext())
41 (stack.Pop() as IDisposable)?.Dispose();
42 else if (next.Current is IEnumerator child) stack.Push(child);
43 }
44 }
45 finally
46 {
47 while (stack.Count > 0) (stack.Pop() as IDisposable)?.Dispose();
48 }
49 }

64 [Test]
65 public void RangeIncludesEndpointsAndTiedEventsAndPortableStatesSurviveReload()
66 {
67 var provider = new Provider();
68 var track = Track(provider);
69 var document = Document();
70 var workspace = document.Timeline.Workspace;
71 workspace.SelectRange(Epoch.AddHours(1), Epoch); // reverse mouse drag
72 workspace.SetView(Epoch.AddMinutes(10), Epoch.AddMinutes(50));
73 Drive(workspace.CacheSelection(document, track, provider, CancellationToken.None));
74 Assert.That(provider.Loads, Is.EqualTo(3));
75 Assert.That(workspace.CachedEvents.Count, Is.EqualTo(3));
76 var json = GraphJson.Serialize(document);
77 var restored = GraphJson.Parse(json);
78
79 // Identical snapshots share one serialized state even when several descriptors reference it.
80 using (var parsed = JsonDocument.Parse(json))
81 {
82 Assert.That(
83 parsed.RootElement.GetProperty("timelineWorkspace").GetProperty("snapshots").EnumerateObject()
84 .Count(), Is.EqualTo(1), "Identical states must be stored once.");
85 }
86
87 var saved = restored.Timeline.Workspace;
88 Assert.That(saved.SelectionStartUtc, Is.EqualTo(Epoch));
89 Assert.That(saved.ViewStartUtc, Is.EqualTo(Epoch.AddMinutes(10)));
90 Assert.That(saved.ViewEndUtc, Is.EqualTo(Epoch.AddMinutes(50)));
91 var offline = new GraphTimelineEventTrack();
92
93 // Restore descriptors and load a cached event with no attached provider.
94 offline.RestoreCachedEvents(saved);
95 GraphSnapshot loaded = null;
96 Drive(offline.LoadEvent(restored, "tie", CancellationToken.None, value => loaded = value,
97 error => Assert.Fail(error.Message)));
98 Assert.That(loaded.Nodes[0].Id, Is.EqualTo("a"));
99 Assert.That(offline.Events.Count, Is.EqualTo(3));
100 var copy = Document();
101 copy.RestoreTimelineFrom(restored);
102 Assert.That(copy.Timeline.Workspace.CachedEvents.Count, Is.EqualTo(3));
103 }
