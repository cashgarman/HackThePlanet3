266 public IReadOnlyList<string> Validate(GraphDocument document)
267 {
268 if (document == null) throw new ArgumentNullException(nameof(document));
269 var issues = new List<string>();
270
271 // Check node types and declared fields independently from relation definitions.
272 foreach (var node in document.Nodes.Values)
273 {
274 if (!types.ContainsKey(node.TypeId)) issues.Add("Unregistered node type: " + node.TypeId);
275 ValidateFields(node.Id, node.Properties, issues);
276 }
277
278 foreach (var edge in document.Edges.Values)
279 {
280 if (!relations.ContainsKey(edge.TypeId)) issues.Add("Unregistered relation type: " + edge.TypeId);
281 ValidateFields(edge.Id, edge.Properties, issues);
282 }
283
284 return issues.AsReadOnly();
285 }
286
287 private void ValidateFields(string owner, IReadOnlyDictionary<string, GraphValue> properties,
288 List<string> issues)
289 {
290 // Undeclared and explicit-null fields are allowed; report only mismatches against known scalar types.
291 foreach (var pair in properties)
292 if (fields.TryGetValue(pair.Key, out var field) && pair.Value.Type != GraphValueType.Null &&
293 pair.Value.Type != field.Type)
294 issues.Add(owner + ": field " + pair.Key + " expects " + field.Type);
295 }
296 }
