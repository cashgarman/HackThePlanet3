19 public static string Serialize(GraphDocument document)
20 {
21 if (document == null) throw new ArgumentNullException(nameof(document));
22 using (var output = new MemoryStream())
23 {
24 using (var writer = new Utf8JsonWriter(output, new JsonWriterOptions { Indented = true }))
25 {
26 // Write records in ordinal order so equivalent documents produce stable JSON.
27
28 // The timeline is an optional extension: v1 readers continue to load the semantic snapshot.
29 writer.WriteStartObject();
30 writer.WriteNumber("schemaVersion", 1);
31 writer.WriteString("datasetId", document.DatasetId);
32 writer.WriteString("displayName", document.DisplayName);
33 writer.WriteStartArray("nodes");
34 foreach (var node in document.Nodes.Values.OrderBy(node => node.Id, StringComparer.Ordinal))
35 WriteNode(writer, node);
36 writer.WriteEndArray();
37 writer.WriteStartArray("edges");
38 foreach (var edge in document.Edges.Values.OrderBy(edge => edge.Id, StringComparer.Ordinal))
39 WriteEdge(writer, edge);
40 writer.WriteEndArray();
41 WriteTimeline(writer, document.Timeline);
42 WriteWorkspace(writer, document.Timeline.Workspace);
43 writer.WriteEndObject();
44 }
45
46 return Encoding.UTF8.GetString(output.ToArray());
47 }
48 }

58 private static void WriteProperties(Utf8JsonWriter writer, IReadOnlyDictionary<string, GraphValue> properties)
59 {
60 // Write native JSON scalar values in ordinal key order, retaining explicit null properties.
61 writer.WriteStartObject("properties");
62 foreach (var pair in properties.OrderBy(pair => pair.Key, StringComparer.Ordinal))
63 switch (pair.Value.Type)
64 {
65 case GraphValueType.Text: writer.WriteString(pair.Key, pair.Value.Text); break;
66 case GraphValueType.Number: writer.WriteNumber(pair.Key, pair.Value.Number); break;
67 case GraphValueType.Boolean: writer.WriteBoolean(pair.Key, pair.Value.Boolean); break;
68 default: writer.WriteNull(pair.Key); break;
69 }
70
71 writer.WriteEndObject();
72 }

100 private static void WriteTimeline(Utf8JsonWriter writer, GraphTimeline timeline)
101 {
102 writer.WriteStartObject("timeline");
103 writer.WriteBoolean("positionRecordingEnabled", timeline.PositionRecordingEnabled);
104 writer.WriteNumber("positionSampleIntervalSeconds", timeline.PositionSampleIntervalSeconds);
105 writer.WriteString("currentTimeUtc", timeline.CurrentTimeUtc.ToString("O"));
106
107 // Persist the reconstruction baseline separately from timestamped semantic and motion tracks.
108 if (timeline.HasBaseline)
109 {
110 writer.WriteString("baselineTimeUtc", timeline.BaselineTimeUtc.ToString("O"));
111 writer.WriteNumber("cursor", timeline.Cursor);
112 writer.WriteStartObject("baseline");
113 writer.WriteString("datasetId", timeline.Baseline.DatasetId);
114 writer.WriteString("displayName", timeline.Baseline.DisplayName);
115 writer.WriteStartArray("nodes");
116 foreach (var node in timeline.Baseline.Nodes.OrderBy(item => item.Id, StringComparer.Ordinal))
117 WriteNode(writer, node);
118 writer.WriteEndArray();
119 writer.WriteStartArray("edges");
120 foreach (var edge in timeline.Baseline.Edges.OrderBy(item => item.Id, StringComparer.Ordinal))
121 WriteEdge(writer, edge);
122 writer.WriteEndArray();
123 writer.WriteEndObject();
124 }
125
126 // Position frames store canonical node coordinates independently from semantic edits.
127 writer.WriteStartArray("positionFrames");
128 foreach (var frame in timeline.PositionFrames)
129 {
130 writer.WriteStartObject();
131 writer.WriteString("timestampUtc", frame.TimestampUtc.ToString("O"));
132 writer.WriteStartArray("positions");
133 foreach (var pair in frame.Positions.OrderBy(item => item.Key, StringComparer.Ordinal))
134 {
135 writer.WriteStartObject();
136 writer.WriteString("nodeId", pair.Key);
137 writer.WriteNumber("x", pair.Value.x);
138 writer.WriteNumber("y", pair.Value.y);
139 writer.WriteNumber("z", pair.Value.z);
140 writer.WriteEndObject();
141 }
142
143 writer.WriteEndArray();
144 writer.WriteEndObject();
145 }
146
147 writer.WriteEndArray();
148 writer.WriteStartArray("entries");
149
150 // Keep both record versions so each edit retains its full before/after evidence.
151 foreach (var entry in timeline.Entries)
152 {
153 writer.WriteStartObject();
154 writer.WriteNumber("sequence", entry.Sequence);
155 writer.WriteString("timestampUtc", entry.TimestampUtc.ToString("O"));
156 writer.WriteString("displayNameBefore", entry.Change.DisplayNameBefore);
157 writer.WriteString("displayNameAfter", entry.Change.DisplayNameAfter);
158 WriteRecordArray(writer, "nodesBefore", entry.Change.NodesBefore, WriteNode);
159 WriteRecordArray(writer, "nodesAfter", entry.Change.NodesAfter, WriteNode);
160 WriteRecordArray(writer, "edgesBefore", entry.Change.EdgesBefore, WriteEdge);
161 WriteRecordArray(writer, "edgesAfter", entry.Change.EdgesAfter, WriteEdge);
162 writer.WriteEndObject();
163 }
164
165 writer.WriteEndArray();
166 writer.WriteEndObject();
167 }

180 public static GraphDocument Parse(string json)
181 {
182 using (var parsed = JsonDocument.Parse(json))
183 {
184 var root = parsed.RootElement;
185
186 // Reject unsupported formats before constructing a document from the semantic snapshot.
187 var version = root.GetProperty("schemaVersion").GetInt32();
188 if (version != 1) throw new FormatException("Unsupported graph document version.");
189 var document = new GraphDocument(root.GetProperty("datasetId").GetString(),
190 root.GetProperty("displayName").GetString());
191 var batch = new GraphBatch();
192 foreach (var node in root.GetProperty("nodes").EnumerateArray()) batch.Nodes.Add(ReadNode(node));
193 foreach (var edge in root.GetProperty("edges").EnumerateArray()) batch.Edges.Add(ReadEdge(edge));
194 document.Apply(batch);
195
196 // Older documents may omit history; restore it only when a baseline is present.
197 if (root.TryGetProperty("timeline", out var timeline) &&
198 timeline.TryGetProperty("baseline", out var baseline) && baseline.ValueKind == JsonValueKind.Object)
199 {
200 var baselineNodes = baseline.GetProperty("nodes").EnumerateArray().Select(ReadNode).ToArray();
201 var baselineEdges = baseline.GetProperty("edges").EnumerateArray().Select(ReadEdge).ToArray();
202 var baselineSnapshot = new GraphSnapshot(baseline.GetProperty("datasetId").GetString(),
203 baseline.GetProperty("displayName").GetString(),
204 baselineNodes, baselineEdges);
205
206 // Deserialize ordered semantic changes before restoring the saved cursor.
207 var entries = new List<GraphTimelineEntry>();
208 foreach (var item in timeline.GetProperty("entries").EnumerateArray())
209 {
210 var change = new GraphTimelineChange(
211 item.GetProperty("nodesBefore").EnumerateArray().Select(ReadNode),
212 item.GetProperty("nodesAfter").EnumerateArray().Select(ReadNode),
213 item.GetProperty("edgesBefore").EnumerateArray().Select(ReadEdge),
214 item.GetProperty("edgesAfter").EnumerateArray().Select(ReadEdge),
215 item.GetProperty("displayNameBefore").GetString(),
216 item.GetProperty("displayNameAfter").GetString());
217 entries.Add(new GraphTimelineEntry(item.GetProperty("sequence").GetInt64(),
218 DateTimeOffset.Parse(
219 item.GetProperty("timestampUtc").GetString(),
220 CultureInfo.InvariantCulture, DateTimeStyles.None),
221 change));
222 }
223
224 // Load optional motion samples and recording settings with backward-compatible defaults.
225 var positionFrames = new List<GraphPositionFrame>();
226 if (timeline.TryGetProperty("positionFrames", out var savedFrames))
227 foreach (var item in savedFrames.EnumerateArray())
228 positionFrames.Add(new GraphPositionFrame(
229 DateTimeOffset.Parse(item.GetProperty("timestampUtc").GetString(),
230 CultureInfo.InvariantCulture,
231 DateTimeStyles.None),
232 item.GetProperty("positions").EnumerateArray()
233 .Select(position => new KeyValuePair<string, Vector3>(
234 position.GetProperty("nodeId").GetString(),
235 new Vector3(
236 position.GetProperty("x").GetSingle(),
237 position.GetProperty("y").GetSingle(),
238 position.GetProperty("z").GetSingle())))));
239 var baselineTime = DateTimeOffset.Parse(timeline.GetProperty("baselineTimeUtc").GetString(),
240 CultureInfo.InvariantCulture, DateTimeStyles.None);
241 var cursor = timeline.TryGetProperty("cursor", out var savedCursor)
242 ? savedCursor.GetInt32()
243 : entries.Count - 1;
244 var recording = timeline.TryGetProperty("positionRecordingEnabled", out var savedRecording) &&
245 savedRecording.GetBoolean();
246 var interval = timeline.TryGetProperty("positionSampleIntervalSeconds", out var savedInterval)
247 ? savedInterval.GetSingle()
248 : 1f;
249 var currentTime = timeline.TryGetProperty("currentTimeUtc", out var savedTime)
250 ? DateTimeOffset.Parse(savedTime.GetString(), CultureInfo.InvariantCulture, DateTimeStyles.None)
251 : (DateTimeOffset?)null;
252 document.RestoreTimeline(baselineSnapshot, baselineTime, entries, cursor, positionFrames, recording,
253 interval, currentTime);
254 }
255
256 if (root.TryGetProperty("timelineWorkspace", out var workspace)) ReadWorkspace(workspace, document);
257 return document;
258 }
259 }

281 private static void WriteWorkspace(Utf8JsonWriter writer, GraphTimelineWorkspace workspace)
282 {
283 writer.WriteStartObject("timelineWorkspace");
284 writer.WriteNumber("version", 1);
285 if (workspace.HasSelection)
286 {
287 writer.WriteString("selectionStartUtc", workspace.SelectionStartUtc.Value.ToString("O"));
288 writer.WriteString("selectionEndUtc", workspace.SelectionEndUtc.Value.ToString("O"));
289 }
290
291 if (workspace.ViewStartUtc.HasValue)
292 {
293 writer.WriteString("viewStartUtc", workspace.ViewStartUtc.Value.ToString("O"));
294 writer.WriteString("viewEndUtc", workspace.ViewEndUtc.Value.ToString("O"));
295 }
296
297 var snapshots = new Dictionary<string, string>(StringComparer.Ordinal);
298
299 // Reference cached snapshots by content hash so events sharing a state serialize it once.
300 var identities = new Dictionary<GraphSnapshot, string>();
301 writer.WriteStartArray("events");
302 foreach (var item in workspace.CachedEvents.OrderBy(item => item.Context, StringComparer.Ordinal)
303 .ThenBy(item => item.Event.Id, StringComparer.Ordinal))
304 {
305 if (!identities.TryGetValue(item.Snapshot, out var identity))
306 {
307 using var output = new MemoryStream();
308 using (var snapshotWriter = new Utf8JsonWriter(output))
309 {
310 WriteSnapshot(snapshotWriter, item.Snapshot);
311 }
312
313 var bytes = output.ToArray();
314 using var sha = SHA256.Create();
315 identity = BitConverter.ToString(sha.ComputeHash(bytes)).Replace("-", "").ToLowerInvariant();
316 snapshots[identity] = Encoding.UTF8.GetString(bytes);
317 identities[item.Snapshot] = identity;
318 }
319
320 writer.WriteStartObject();
321 writer.WriteString("id", item.Event.Id);
322 writer.WriteString("timestampUtc", item.Event.TimestampUtc.ToString("O"));
323 writer.WriteString("label", item.Event.Label);
324 writer.WriteString("details", item.Event.Details);
325 writer.WriteString("tooltip", item.Event.Tooltip);
326 writer.WriteString("context", item.Context);
327 writer.WriteString("snapshot", identity);
328 writer.WriteEndObject();
329 }
330
331 writer.WriteEndArray();
332 writer.WriteStartObject("snapshots");
333
334 // Emit the deduplicated snapshot table after writing event references.
335 foreach (var pair in snapshots.OrderBy(pair => pair.Key, StringComparer.Ordinal))
336 {
337 writer.WritePropertyName(pair.Key);
338 using var parsed = JsonDocument.Parse(pair.Value);
339 parsed.RootElement.WriteTo(writer);
340 }
341
342 writer.WriteEndObject();
343 writer.WriteEndObject();
344 }

346 private static void ReadWorkspace(JsonElement value, GraphDocument document)
347 {
348 if (value.GetProperty("version").GetInt32() != 1)
349 throw new FormatException("Unsupported timeline workspace version.");
350
351 DateTimeOffset Date(string key)
352 {
353 return DateTimeOffset.Parse(value.GetProperty(key).GetString(), CultureInfo.InvariantCulture,
354 DateTimeStyles.None);
355 }
356
357 var workspace = document.Timeline.Workspace;
358
359 // Restore portable selection and viewport bounds before reconnecting cached event snapshots.
360 if (value.TryGetProperty("selectionStartUtc", out _))
361 workspace.SelectRange(Date("selectionStartUtc"), Date("selectionEndUtc"));
362 if (value.TryGetProperty("viewStartUtc", out _))
363 workspace.SetView(Date("viewStartUtc"), Date("viewEndUtc"));
364 var snapshots = value.GetProperty("snapshots").EnumerateObject()
365 .ToDictionary(item => item.Name, item => ReadSnapshot(item.Value),
366 StringComparer.Ordinal);
367 foreach (var snapshot in snapshots.Values)
368 {
369 // Validate dataset ownership and graph consistency before admitting a cached state.
370 if (snapshot.DatasetId != document.DatasetId)
371 throw new FormatException("Cached timeline state belongs to another dataset.");
372 var validation = new GraphDocument(document.DatasetId);
373 validation.ReplaceSnapshot(snapshot);
374 }
375
376 // Resolve each event's snapshot reference against the validated table.
377 foreach (var item in value.GetProperty("events").EnumerateArray())
378 workspace.Store(new GraphTimelineEvent(item.GetProperty("id").GetString(),
379 DateTimeOffset.Parse(
380 item.GetProperty("timestampUtc").GetString(),
381 CultureInfo.InvariantCulture, DateTimeStyles.None),
382 item.GetProperty("label").GetString(),
383 item.GetProperty("details").GetString(),
384 item.GetProperty("tooltip").GetString()),
385 item.GetProperty("context").GetString(),
386 snapshots[item.GetProperty("snapshot").GetString()]);
387 }

413 private static Dictionary<string, GraphValue> ReadProperties(JsonElement owner)
414 {
415 var values = new Dictionary<string, GraphValue>(StringComparer.Ordinal);
416 if (!owner.TryGetProperty("properties", out var properties)) return values;
417
418 // Preserve scalar types and reject nested metadata rather than silently discarding it.
419 foreach (var property in properties.EnumerateObject())
420 switch (property.Value.ValueKind)
421 {
422 case JsonValueKind.String:
423 values.Add(property.Name, GraphValue.From(property.Value.GetString())); break;
424 case JsonValueKind.Number:
425 values.Add(property.Name, GraphValue.From(property.Value.GetDouble())); break;
426 case JsonValueKind.True: values.Add(property.Name, GraphValue.From(true)); break;
427 case JsonValueKind.False: values.Add(property.Name, GraphValue.From(false)); break;
428 case JsonValueKind.Null: values.Add(property.Name, GraphValue.Null); break;
429 default: throw new FormatException("Graph metadata values must be scalar: " + property.Name);
430 }
431
432 return values;
433 }
