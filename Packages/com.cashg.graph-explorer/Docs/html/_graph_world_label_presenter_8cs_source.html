44 public static float LabelInk(string text)
45 {
46 // Estimate text width from narrow, normal, and wide glyph classes without allocating layout elements.
47 if (string.IsNullOrEmpty(text)) return 0f;
48 var width = 0f;
49 for (var i = 0; i < text.Length; i++)
50 {
51 var c = text[i];
52 if (c == ' ' || c == '.' || c == ',' || c == '\'' || c == 'i' || c == 'l' || c == 'I' || c == 'j' ||
53 c == 't') width += 3.8f;
54 else if (char.IsUpper(c) || c == 'm' || c == 'w') width += 8.4f;
55 else width += 6.5f;
56 }
57
58 return width;
59 }

78 public static float NodeLabelBlockWidth(GraphNodeLabelSettings s, string typeName, string name, string meta,
79 float nameFontSize = 0f, float extraWidth = 0f)
80 {
81 if (nameFontSize <= 0f) nameFontSize = s.NameFontSize;
82 var text = Mathf.Max(LabelInk(name) * LabelScale(nameFontSize),
83 LabelInk(meta) * LabelScale(s.MetaFontSize));
84 var typeWidth = LabelInk(typeName) * LabelScale(s.TypeFontSize) +
85 Mathf.Max(0, (typeName == null ? 0 : typeName.Length) - 1);
86 return Mathf.Clamp(Mathf.Max(text, typeWidth) + s.Padding * 2f + extraWidth, s.MinWidth, s.MaxWidth);
87 }

89 // The full name always stays visible: it wraps word by word inside the panel instead of being clipped.
96 public static int NodeNameLineCount(GraphNodeLabelSettings s, string name, float available,
97 float nameFontSize = 0f)
98 {
99 if (string.IsNullOrEmpty(name)) return 1;
100 if (nameFontSize <= 0f) nameFontSize = s.NameFontSize;
101 var scale = LabelScale(nameFontSize);
102 var space = LabelInk(" ") * scale;
103 var limit = Mathf.Max(1f, available);
104 var lines = 1;
105 var column = 0f;
106 foreach (var word in name.Split(' '))
107 {
108 // Wrap at word boundaries while leaving a single oversized word on its own line.
109 var wordWidth = LabelInk(word) * scale;
110 if (column > 0f && column + space + wordWidth > limit)
111 {
112 lines++;
113 column = wordWidth;
114 }
115 else
116 {
117 column += (column > 0f ? space : 0f) + wordWidth;
118 }
119 }
120
121 return lines;
122 }
