66 public static bool RegisterPrimaryClick(
67 ref PrimaryClickState state,
68 IGraphNode node,
69 float time,
70 float doubleClickSeconds = DefaultDoubleClickSeconds)
71 {
72 if (node == null)
73 {
74 // An empty-space click breaks the same-node double-click sequence.
75 state = default;
76 return false;
77 }
78
79 var doubleClick = node.Id == state.LastNodeId && time - state.LastTime < doubleClickSeconds;
80
81 // Consume a successful pair so a third click starts a fresh sequence.
82 if (doubleClick)
83 {
84 state = default;
85 }
86 else
87 {
88 state.LastNodeId = node.Id;
89 state.LastTime = time;
90 }
91
92 return doubleClick;
93 }

103 public static void HandlePrimaryClick(
104 HoverSample sample,
105 Mouse mouse,
106 ref PrimaryClickState clickState,
107 GraphNavigation navigation,
108 GraphViewState view,
109 Action<IGraphNode> select,
110 float doubleClickSeconds = DefaultDoubleClickSeconds)
111 {
112 if (sample.Blocked || mouse == null || !mouse.leftButton.wasPressedThisFrame || select == null) return;
113
114 // Select immediately, using a repeated click on the same node to begin camera tracking.
115 if (sample.Node != null)
116 {
117 select(sample.Node);
118 if (RegisterPrimaryClick(ref clickState, sample.Node, Time.unscaledTime, doubleClickSeconds))
119 navigation?.Focus(sample.Node);
120 return;
121 }
122
123 clickState = default;
124
125 // Empty graph space clears both selection and any camera tracking lock.
126 ReleaseCameraFocus(view, navigation);
127 select(null);
128 }
