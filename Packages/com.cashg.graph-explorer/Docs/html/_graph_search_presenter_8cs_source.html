118 virtualizationMethod = CollectionVirtualizationMethod.FixedHeight

157 RefreshNodeSearch();

159 nodeSearchField.RegisterCallback<FocusInEvent>(_ =>
160 {
161 options.SetOpen(true);
162 RefreshNodeSearch();
163 });
164 RegisterRoot<KeyDownEvent>(e =>
165 {
166 // Restrict result navigation shortcuts to focus inside the search field or popup.
167 var target = e.target as VisualElement;
168 if (target == null || !(target == nodeSearchField || target == nodeSearchPopup ||
169 nodeSearchField.Contains(target) || nodeSearchPopup.Contains(target))) return;
170 if (e.keyCode != KeyCode.DownArrow && e.keyCode != KeyCode.UpArrow && e.keyCode != KeyCode.Return &&
171 e.keyCode != KeyCode.KeypadEnter) return;
172 if (!options.IsOpen())
173 {
174 options.SetOpen(true);
175 RefreshNodeSearch();
176 }
177
178 if (e.keyCode == KeyCode.Return || e.keyCode == KeyCode.KeypadEnter)
179 {
180 ActivateNodeSearch();
181 }
182 else if (nodeSearchResults.Count > 0)
183 {
184 var index = Mathf.Clamp(nodeSearchList.selectedIndex + (e.keyCode == KeyCode.DownArrow ? 1 : -1), 0,
185 nodeSearchResults.Count - 1);
186 nodeSearchList.selectedIndex = index;
187 nodeSearchList.ScrollToItem(index);
188 }
189
190 e.StopImmediatePropagation();
191 e.PreventDefault();
192 }, TrickleDown.TrickleDown);
193 RegisterRoot<PointerDownEvent>(e =>
194 {
195 var target = e.target as VisualElement;
196 if (target != null && !nodeSearchPopup.Contains(target) && !shellSearchHost.Contains(target))
197 CloseNodeSearch();
198 }, TrickleDown.TrickleDown);
199 RegisterRoot<FocusInEvent>(e =>
200 {
201 var target = e.target as VisualElement;
202 if (target != null && !nodeSearchPopup.Contains(target) && !shellSearchHost.Contains(target))
203 CloseNodeSearch();
204 });
205 RegisterRoot<GeometryChangedEvent>(_ => PositionNodeSearch());
206 RegisterOn<GeometryChangedEvent>(shellSearchHost, _ => PositionNodeSearch());

210 public void PositionNodeSearch()
211 {
212 // Anchor the popup under the search host and cap its list height to the available root space.
213 if (nodeSearchPopup == null || !options.IsOpen()) return;
214 var bounds = shellSearchHost.worldBound;
215 var point = root.WorldToLocal(new Vector2(bounds.xMin, bounds.yMax));
216 nodeSearchPopup.style.left = point.x;
217 nodeSearchPopup.style.top = point.y + 6;
218 nodeSearchPopup.style.width = bounds.width;
219 nodeSearchList.style.height = Mathf.Min(384, Mathf.Max(0, root.resolvedStyle.height - point.y - 84),
220 nodeSearchResults.Count * 48);
221 }

224 public void RefreshNodeSearch()
225 {
226 // Rebuild results from current graph visibility and query state.
227 if (nodeSearchPopup == null || !options.IsOpen()) return;
228 var selectedId = nodeSearchList.selectedItem is TNode selected ? selected.Id : null;
229 var query = (nodeSearchField.value ?? "").Trim().ToLowerInvariant();
230 nodeSearchResults.Clear();
231 if (query.Length > 0 && options.Nodes() != null)
232 nodeSearchResults.AddRange(
233 options.Nodes().Select(n => new { Node = n, Score = NodeSearchScore(query, n) })
234 .Where(hit => hit.Score >= 0).OrderByDescending(hit => hit.Node.Favorite)
235 .ThenByDescending(hit => hit.Score)
236 .ThenBy(hit => hit.Node.Label, StringComparer.OrdinalIgnoreCase)
237 .ThenBy(hit => hit.Node.Id, StringComparer.Ordinal)
238 .Select(hit => hit.Node));
239
240 // Preserve selection by node ID after score ordering changes, falling back to the first result.
241 nodeSearchList.Rebuild();
242 var index = selectedId == null ? -1 : nodeSearchResults.FindIndex(n => n.Id == selectedId);
243 nodeSearchList.SetSelectionWithoutNotify(nodeSearchResults.Count == 0
244 ? Array.Empty<int>()
245 : new[] { Mathf.Max(0, index) });
246 nodeSearchStatus.text = query.Length == 0 ? "Search all loaded nodes · ↑ ↓ browse · Enter focus" :
247 nodeSearchResults.Count == 0 ? "No matching nodes" :
248 nodeSearchResults.Count + " nodes · ↑ ↓ browse · Enter focus";
249 nodeSearchPopup.style.display = DisplayStyle.Flex;
250 nodeSearchPopup.BringToFront();
251 PositionNodeSearch();
252 }

279 public static int NodeSearchScore(string query, TNode node)
280 {
281 // Rank literal name matches ahead of token matches across name, type, and opaque identity.
282 var name = (node.Label ?? "").ToLowerInvariant();
283 if (name == query) return 10000;
284 if (name.StartsWith(query, StringComparison.Ordinal)) return 9000 - Mathf.Min(name.Length, 1000);
285 if (name.Contains(query)) return 7000 - Mathf.Min(name.Length, 1000);
286 var text = name + " " + node.Record.TypeId.ToLowerInvariant() + " " + (node.Id ?? "").ToLowerInvariant();
287 var score = 0;
288 foreach (var token in query.Split(new[] { ' ' }, StringSplitOptions.RemoveEmptyEntries))
289 {
290 if (text.Contains(token))
291 {
292 score += 100;
293 continue;
294 }
295
296 // Match real words with at most one edit; scattered letters are too permissive.
297 // Short queries require literal matches to avoid unrelated one-letter variations.
298 if (token.Length >= 4 && text.Split(' ', '-', '_', '/', ':').Any(word => OneSearchTypo(token, word)))
299 {
300 score += 20;
301 continue;
302 }
303
304 return -1;
305 }
306
307 return score;
308 }

310 private static bool OneSearchTypo(string a, string b)
311 {
312 // Skip the common prefix, then allow one transposition, insertion, deletion, or substitution.
313 if (Math.Abs(a.Length - b.Length) > 1) return false;
314 int i = 0, j = 0;
315 while (i < a.Length && j < b.Length && a[i] == b[j])
316 {
317 i++;
318 j++;
319 }
320
321 if (i == a.Length || j == b.Length) return true;
322 if (a.Length == b.Length && i + 1 < a.Length && a[i] == b[j + 1] && a[i + 1] == b[j] &&
323 a.Substring(i + 2) == b.Substring(j + 2)) return true;
324 if (a.Length >= b.Length) i++;
325 if (b.Length >= a.Length) j++;
326 while (i < a.Length && j < b.Length && a[i] == b[j])
327 {
328 i++;
329 j++;
330 }
331
332 return i == a.Length && j == b.Length;
333 }
