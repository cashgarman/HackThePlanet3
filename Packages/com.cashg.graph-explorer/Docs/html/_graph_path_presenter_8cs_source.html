14 public static IDisposable Bind(GraphPathState state, GraphRenderer renderer)
15 {
16 if (state == null) throw new ArgumentNullException(nameof(state));
17 if (renderer == null) throw new ArgumentNullException(nameof(renderer));
18
19 void Apply(GraphPathState _)
20 {
21 renderer.ActivePath = state.Result;
22 }
23
24 state.Changed += Apply;
25
26 // Apply an existing result immediately, then keep highlighting synchronized until the binding is disposed.
27 renderer.ActivePath = state.Result;
28 return new Binding(state, renderer, Apply);
29 }

31 private sealed class Binding : IDisposable
32 {

34 private GraphRenderer renderer;
35 private GraphPathState state;

37 public Binding(GraphPathState state, GraphRenderer renderer, Action<GraphPathState> apply)
38 {
39 this.state = state;
40 this.renderer = renderer;
41 this.apply = apply;
42 }

44 public void Dispose()
45 {
46 // Unsubscribe and clear highlighting so a dismissed panel cannot leave a stale route rendered.
47 if (state != null) state.Changed -= apply;
48 if (renderer != null) renderer.ActivePath = null;
49 state = null;
50 renderer = null;
51 }

117 private TextField EndpointRow(VisualElement parent, string label, GraphPathEndpoint endpoint,
118 out VisualElement choices)
119 {
120 // Support graph picking through the host callback, falling back to the shared pending-pick state.
121 parent.Add(GraphUi.Label(label, "eyebrow"));

123 parent.Add(row);

126 GraphUi.HideFieldLabel(field);
127 field.AddToClassList("nolabel");
128 row.Add(field);
129 row.Add(GraphUi.Button("Pick", () =>
130 {
131 if (pick != null) pick(endpoint);
132 else state.BeginPick(endpoint);
133 }, "path-pick"));
134 choices = GraphUi.Element("path-choices");
135 parent.Add(choices);
136 return field;

139 private void PopulateChoices(VisualElement choices, GraphPathEndpoint endpoint)
140 {
141 // Match labels or opaque IDs and bound the deterministic suggestions to eight nodes.
142 choices.Clear();
143 var query = endpoint == GraphPathEndpoint.Start ? startSearch.value : endSearch.value;
144 if (string.IsNullOrWhiteSpace(query)) return;
145 foreach (var node in scene.Nodes.Values
146 .Where(n =>
147 n.Label.IndexOf(query.Trim(), StringComparison.OrdinalIgnoreCase) >=
148 0 || n.Id.IndexOf(query.Trim(), StringComparison.OrdinalIgnoreCase) >=
149 0)
150 .OrderBy(n => n.Label, StringComparer.OrdinalIgnoreCase)
151 .ThenBy(n => n.Id, StringComparer.Ordinal).Take(8))
152 choices.Add(GraphUi.Button(nodeLabel(node), () => Select(endpoint, node), "link"));
153 }

174 private void Refresh(GraphPathState changed)
175 {
176 // Synchronize endpoint fields and choice lists with the current path state.
177 startId = changed.StartId;
178 endId = changed.EndId;
179 startSearch.SetValueWithoutNotify(startId != null && scene.Nodes.TryGetValue(startId, out var startNode)
180 ? nodeLabel(startNode)
181 : startId ?? "");
182 endSearch.SetValueWithoutNotify(endId != null && scene.Nodes.TryGetValue(endId, out var endNode)
183 ? nodeLabel(endNode)
184 : endId ?? "");
185 rows.Clear();
186 if (!changed.IsActive)
187 {
188 summary.text = changed.PendingPick.HasValue
189 ? "Click a graph node to set the " + changed.PendingPick.Value.ToString().ToLowerInvariant() +
190 " node · Esc cancels."
191 : "Choose start and end nodes.";
192 return;
193 }
194
195 if (changed.Result == null || !changed.Result.IsReachable)
196 {
197 summary.text = "No directed path exists between these nodes.";
198 return;
199 }
200
201 // Show the ordered nodes with each outgoing relationship labeled from the scene schema.
202 summary.text = changed.Result.Jumps + (changed.Result.Jumps == 1 ? " jump" : " jumps");
203 for (var i = 0; i < changed.Result.NodeIds.Count; i++)
204 {
205 var id = changed.Result.NodeIds[i];
206 if (!scene.Nodes.TryGetValue(id, out var node)) continue;
207 var row = GraphUi.Element("connection");
208 rows.Add(row);
209 row.Add(GraphUi.Label((i + 1).ToString(), "connection-kind"));
210 row.Add(GraphUi.Button(nodeLabel(node), () => navigate?.Invoke(node), "link"));
211 if (i < changed.Result.EdgeIds.Count &&
212 scene.Edges.TryGetValue(changed.Result.EdgeIds[i], out var edge))
213 row.Add(GraphUi.Label(
214 "→ " + (scene.Schema.Relations.TryGetValue(edge.Relation, out var relation)
215 ? relation.Label
216 : edge.Relation), "connection-relation"));
217 }
218 }
