59def brace_delta(line: str) -> int:
60 return line.count("{") - line.count("}")

63def parse_file(path: Path):
64 text = path.read_text(encoding="utf-8")
65 ns = "Cashg.GraphExplorer"
66 if "namespace Cashg.GraphExplorer.Codex" in text:
67 ns = "Cashg.GraphExplorer.Codex"
68 elif "namespace Cashg.GraphExplorer.Editor" in text:
69 ns = "Cashg.GraphExplorer.Editor"
70
71 lines = text.splitlines()
72 depth = 0
73 in_namespace = False
74 types = []
75 pending_summary = []
76
77 type_re = re.compile(
78 r"^\s*public\s+(static\s+)?(sealed\s+)?(readonly\s+)?(class|interface|enum|struct)\s+(\w+)"
79 )
80 member_re = re.compile(
81 r"^\s*public\s+(static\s+)?(.+?)\s+(\w+)\s*(\‍([^)]*\‍))?\s*(?:\{[^}]*\}|=>|[;=])"
82 )
83 iface_member_re = re.compile(r"^\s*([\w<>,\‍[\‍]\s.?]+)\s+(\w+)\s*\‍([^)]*\‍)\s*;")
84
85 for line in lines:
86 stripped = line.strip()
87 if not in_namespace:
88 if stripped.startswith("namespace "):
89 in_namespace = True
90 continue
91 if stripped.startswith("///"):
92 pending_summary.append(stripped[3:].strip().lstrip("/").strip())
93 continue
94
95 m = type_re.match(line)
96 rel_depth = depth # depth before this line's braces
97 if m and rel_depth == 1:
98 summary = " ".join(pending_summary).replace("<summary>", "").replace("</summary>", "").strip()
99 pending_summary = []
100 types.append(
101 {
102 "kind": m.group(4),
103 "name": m.group(5),
104 "summary": summary,
105 "members": [],
106 }
107 )
108 depth += brace_delta(line)
109 continue
110
111 if types and rel_depth == 2:
112 if types[-1]["kind"] == "interface":
113 im = iface_member_re.match(line)
114 if im:
115 types[-1]["members"].append(f"{im.group(1).strip()} {im.group(2)}(...)")
116 else:
117 mm = member_re.match(line)
118 if mm and " class " not in line and " interface " not in line:
119 sig = stripped.rstrip("{").strip()
120 if not sig.endswith(";") and "{" in sig:
121 sig = sig.split("{")[0].strip()
122 types[-1]["members"].append(sig)
123
124 depth += brace_delta(line)
125 if depth == 0:
126 pending_summary = []
127
128 return ns, types
