25 public static GraphPathResult Find(IGraphScene scene, string startId, string endId,
26 Func<IGraphEdge, bool> edgeAllowed = null)
27 {
28 if (scene == null) throw new ArgumentNullException(nameof(scene));
29 if (string.IsNullOrEmpty(startId) || string.IsNullOrEmpty(endId) ||
30 !scene.Nodes.ContainsKey(startId) || !scene.Nodes.ContainsKey(endId))
31 return GraphPathResult.Unreachable(startId, endId);
32 if (startId == endId) return new GraphPathResult(startId, endId, new[] { startId }, Array.Empty<string>());
33
34 // Build deterministic outgoing adjacency from the permitted directed edges.
35 var outgoing = new Dictionary<string, List<IGraphEdge>>(StringComparer.Ordinal);
36 foreach (var edge in scene.Edges.Values)
37 {
38 if (edgeAllowed != null && !edgeAllowed(edge)) continue;
39 if (!outgoing.TryGetValue(edge.From, out var list))
40 outgoing.Add(edge.From, list = new List<IGraphEdge>());
41 list.Add(edge);
42 }
43
44 foreach (var list in outgoing.Values)
45 list.Sort((a, b) =>
46 {
47 var c = string.CompareOrdinal(a.To, b.To);
48 return c != 0 ? c : string.CompareOrdinal(a.Id, b.Id);
49 });
50
51 var previousNode = new Dictionary<string, string>(StringComparer.Ordinal);
52 var previousEdge = new Dictionary<string, string>(StringComparer.Ordinal);
53 var visited = new HashSet<string>(StringComparer.Ordinal) { startId };
54 var queue = new Queue<string>();
55 queue.Enqueue(startId);
56
57 // Breadth-first traversal discovers the shortest path and records predecessors.
58 while (queue.Count > 0)
59 {
60 var current = queue.Dequeue();
61 if (!outgoing.TryGetValue(current, out var edges)) continue;
62 foreach (var edge in edges)
63 {
64 if (!visited.Add(edge.To)) continue;
65 previousNode[edge.To] = current;
66 previousEdge[edge.To] = edge.Id;
67 if (edge.To == endId)
68 {
69 // Walk predecessors backward, then restore source-to-destination order.
70 var nodes = new List<string> { endId };
71 var edgeIds = new List<string>();
72 var cursor = endId;
73 while (cursor != startId)
74 {
75 edgeIds.Add(previousEdge[cursor]);
76 cursor = previousNode[cursor];
77 nodes.Add(cursor);
78 }
79
80 nodes.Reverse();
81 edgeIds.Reverse();
82 return new GraphPathResult(startId, endId, nodes, edgeIds);
83 }
84
85 queue.Enqueue(edge.To);
86 }
87 }
88
89 return GraphPathResult.Unreachable(startId, endId);
90 }
91 }
