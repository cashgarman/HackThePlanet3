80 public GraphRenderer(GraphPresentationProfile profile = null, GraphExplorerOptions options = null,
81 GraphResourceConfiguration resources = null)
82 {
83 this.profile = profile ?? new GraphPresentationProfile();
84 this.resources = resources ?? new GraphResourceConfiguration();
85 MaximumVisible = (options ?? new GraphExplorerOptions()).NodeRenderBudget;
86
87 // Clone resource materials so renderer-specific properties cannot mutate imported assets.
88 var shader = this.resources.Load<Shader>("GraphNodeToon");
89 if (shader == null) throw new InvalidOperationException("Missing Resources/GraphNodeToon shader.");
90 VisualSettings = this.resources.Load<GraphVisualSettings>("GraphVisualSettings");
91 var sourceMaterial = this.resources.Load<Material>("GraphNodeMaterial");
92 nodeMaterial = sourceMaterial != null ? new Material(sourceMaterial) : new Material(shader);

94 nodeMaterial.enableInstancing = true;
95 nodeMaterial.name = "Graph nodes";
96 haloMesh = Ring();
97 var edgeShader = this.resources.Load<Shader>("GraphEdge");
98 if (edgeShader == null) edgeShader = shader;
99 edgeMaterial = new Material(edgeShader) { name = "Graph edges" };
100 edgeMaterial.SetColor("_BaseColor", Color.white);
101 edgeMaterial.SetFloat("_Filter", EdgeFilterPixels);
102 edgeMaterial.renderQueue = (int)RenderQueue.Geometry + 1;
103 if (edgeMaterial.HasProperty("_Cull")) edgeMaterial.SetInt("_Cull", (int)CullMode.Off);
104 edgeMesh = new Mesh { name = "Graph relationships", indexFormat = IndexFormat.UInt32 };
105 edgeMesh.MarkDynamic();

142 private void EnsureShape(IGraphNode node)
143 {
144 var shape = profile.Type(scene, node).ShapeId;
145 if (shapes.ContainsKey(shape)) return;
146
147 // Initialize the owned model copy with neutral vertex colors for per-instance tinting.
148 var mesh = GraphNodeModels.Create(shape, resources);
149 var colors = new Color32[mesh.vertexCount];
150 for (var i = 0; i < colors.Length; i++) colors[i] = new Color32(255, 255, 255, 255);
151 mesh.colors32 = colors;
152 shapes[shape] = mesh;
153 batches[shape] = new List<Batch>();
154 }

159 public int Degree(IGraphNode node)
160 {
161 return degrees.TryGetValue(node.Id, out var value) ? value : node.Neighbors.Count;
162 }

167 public float Radius(IGraphNode node)
168 {
169 if (Size == SizeMapping.Connections) return .55f + Mathf.Min(1.6f, Mathf.Sqrt(Degree(node)) * .17f);
170 if (Size == SizeMapping.Metric && profile.TryMetric(scene, node, out var value, out var partition) &&
171 metricRanges.TryGetValue(partition, out var range))
172 {
173 var fraction = Mathf.Approximately(range.x, range.y) ? .5f : Mathf.InverseLerp(range.x, range.y, value);
174 // Bounded relative volume: radius is the cube root of normalized volume.
175 return Mathf.Pow(1 + fraction * 7, 1f / 3f) * .65f;
176 }
177
178 return profile.Type(scene, node).DefaultRadius;
179 }

184 public Color NodeColor(IGraphNode node)
185 {
186 // Resolve the chosen data mapping before applying filter dimming and workflow emphasis.
187 var color = ColorBy == ColorMapping.Metric ? MetricColor(node) :
188 ColorBy == ColorMapping.Type ? profile.Type(scene, node).Color :
189 ColorBy == ColorMapping.Category ? profile.CategoryColor(scene, node, Unmapped) :
190 Color.Lerp(new Color(.2f, .45f, .65f), new Color(1, .65f, .25f),
191 Mathf.Clamp01(Mathf.Log(Degree(node) + 1, 2) / 7));
192 if (!node.Matches && !spotlight.Contains(node.Id)) color *= .3f;
193 if (node.Emphasis == ToolEmphasis.Done)
194 {
195 var grey = (color.r + color.g + color.b) / 3f;
196 color = Color.Lerp(new Color(grey, grey, grey), new Color(.4f, .42f, .46f), .78f) * .55f;
197 }
198 else if (node.Emphasis == ToolEmphasis.Next)
199 {
200 color = Color.Lerp(color, new Color(1f, .9f, .35f), .7f) * 1.45f;
201 }
202
203 return color;
204 }

206 // Normalize only within the schema-defined partition; unrelated units are never compared.
207 private Color MetricColor(IGraphNode node)
208 {
209 if (!profile.TryMetric(scene, node, out var value, out var partition) ||
210 !metricRanges.TryGetValue(partition, out var range))
211 return Unmapped;
212 var fraction = Mathf.Approximately(range.x, range.y) ? .5f : Mathf.InverseLerp(range.x, range.y, value);
213 return profile.MetricColor(fraction);
214 }

219 public void UpdateMetrics(IGraphScene graph, IGraphEdgeFilter filter)
220 {
221 // Recompute metric ranges from the currently eligible graph partition.
222 scene = graph;
223 degrees.Clear();
224 metricRanges.Clear();
225 var neighbors = VisibleDegree ? new Dictionary<string, HashSet<string>>() : null;
226 foreach (var node in graph.Nodes.Values)
227 {
228 if (VisibleDegree) neighbors[node.Id] = new HashSet<string>();
229 else degrees[node.Id] = node.Neighbors.Count;
230 if (profile.TryMetric(scene, node, out var value, out var partition))
231 {
232 if (!metricRanges.TryGetValue(partition, out var range)) range = new Vector2(value, value);
233 metricRanges[partition] = new Vector2(Mathf.Min(range.x, value), Mathf.Max(range.y, value));
234 }
235 }
236
237 if (VisibleDegree)
238 {
239 // Count unique neighbors only across matching edges with two visible endpoints.
240 foreach (var edge in graph.Edges.Values)
241 if (filter.EdgeMatches(edge) && graph.Nodes[edge.From].Visible && graph.Nodes[edge.To].Visible)
242 {
243 neighbors[edge.From].Add(edge.To);
244 neighbors[edge.To].Add(edge.From);
245 }
246
247 foreach (var entry in neighbors) degrees[entry.Key] = entry.Value.Count;
248 }
249 }

258 public void Draw(IGraphScene graph, IGraphEdgeFilter filter, Camera camera, IGraphNode selected,
259 IGraphNode hovered, IGraphEdge hoveredEdge)
260 {
261 using var measurement = GraphDiagnostics.Draw.Auto();
262 scene = graph;
263 VisualSettings?.ApplySurface(nodeMaterial);
264 selectedVisualId = selected?.Id;
265 hoveredVisualId = hovered?.Id;
266 selectionPulse = VisualSettings != null
267 ? VisualSettings.SelectionPulse(AnimationTimeOverride ?? Time.unscaledTime, ReducedMotion)
268 : 1;
269 VisibleCount = 0;
270 VisibleEdges = 0;
271 Truncated = false;
272
273 // Reuse allocated batches but clear every per-frame visibility and highlight set.
274 foreach (var list in batches.Values)
275 foreach (var batch in list)
276 batch.Count = 0;
277 rendered.Clear();
278 spotlight.Clear();
279 spotlightFocus.Clear();
280 activePathNodes.Clear();
281 activePathEdges.Clear();
282 if (ActivePath != null && ActivePath.IsReachable)
283 {
284 foreach (var id in ActivePath.NodeIds)
285 if (graph.Nodes.ContainsKey(id))
286 activePathNodes.Add(id);
287 foreach (var id in ActivePath.EdgeIds)
288 if (graph.Edges.ContainsKey(id))
289 activePathEdges.Add(id);
290 }
291
292 CollectSpotlight(selected);
293 GeometryUtility.CalculateFrustumPlanes(camera, frustum);
294
295 // Reserve instances for selection, hover, path, and selected neighbors before filling the budget.
296 if (selected != null) AddNode(selected);
297 if (hovered != null && hovered.Visible) AddNode(hovered);
298 foreach (var id in activePathNodes) AddNode(graph.Nodes[id]);
299 RevealNeighbors(graph, selected);
300 foreach (var node in graph.Nodes.Values)
301 {
302 if (!node.Visible || rendered.Contains(node.Id) ||
303 !GeometryUtility.TestPlanesAABB(frustum, new Bounds(node.Position, Vector3.one * Radius(node) * 4)))
304 continue;
305 if (VisibleCount >= MaximumVisible)
306 {
307 Truncated = true;
308 continue;
309 }
310
311 AddNode(node);
312 }
313
314 var parameters = new RenderParams(nodeMaterial)
315 { camera = camera, shadowCastingMode = ShadowCastingMode.Off, receiveShadows = false };
316
317 // Submit node instances by shape, supplying per-instance colors and selection data.
318 foreach (var entry in batches)
319 foreach (var batch in entry.Value)
320 if (batch.Count > 0)
321 {
322 batch.Properties.SetVectorArray("_BaseColor", batch.Colors);
323 batch.Properties.SetVectorArray("_NodeEffects", batch.Effects);
324 parameters.matProps = batch.Properties;
325 Graphics.RenderMeshInstanced(parameters, shapes[entry.Key], 0, batch.Matrices, batch.Count);
326 }
327
328 using (GraphDiagnostics.Edges.Auto())
329 {
330 // Build one dynamic ribbon mesh for eligible relationships and task decorations.
331 edgeVertices.Clear();
332 edgeColors.Clear();
333 edgeUvs.Clear();
334 edgeIndices.Clear();
335 foreach (var edge in graph.Edges.Values)
336 {
337 var pathEdge = activePathEdges.Contains(edge.Id);
338 if (!EdgeShown(edge, filter, graph, pathEdge)) continue;
339 var fromDrawn = rendered.Contains(edge.From);
340 var toDrawn = rendered.Contains(edge.To);
341 if (!fromDrawn && !toDrawn) continue;
342 var a = graph.Nodes[edge.From];
343 var b = graph.Nodes[edge.To];
344 if (!TryVisibleSegment(a, b, out var start, out var end)) continue;
345 var emphasized = selected != null && (a == selected || b == selected);
346 var incident = emphasized || (hovered != null && (a == hovered || b == hovered));
347 var directHover = hoveredEdge != null && hoveredEdge.Id == edge.Id;
348 var color = pathEdge
349 ? new Color(.98f, .76f, .27f, 1f)
350 : EdgeColor(edge, a, b, incident, directHover);
351 var width = pathEdge ? .16f :
352 directHover ? EdgeWidthHovered :
353 incident ? EdgeWidthIncident : EdgeWidth;
354 // Dim unrelated links only for a selection. Hover brightens the link under the pointer and leaves the rest alone.
355 if (!pathEdge && !directHover && selected != null && !emphasized)
356 {
357 color = new Color(color.r * .22f, color.g * .22f, color.b * .22f, color.a);
358 width = EdgeWidth;
359 }
360
361 Ribbon(start, end, color, width * .5f, camera);
362 if (pathEdge || directHover || (hoveredEdge == null && emphasized))
363 {
364 // Add an arrowhead at the destination for emphasized directed relationships.
365 var direction = end - start;
366 if (direction.sqrMagnitude > 1e-8f)
367 {
368 direction.Normalize();
369 var side = Vector3.Cross(direction, camera.transform.forward);
370 if (side.sqrMagnitude < 1e-8f) side = camera.transform.right;
371 else side.Normalize();
372 side *= .35f;
373 var tip = b.Position;
374 Ribbon(tip, tip - direction * .85f + side, color, width * .45f, camera);
375 Ribbon(tip, tip - direction * .85f - side, color, width * .45f, camera);
376 }
377 }
378
379 VisibleEdges++;
380 }
381
382 DrawTaskRings(graph, camera);
383 edgeMesh.Clear();
384 if (edgeVertices.Count > 0)
385 {
386 // Upload the combined geometry and keep its bounds valid while the camera is inside it.
387 edgeMesh.SetVertices(edgeVertices);
388 edgeMesh.SetColors(edgeColors);
389 edgeMesh.SetUVs(0, edgeUvs);
390 edgeMesh.SetIndices(edgeIndices, MeshTopology.Triangles, 0);
391 edgeMesh.RecalculateBounds();
392 var bounds = edgeMesh.bounds;
393 bounds.Encapsulate(camera.transform.position);
394 bounds.Expand(8f);
395 edgeMesh.bounds = bounds;
396 Graphics.RenderMesh(
397 new RenderParams(edgeMaterial)
398 { camera = camera, worldBounds = bounds, shadowCastingMode = ShadowCastingMode.Off },
399 edgeMesh, 0, Matrix4x4.identity);
400 }
401 }
402
403 if (VisualSettings != null && VisualSettings.ShowSelectionRing)
404 {
405 if (selected != null && rendered.Contains(selected.Id)) Halo(selected, camera, new Color(1, .85f, .5f));
406 if (hovered != null && hovered != selected && rendered.Contains(hovered.Id))
407 Halo(hovered, camera, new Color(.65f, .85f, 1));
408 }
409
410 foreach (var id in activePathNodes)
411 if (graph.Nodes.TryGetValue(id, out var pathNode) && rendered.Contains(id))
412 Halo(pathNode, camera, new Color(1f, .72f, .2f));
413 DrawNodeProgress(graph, camera);
414 }

459 private void DrawTaskRings(IGraphScene graph, Camera camera)
460 {
461 // Draw workflow emphasis first, then task-specific progress and status marks for rendered nodes.
462 foreach (var node in graph.Nodes.Values)
463 if (node.Emphasis == ToolEmphasis.Next && rendered.Contains(node.Id))
464 TaskRing(node, camera, new Color(1f, .88f, .3f), 1.9f, 0, 1, .05f);
465 if (TaskVisuals != null)
466 foreach (var pair in TaskVisuals)
467 {
468 if (!rendered.Contains(pair.Key) || !graph.Nodes.TryGetValue(pair.Key, out var node)) continue;
469 var visual = pair.Value;
470 var color = GraphTaskVisual.Color(visual.Status);
471 var start = visual.Status == "running" && !visual.Progress.HasValue && !ReducedMotion
472 ? Time.unscaledTime * 2
473 : 0;
474 var fraction = visual.Status == "running" ? visual.Progress ?? .25f : 1;
475 if (visual.Status == "draft")
476 for (var segment = 0; segment < 8; segment++)
477 TaskRing(node, camera, color, 1.55f, segment * Mathf.PI / 4, .075f, .05f);
478 else TaskRing(node, camera, color, 1.55f, start, 1, .035f);
479 if (visual.Status == "running")
480 TaskRing(node, camera, color * 1.3f, 1.7f, start, Mathf.Clamp01(fraction), .08f);
481 if (visual.Status == "failed" || visual.Status == "blocked" || visual.Status == "needs_input")
482 {
483 var offset = camera.transform.up * Radius(node) * 2.2f;
484 Ribbon(node.Position + offset, node.Position + offset + camera.transform.up * .65f, color, .1f,
485 camera);
486 Ribbon(node.Position + offset - camera.transform.up * .25f - camera.transform.right * .06f,
487 node.Position + offset - camera.transform.up * .25f + camera.transform.right * .06f,
488 color, .1f, camera);
489 }
490
491 if (visual.Status == "completed")
492 {
493 var anchor = node.Position + camera.transform.right * Radius(node) * 1.9f;
494 var elbow = anchor + camera.transform.right * .25f - camera.transform.up * .25f;
495 Ribbon(anchor, elbow, color, .08f, camera);
496 Ribbon(elbow, elbow + camera.transform.right * .5f + camera.transform.up * .65f, color, .08f,
497 camera);
498 }
499 }
500
501 if (!ReducedMotion && Time.unscaledTime < FlashUntil && FlashTarget != null &&
502 graph.Nodes.TryGetValue(FlashTarget, out var target) && rendered.Contains(FlashTarget))
503 {
504 // Briefly connect the camera to the task target after dispatch when motion effects are enabled.
505 TaskRing(target, camera, new Color(1, .9f, .65f), 2.1f, 0, 1, .12f);
506 var origin = camera.transform.position + camera.transform.forward * 2 + camera.transform.right * .7f -
507 camera.transform.up * .5f;
508 Ribbon(origin, target.Position, new Color(.6f, .85f, 1), .04f, camera);
509 }
510 }

512 private void DrawNodeProgress(IGraphScene graph, Camera camera)
513 {
514 VisibleProgressIndicators = 0;
515 if (NodeProgress == null || (VisualSettings != null && !VisualSettings.ShowNodeProgress)) return;
516 var color = VisualSettings != null ? VisualSettings.ProgressColor : new Color(.25f, .95f, .48f);
517 var track = VisualSettings != null ? VisualSettings.ProgressTrackColor : new Color(.16f, .23f, .20f);
518 var scale = VisualSettings != null ? Mathf.Max(2.3f, VisualSettings.ProgressRadiusScale) : 2.45f;
519 var width = VisualSettings != null ? VisualSettings.ProgressWidth : .07f;
520 var badgeSize = VisualSettings != null ? VisualSettings.CompletionBadgeSize : .4f;
521 foreach (var batch in progressBatches) batch.Count = 0;
522
523 // Pack finite progress values into camera-facing instance quads, capped at 1023 per batch.
524 foreach (var pair in NodeProgress)
525 {
526 if (!rendered.Contains(pair.Key) || !graph.Nodes.TryGetValue(pair.Key, out var node) ||
527 pair.Value == null || !pair.Value.TryFraction(out var fraction)) continue;
528 var index = VisibleProgressIndicators++ / 1023;
529 if (index == progressBatches.Count) progressBatches.Add(new Batch());
530 var batch = progressBatches[index];
531 var radius = Radius(node);
532 batch.Matrices[batch.Count] = Matrix4x4.TRS(node.Position - camera.transform.forward * radius * .02f,
533 camera.transform.rotation, Vector3.one * radius * scale);
534 batch.Effects[batch.Count] = new Vector4(fraction, pair.Value.Complete ? 1 : 0,
535 Mathf.Max(.005f, width) / scale,
536 Mathf.Clamp(badgeSize, .15f, .8f) / scale);
537 batch.Count++;
538 }
539
540 if (VisibleProgressIndicators == 0) return;
541 if (progressMaterial == null)
542 {
543 // Allocate overlay resources lazily once an eligible progress indicator actually exists.
544 var shader = resources.Load<Shader>("GraphNodeProgress");
545 if (shader == null) throw new InvalidOperationException("Missing Resources/GraphNodeProgress shader.");
546 var source = resources.Load<Material>("GraphNodeProgressMaterial");
547 progressMaterial = source != null ? new Material(source) : new Material(shader);
548 progressMaterial.shader = shader;
549 progressMaterial.name = "Graph node progress";
550 progressMaterial.enableInstancing = true;
551 progressMesh = new Mesh { name = "Graph progress quad" };
552 progressMesh.vertices = new[]
553 {
554 new Vector3(-1.35f, -1.35f, 0), new Vector3(1.35f, -1.35f, 0), new Vector3(1.35f, 1.35f, 0),
555 new Vector3(-1.35f, 1.35f, 0)
556 };
557 progressMesh.triangles = new[] { 0, 1, 2, 0, 2, 3 };
558 progressMesh.RecalculateBounds();
559 }
560
561 progressMaterial.SetColor("_ProgressColor", color);
562 progressMaterial.SetColor("_TrackColor", track);
563 foreach (var batch in progressBatches)
564 if (batch.Count > 0)
565 {
566 batch.Properties.SetVectorArray("_ProgressData", batch.Effects);
567 if (camera.GetComponent<GraphNodeEffects>()?.Renderer != this)
568 Graphics.RenderMeshInstanced(
569 new RenderParams(progressMaterial)
570 {
571 camera = camera, matProps = batch.Properties, shadowCastingMode = ShadowCastingMode.Off,
572 receiveShadows = false
573 }, progressMesh, 0, batch.Matrices, batch.Count);
574 }
575 }

588 private void TaskRing(IGraphNode node, Camera camera, Color color, float scale, float start, float fraction,
589 float width)
590 {
591 // Approximate a camera-facing circular arc with the shared ribbon builder.
592 var segments = Mathf.Max(1, Mathf.CeilToInt(32 * fraction));
593 var radius = Radius(node) * scale;
594 for (var i = 0; i < segments; i++)
595 {
596 float a = start + i / (float)segments * Mathf.PI * 2 * fraction,
597 b = start + (i + 1) / (float)segments * Mathf.PI * 2 * fraction;
598 var p = node.Position + (camera.transform.right * Mathf.Cos(a) + camera.transform.up * Mathf.Sin(a)) *
599 radius;
600 var q = node.Position + (camera.transform.right * Mathf.Cos(b) + camera.transform.up * Mathf.Sin(b)) *
601 radius;
602 Ribbon(p, q, color, width, camera);
603 }
604 }

606 private void AddNode(IGraphNode node)
607 {
608 if (!rendered.Add(node.Id)) return;
609 VisibleCount++;
610 EnsureShape(node);
611
612 // Split instances at Unity's batch limit while preserving independent color and effect data.
613 var list = batches[profile.Type(scene, node).ShapeId];
614 var index = 0;
615 while (index < list.Count && list[index].Count == 1023) index++;
616 if (index == list.Count) list.Add(new Batch());
617 var batch = list[index];
618 var radius = Radius(node);
619 batch.Matrices[batch.Count] = Matrix4x4.TRS(node.Position, Quaternion.identity, Vector3.one * radius * 2);
620 batch.Colors[batch.Count] = NodeInstanceColor(node);
621 batch.Effects[batch.Count] =
622 new Vector4(node.Id == selectedVisualId ? 1 : node.Id == hoveredVisualId ? .16f : 0, selectionPulse, 0,
623 0);
624 batch.Count++;
625 }

627 // Instanced alpha drives the surface pulse in GraphUnlit. Only a running task sets it.
628 private Color NodeInstanceColor(IGraphNode node)
629 {
630 var color = NodeColor(node);
631 color.a = 0f;
632 if (TaskVisuals == null || !TaskVisuals.TryGetValue(node.Id, out var task) ||
633 task.Status != "running") return color;
634 var pulse = RunningPulse(node.Id);
635 var glow = GraphTaskVisual.Color(task.Status);
636 color = Color.Lerp(color, glow, .12f + .38f * pulse) * (1f + .7f * pulse);
637 color.a = pulse;
638 return color;
639 }

641 private float RunningPulse(string id)
642 {
643 if (ReducedMotion) return .8f;

759 private void Ribbon(Vector3 a, Vector3 b, Color color, float halfWidth, Camera camera)
760 {
761 if (!ClipNear(ref a, ref b, camera)) return;
762
763 // Convert world width to endpoint pixel widths so filtering remains consistent under perspective.
764 var span = b - a;
765 var length = span.magnitude;
766 if (length < 1e-5f) return;
767 var direction = span / length;
768 var forward = camera.transform.forward;
769 var right = camera.transform.right;
770 var up = camera.transform.up;
771 var origin = camera.transform.position;
772 var near = camera.nearClipPlane + .08f;
773 var tanHalf = Mathf.Tan(camera.fieldOfView * Mathf.Deg2Rad * .5f);
774 if (tanHalf < 1e-4f) return;
775 var viewScale = Mathf.Max(1f, camera.pixelHeight) / (2f * tanHalf);
776 var depthA = Mathf.Max(Vector3.Dot(a - origin, forward), near);
777 var depthB = Mathf.Max(Vector3.Dot(b - origin, forward), near);
778 var coreA = halfWidth * viewScale / depthA;
779 var coreB = halfWidth * viewScale / depthB;
780 var halfA = (coreA + EdgePadPixels) * depthA / viewScale;
781 var halfB = (coreB + EdgePadPixels) * depthB / viewScale;
782 var lateral = new Vector2(Vector3.Dot(direction, right), Vector3.Dot(direction, up)).magnitude;
783 var extendA = CapExtension(lateral, depthA, viewScale, depthA - near, Vector3.Dot(direction, forward));
784 var extendB = CapExtension(lateral, depthB, viewScale, depthB - near, -Vector3.Dot(direction, forward));
785 var start = a - direction * extendA;
786 var end = b + direction * extendB;
787 var za = Mathf.Max(Vector3.Dot(a - origin, forward), near);
788 var zb = Mathf.Max(Vector3.Dot(b - origin, forward), near);
789 var pa = new Vector2(Vector3.Dot(a - origin, right), Vector3.Dot(a - origin, up)) / za;
790 var pb = new Vector2(Vector3.Dot(b - origin, right), Vector3.Dot(b - origin, up)) / zb;
791 var screenDir = pb - pa;
792 var segmentPixels = screenDir.magnitude * viewScale;
793 var perp = screenDir.sqrMagnitude < 1e-10f
794 ? Vector2.right
795 : new Vector2(-screenDir.y, screenDir.x).normalized;
796 var side = right * perp.x + up * perp.y;
797 var capA = extendA * lateral * viewScale / depthA;
798 var capB = extendB * lateral * viewScale / depthB;
799 var index = edgeVertices.Count;
800
801 // Pad the quad and encode core width and cap distances for shader coverage filtering.
802 AddEdgeVertex(start - side * halfA, color,
803 new Vector4(-(coreA + EdgePadPixels), coreA, -capA, segmentPixels));
804 AddEdgeVertex(start + side * halfA, color, new Vector4(coreA + EdgePadPixels, coreA, -capA, segmentPixels));
805 AddEdgeVertex(end - side * halfB, color,
806 new Vector4(-(coreB + EdgePadPixels), coreB, segmentPixels + capB, segmentPixels));
807 AddEdgeVertex(end + side * halfB, color,
808 new Vector4(coreB + EdgePadPixels, coreB, segmentPixels + capB, segmentPixels));
809 edgeIndices.Add(index);
810 edgeIndices.Add(index + 1);
811 edgeIndices.Add(index + 2);
812 edgeIndices.Add(index + 1);
813 edgeIndices.Add(index + 3);
814 edgeIndices.Add(index + 2);
815 }

824 private static float CapExtension(float lateral, float depth, float viewScale, float depthRoom,
825 float slopeTowardCamera)
826 {
827 // Add enough world-space padding for the screen filter without extending through the near plane.
828 var pixelsPerWorld = Mathf.Max(lateral, .05f) * viewScale / Mathf.Max(depth, 1e-4f);
829 var extend = EdgePadPixels / pixelsPerWorld;
830 if (slopeTowardCamera > 1e-4f) extend = Mathf.Min(extend, Mathf.Max(0f, depthRoom) / slopeTowardCamera);
831 return extend;
832 }

834 private static bool ClipNear(ref Vector3 a, ref Vector3 b, Camera camera)
835 {
836 // Reject fully hidden segments and trim crossing segments to a plane just beyond the camera near clip.
837 var origin = camera.transform.position + camera.transform.forward * (camera.nearClipPlane + .08f);
838 var da = Vector3.Dot(a - origin, camera.transform.forward);
839 var db = Vector3.Dot(b - origin, camera.transform.forward);
840 if (da < 0f && db < 0f) return false;
841 if (da >= 0f && db >= 0f) return true;
842 var t = da / (da - db);
843 var hit = Vector3.Lerp(a, b, t);
844 if (da < 0f) a = hit;
845 else b = hit;
846 return true;
847 }

849 private static Vector3 ClosestWorld(Vector3 a, Vector3 b, Vector3 point, out float t)
850 {
851 var span = b - a;
852 var length = span.sqrMagnitude;
853 t = length < 1e-8f ? 0f : Mathf.Clamp01(Vector3.Dot(point - a, span) / length);
854 return a + span * t;

864 public IGraphEdge PickEdge(IGraphScene graph, IGraphEdgeFilter filter, Camera camera, Vector2 mouse,
865 float maxPixels = 16f)
866 {
867 using var profile = GraphDiagnostics.PickEdge.Auto();
868 IGraphEdge hit = null;
869 var nearest = float.MaxValue;
870 var tanHalf = Mathf.Tan(camera.fieldOfView * Mathf.Deg2Rad * .5f);
871 var pixelHeight = Mathf.Max(1f, camera.pixelHeight);
872
873 // Choose the closest screen-space relationship using a width-aware hit tolerance.
874 foreach (var edge in graph.Edges.Values)
875 {
876 if (!EdgeShown(edge, filter, graph)) continue;
877 if (!rendered.Contains(edge.From) && !rendered.Contains(edge.To)) continue;
878 if (!TryVisibleSegment(graph.Nodes[edge.From], graph.Nodes[edge.To], out var a, out var b) ||
879 !ClipNear(ref a, ref b, camera)) continue;
880 var distance = ScreenSegmentDistance(camera, mouse, a, b);
881 var depth = Mathf.Max(camera.nearClipPlane,
882 Vector3.Dot((a + b) * .5f - camera.transform.position, camera.transform.forward));
883 var pixelHalf = EdgeWidth * .5f * pixelHeight / (2f * depth * tanHalf);
884 var limit = Mathf.Clamp(pixelHalf + 12f, maxPixels, 64f);
885 if (distance < limit && distance < nearest)
886 {
887 nearest = distance;
888 hit = edge;
889 }
890 }
891
892 return hit;
893 }

913 private static float ScreenSegmentDistance(Camera camera, Vector2 mouse, Vector3 a, Vector3 b)
914 {
915 // Project onto the finite screen segment, including degenerate segments that collapse to one pixel.
916 var sa = camera.WorldToScreenPoint(a);
917 var sb = camera.WorldToScreenPoint(b);
918 if (sa.z <= 0 || sb.z <= 0) return float.MaxValue;
919 var p0 = new Vector2(sa.x, sa.y);
920 var p1 = new Vector2(sb.x, sb.y);
921 var delta = p1 - p0;
922 var length = delta.sqrMagnitude;
923 if (length < 1e-4f) return Vector2.Distance(mouse, p0);
924 var t = Mathf.Clamp01(Vector2.Dot(mouse - p0, delta) / length);
925 return Vector2.Distance(mouse, p0 + delta * t);
926 }

933 public IGraphNode Pick(IGraphScene graph, Camera camera, Vector2 mouse)
934 {
935 using var profile = GraphDiagnostics.PickNode.Auto();
936 var ray = camera.ScreenPointToRay(mouse);
937 IGraphNode hit = null;
938 var nearest = float.MaxValue;
939 foreach (var node in graph.Nodes.Values)
940 {
941 if ((!node.Visible && !spotlight.Contains(node.Id)) || !rendered.Contains(node.Id)) continue;
942 // All models fit the same radius; a forgiving sphere also keeps thin
943 // badges, pins and the globe's holes easy to target from any angle.
944 var offset = ray.origin - node.Position;
945 var b = Vector3.Dot(offset, ray.direction);
946 var radius = Radius(node) * 1.15f;
947 var discriminant = b * b - offset.sqrMagnitude + radius * radius;
948 if (discriminant < 0) continue;
949 var distance = -b - Mathf.Sqrt(discriminant);
950 if (distance < 0) distance = -b + Mathf.Sqrt(discriminant);
951 if (distance >= 0 && distance < nearest)
952 {
953 nearest = distance;
954 hit = node;
955 }
956 }
957
958 return hit;
959 }

961 private static Mesh Ring()
962 {
963 // Build three perpendicular line circles to keep the selection halo readable from any viewing angle.
964 var mesh = new Mesh();
965 var vertices = new Vector3[96];
966 var indices = new int[192];
967 for (var axis = 0; axis < 3; axis++)
968 for (var i = 0; i < 32; i++)
969 {
970 var a = i * Mathf.PI * 2 / 32;
971 var p = new Vector3(Mathf.Cos(a) * .5f, Mathf.Sin(a) * .5f, 0);
972 vertices[axis * 32 + i] =
973 axis == 0 ? p : axis == 1 ? new Vector3(p.x, 0, p.y) : new Vector3(0, p.x, p.y);
974 indices[(axis * 32 + i) * 2] = axis * 32 + i;
975 indices[(axis * 32 + i) * 2 + 1] = axis * 32 + (i + 1) % 32;
976 }
977
978 var colors = new Color32[vertices.Length];
979 for (var i = 0; i < colors.Length; i++) colors[i] = new Color32(255, 255, 255, 255);
980 mesh.vertices = vertices;
981 mesh.colors32 = colors;
982 mesh.SetIndices(indices, MeshTopology.Lines, 0);
983 return mesh;
