66 public void BuildFavoritesDrawer()
67 {
68 // Share the host's favorites filter while virtualizing the independent favorites navigation list.
69 shellFavoritesBody = Element("drawer-section");
70 filterScroll.Add(shellFavoritesBody);
71 Section(shellFavoritesBody, "FAVORITES");
72 favoriteStatus = Label("", "note");
73 shellFavoritesBody.Add(favoriteStatus);
74 favoritesOnlyToggle = new Toggle("Show only favorites")
75 { name = "shell-favorites-only", value = options.FavoritesOnly() };
76 UseShellCheckbox(favoritesOnlyToggle);
77 favoritesOnlyToggle.RegisterValueChangedCallback(e => options.SetFavoritesOnly(e.newValue));
78 shellFavoritesBody.Add(favoritesOnlyToggle);
79 favoritesList = new ListView
80 {
81 name = "shell-favorites-list", itemsSource = options.Results, fixedItemHeight = 48,
82 selectionType = SelectionType.Single, virtualizationMethod = CollectionVirtualizationMethod.FixedHeight
83 };
84 favoritesList.style.flexGrow = 1;
85 favoritesList.style.minHeight = 120;
86 HideHorizontalScroller(favoritesList);
87 favoritesList.makeItem = () =>
88 {
89 var row = Element("result-row");
90 row.Add(Label("", "result-name"));
91 row.Add(Label("", "result-meta"));
92 row.Add(GraphContentUi.FavoriteStar(14, options.FavoriteColor));
93 return row;
94 };
95 favoritesList.bindItem = (element, index) =>
96 {
97 var node = options.Results[index];
98 var title = element.Q<Label>(className: "result-name");
99 title.text = node.Label;
100 title.style.color = options.Color(node);
101 element.Q<Label>(className: "result-meta").text = options.TypeLabel(node) + " · " +
102 node.Neighbors.Count + " connections" +
103 (node.Visible ? "" : " · hidden by filters");
104 GraphContentUi.BindFavoriteRow(element, node);
105 };
106 favoritesList.selectionChanged += items =>
107 {
108 var node = items.OfType<TNode>().FirstOrDefault();
109 if (node == null) return;
110 // Same rule as global search: a hidden favorite must be revealed, not silently left off-screen.
111 if (!node.Visible) options.ResetFilters();
112 options.Activate(node);
113 };
114 shellFavoritesBody.Add(favoritesList);
115 }
116
117 // Favorites are listed by name regardless of filters, so a hidden favorite can still be found and revealed.
119 public void RefreshFavorites()
120 {
121 if (favoritesList == null || favoriteStatus == null || options.Nodes() == null) return;
122 options.Results.Clear();
123 options.Results.AddRange(options.Nodes().Values.Where(n => n.Favorite)
124 .OrderBy(n => n.Label, StringComparer.OrdinalIgnoreCase)
125 .ThenBy(n => n.Id, StringComparer.Ordinal));
126 favoritesList.Rebuild();
127 if (options.FavoriteIds.Count == 0)
128 {
129 favoriteStatus.text =
130 "No favorites yet. Select an entity and press its star in Overview. Favorites rank first in search and stay saved between sessions.";
131 return;
132 }
133
134 var missing = options.FavoriteIds.Count(id => !options.Nodes().ContainsKey(id));
135
136 // Report saved favorites absent from this loaded graph without dropping their persisted identities.
137 var summary = options.Results.Count + (options.Results.Count == 1 ? " favorite" : " favorites") +
138 " in this graph";
139 if (missing > 0) summary += " · " + missing + " not in the loaded graph";
140 favoriteStatus.text = summary;
141 }
142 }
