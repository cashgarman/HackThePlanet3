114 public void UpsertNode(GraphNodeRecord node)
115 {
116 if (node == null) throw new ArgumentNullException(nameof(node));
117 EnsureLive();
118
119 // Compare before updating the adjacency index so no-op upserts preserve the revision.
120 nodes.TryGetValue(node.Id, out var previous);
121 if (previous != null && Equivalent(previous, node)) return;
122 if (!nodes.ContainsKey(node.Id))
123 {
124 adjacency.Add(node.Id, new Adjacency());
125 TopologyRevision++;
126 }
127 else if (ReferenceEquals(nodes[node.Id], node))
128 {
129 return;
130 }
131 else if (nodes[node.Id].TypeId != node.TypeId)
132 {
133 TopologyRevision++;
134 }
135
136 // Publish the replacement and record its before/after values outside batch replay.
137 nodes[node.Id] = node;
138 Revision++;
139 if (!suppressTimeline)
140 {
141 Timeline.Append(new GraphTimelineChange(previous == null ? null : new[] { previous }, new[] { node },
142 null, null, DisplayName, DisplayName));
143 Changed?.Invoke();
144 }
145 }

149 public void UpsertEdge(GraphEdgeRecord edge)
150 {
151 if (edge == null) throw new ArgumentNullException(nameof(edge));
152 EnsureLive();
153
154 // Reject dangling edges before changing records or adjacency indexes.
155 if (!nodes.ContainsKey(edge.From) || !nodes.ContainsKey(edge.To))
156 throw new ArgumentException("Both edge endpoints must exist.", nameof(edge));
157 edges.TryGetValue(edge.Id, out var prior);
158 if (prior != null && Equivalent(prior, edge)) return;
159 if (edges.TryGetValue(edge.Id, out var previous))
160 {
161 if (ReferenceEquals(edge, previous)) return;
162 if (previous.From != edge.From || previous.To != edge.To)
163 {
164 // Retargeting must repair both endpoint indexes before replacing the edge.
165 Unlink(previous);
166 Link(edge);
167 TopologyRevision++;
168 }
169 else if (previous.TypeId != edge.TypeId)
170 {
171 TopologyRevision++;
172 }
173 }
174 else
175 {
176 Link(edge);
177 TopologyRevision++;
178 }
179
180 // Preserve evidence-only updates in history without rebuilding the topology.
181 edges[edge.Id] = edge;
182 Revision++;
183 if (!suppressTimeline)
184 {
185 Timeline.Append(new GraphTimelineChange(null, null, prior == null ? null : new[] { prior },
186 new[] { edge }, DisplayName, DisplayName));
187 Changed?.Invoke();
188 }
189 }

191 private void Link(GraphEdgeRecord edge)
192 {
193 adjacency[edge.From].Edges.Values.Add(edge.Id);
194 adjacency[edge.To].Edges.Values.Add(edge.Id);
195 if (edge.From == edge.To) return; // Preserve recursive facts, but exclude self from unique-neighbor degree.
196 AddNeighbor(edge.From, edge.To);
197 AddNeighbor(edge.To, edge.From);
198 }

200 private void AddNeighbor(string from, string to)
201 {
202 var index = adjacency[from];
203
204 // Keep duplicate parallel edges counted while exposing each neighbor only once.
205 index.Counts.TryGetValue(to, out var count);
206 index.Counts[to] = count + 1;
207 index.Neighbors.Values.Add(to);
208 }
209
210 private void Unlink(GraphEdgeRecord edge)
211 {
212 adjacency[edge.From].Edges.Values.Remove(edge.Id);
213 adjacency[edge.To].Edges.Values.Remove(edge.Id);
214 if (edge.From == edge.To) return;
215 RemoveNeighbor(edge.From, edge.To);
216 RemoveNeighbor(edge.To, edge.From);
217 }
218
219 private void RemoveNeighbor(string from, string to)
220 {
221 var index = adjacency[from];
222 var count = index.Counts[to] - 1;
223 if (count > 0)
224 {
225 index.Counts[to] = count;
226 }
227 else
228 {
229 // Remove the neighbor only after the last parallel edge has disappeared.
230 index.Counts.Remove(to);
231 index.Neighbors.Values.Remove(to);
232 }

260 public bool RemoveNode(string id)
261 {
262 EnsureLive();
263 if (!nodes.ContainsKey(id)) return false;
264 var before = Snapshot();
265
266 // Suppress per-edge history entries so removing the node records one coherent change.
267 suppressTimeline = true;
268 try
269 {
270 foreach (var edge in adjacency[id].Edges.ToArray()) RemoveEdge(edge);
271 adjacency.Remove(id);
272 nodes.Remove(id);
273 Revision++;

275 }
276 finally
277 {

289 public HashSet<string> Neighborhood(string id, int depth)
290 {
291 if (depth < 0) throw new ArgumentOutOfRangeException(nameof(depth));
292 var found = new HashSet<string>(StringComparer.Ordinal);
293 if (id == null || !nodes.ContainsKey(id)) return found;
294 var pending = new Queue<(string id, int level)>();
295 found.Add(id);
296 pending.Enqueue((id, 0));
297
298 // Visit each node once and stop expanding when its hop limit is reached.
299 while (pending.Count > 0)
300 {
301 var current = pending.Dequeue();
302 if (current.level >= depth) continue;
303 foreach (var neighbor in adjacency[current.id].Neighbors)
304 if (found.Add(neighbor))
305 pending.Enqueue((neighbor, current.level + 1));
306 }
307
308 return found;
309 }

320 public void ReplaceSnapshot(GraphSnapshot snapshot)
321 {
322 if (snapshot == null) throw new ArgumentNullException(nameof(snapshot));
323 if (snapshot.DatasetId != DatasetId)
324 throw new ArgumentException("A snapshot belongs to a different dataset.", nameof(snapshot));
325 EnsureLive();
326 var beforeSnapshot = Timeline.HasBaseline ? Snapshot() : null;
327 // Validate in a detached document before touching current facts or indexes.
328 var incoming = new GraphDocument(DatasetId, snapshot.DisplayName);
329 var batch = new GraphBatch();
330 batch.Nodes.AddRange(snapshot.Nodes);
331 batch.Edges.AddRange(snapshot.Edges);
332 incoming.Apply(batch);
333 int before = Revision, topologyBefore = TopologyRevision;
334 suppressTimeline = true;
335 try
336 {
337 // Install the validated snapshot while exposing one revision for the replacement.
338 nodes.Clear();
339 edges.Clear();
340 adjacency.Clear();
341 foreach (var node in incoming.Nodes.Values) UpsertNode(node);
342 foreach (var edge in incoming.Edges.Values) UpsertEdge(edge);
343 DisplayName = incoming.DisplayName;
344 Revision = before + 1;
345 TopologyRevision = topologyBefore + 1;
346 }
347 finally
348 {
349 suppressTimeline = false;
350 }
351
352 if (beforeSnapshot != null) RecordDifference(beforeSnapshot, Snapshot());
353 }

357 public void Apply(GraphBatch batch)
358 {
359 if (batch == null) throw new ArgumentNullException(nameof(batch));
360 EnsureLive();
361
362 // Freeze the batch and reject conflicting identities before mutating the document.
363 var nodeUpdates = batch.Nodes.ToArray();
364 var edgeUpdates = batch.Edges.ToArray();
365 var removedNodes =
366 new HashSet<string>(batch.RemoveNodes.Select(id => GraphRecordCopy.Required(id, "nodeId")),
367 StringComparer.Ordinal);
368 var removedEdges =
369 new HashSet<string>(batch.RemoveEdges.Select(id => GraphRecordCopy.Required(id, "edgeId")),
370 StringComparer.Ordinal);
371 var updatedNodes = new HashSet<string>(StringComparer.Ordinal);
372 var updatedEdges = new HashSet<string>(StringComparer.Ordinal);
373 foreach (var node in nodeUpdates)
374 if (node == null || !updatedNodes.Add(node.Id) || removedNodes.Contains(node.Id))
375 throw new ArgumentException("Null, duplicate or conflicting node update.", nameof(batch));
376 foreach (var edge in edgeUpdates)
377 {
378 if (edge == null || !updatedEdges.Add(edge.Id) || removedEdges.Contains(edge.Id))
379 throw new ArgumentException("Null, duplicate or conflicting edge update.", nameof(batch));
380
381 bool Exists(string id)
382 {
383 return !removedNodes.Contains(id) && (nodes.ContainsKey(id) || updatedNodes.Contains(id));
384 }
385
386 // Resolve endpoints against the final batch state, including newly added nodes.
387 if (!Exists(edge.From) || !Exists(edge.To))
388 throw new ArgumentException("An edge references a missing or removed endpoint.", nameof(batch));
389 }
390
391 int before = Revision, topologyBefore = TopologyRevision;
392 var beforeSnapshot = Timeline.HasBaseline ? Snapshot() : null;
393 suppressTimeline = true;
394 try
395 {
396 // Apply removals first, then install nodes before their referencing edges.
397 foreach (var id in removedEdges) RemoveEdge(id);
398 foreach (var id in removedNodes) RemoveNode(id);
399 foreach (var node in nodeUpdates) UpsertNode(node);
400 foreach (var edge in edgeUpdates) UpsertEdge(edge);
401 if (Revision != before) Revision = before + 1;
402 if (TopologyRevision != topologyBefore) TopologyRevision = topologyBefore + 1;
403 }
404 finally
405 {
406 suppressTimeline = false;
407 }
408
409 if (Revision != before && beforeSnapshot != null) RecordDifference(beforeSnapshot, Snapshot());
410 }

535 private GraphSnapshot Reconstruct(int entryIndex)
536 {
537 // Replay into a detached document so seeking cannot append new live history.
538 var state = new GraphDocument(DatasetId, Timeline.Baseline.DisplayName);
539 state.suppressTimeline = true;
540 var baseline = new GraphBatch();
541 baseline.Nodes.AddRange(Timeline.Baseline.Nodes);
542 baseline.Edges.AddRange(Timeline.Baseline.Edges);
543 state.Apply(baseline);
544 for (var i = 0; i <= entryIndex; i++)
545 {
546 // Records absent from the after-set are deletions; the remaining after-set is upserted.
547 var change = Timeline.Entries[i].Change;
548 var batch = new GraphBatch();
549 foreach (var node in change.NodesBefore)
550 if (!change.NodesAfter.Any(item => item.Id == node.Id))
551 batch.RemoveNodes.Add(node.Id);
552 foreach (var edge in change.EdgesBefore)
553 if (!change.EdgesAfter.Any(item => item.Id == edge.Id))
554 batch.RemoveEdges.Add(edge.Id);
555 batch.Nodes.AddRange(change.NodesAfter);
556 batch.Edges.AddRange(change.EdgesAfter);
557 state.Apply(batch);
558 if (change.DisplayNameBefore != change.DisplayNameAfter) state.Rename(change.DisplayNameAfter);
559 }
560
561 return state.Snapshot();
562 }

590 private void RecordDifference(GraphSnapshot before, GraphSnapshot after)
591 {
592 if (!Timeline.HasBaseline) return;
593
594 // Index both snapshots to collect only records that were added, removed, or changed.
595 var beforeNodes = before.Nodes.ToDictionary(item => item.Id, StringComparer.Ordinal);
596 var afterNodes = after.Nodes.ToDictionary(item => item.Id, StringComparer.Ordinal);
597 var beforeEdges = before.Edges.ToDictionary(item => item.Id, StringComparer.Ordinal);
598 var afterEdges = after.Edges.ToDictionary(item => item.Id, StringComparer.Ordinal);
599 var changedBeforeNodes = beforeNodes.Values
600 .Where(item => !afterNodes.TryGetValue(item.Id, out var next) ||
601 !Equivalent(item, next))
602 .OrderBy(item => item.Id, StringComparer.Ordinal).ToArray();
603 var changedAfterNodes = afterNodes.Values
604 .Where(item => !beforeNodes.TryGetValue(item.Id, out var prior) ||
605 !Equivalent(item, prior))
606 .OrderBy(item => item.Id, StringComparer.Ordinal).ToArray();
607 var changedBeforeEdges = beforeEdges.Values
608 .Where(item => !afterEdges.TryGetValue(item.Id, out var next) ||
609 !Equivalent(item, next))
610 .OrderBy(item => item.Id, StringComparer.Ordinal).ToArray();
611 var changedAfterEdges = afterEdges.Values
612 .Where(item => !beforeEdges.TryGetValue(item.Id, out var prior) ||
613 !Equivalent(item, prior))
614 .OrderBy(item => item.Id, StringComparer.Ordinal).ToArray();
615 var changed =
616 changedBeforeNodes.Length + changedAfterNodes.Length + changedBeforeEdges.Length +
617 changedAfterEdges.Length > 0 || before.DisplayName != after.DisplayName;
618 if (!changed) return;
619
620 // Publish a single ordered delta for the completed compound operation.
621 Timeline.Append(new GraphTimelineChange(changedBeforeNodes, changedAfterNodes, changedBeforeEdges,
622 changedAfterEdges, before.DisplayName, after.DisplayName));
623 Changed?.Invoke();
624 }

626 private static bool Equivalent(GraphNodeRecord left, GraphNodeRecord right)
627 {
628 return left.TypeId == right.TypeId && left.Label == right.Label &&
629 left.Sources.SequenceEqual(right.Sources, StringComparer.Ordinal) &&
630 left.Properties.Count == right.Properties.Count &&
631 left.Properties.All(pair => right.Properties.TryGetValue(pair.Key, out var value) &&
632 pair.Value.Equals(value));
633 }
634
635 private static bool Equivalent(GraphEdgeRecord left, GraphEdgeRecord right)
636 {
637 return left.From == right.From && left.To == right.To && left.TypeId == right.TypeId &&
638 left.Status == right.Status && left.Confidence == right.Confidence &&
639 left.Sources.SequenceEqual(right.Sources, StringComparer.Ordinal) &&
640 left.Properties.Count == right.Properties.Count &&
641 left.Properties.All(pair => right.Properties.TryGetValue(pair.Key, out var value) &&
642 pair.Value.Equals(value));
643 }
