16 private GraphRenderingSession(Camera camera, GraphResourceConfiguration resources,
17 GraphPresentationProfile profile, GraphExplorerOptions options)
18 {
19 Camera = camera;
20 Effects = camera.GetComponent<GraphNodeEffects>();
21 if (Effects != null && Effects.Renderer != null)
22 throw new InvalidOperationException("This camera already has an explorer renderer.");
23 ownsEffects = Effects == null;
24
25 // Preserve borrowed effect settings so disposing the session can restore the host camera exactly.
26 if (!ownsEffects)
27 {
28 previousSettings = Effects.Settings;
29 previousResources = Effects.Resources;
30 previousEffectsEnabled = Effects.enabled;
31 }
32
33 try
34 {
35 // Allocate session resources inside the cleanup boundary to release partial initialization on failure.
36 Renderer = new GraphRenderer(profile, options, resources);
37 Layout = new ForceLayout(profile);
38 Navigation = new GraphNavigation(camera);
39 Atmosphere = new GraphAtmosphere(camera, resources);
40 if (ownsEffects) Effects = camera.gameObject.AddComponent<GraphNodeEffects>();
41 Effects.Resources = resources;
42 Effects.Settings = Renderer.VisualSettings;
43 Effects.Renderer = Renderer;
44 Effects.enabled = true;
45 }
46 catch
47 {
48 Dispose();
49 throw;
50 }
51 }

67 public void Dispose()
68 {
69 if (disposed) return;
70 disposed = true;
71 Navigation?.Dispose();
72 if (Effects != null)
73 {
74 // Disable first so its render subscription and pass materials are released immediately.
75 Effects.enabled = false;
76 Effects.Renderer = null;
77 if (ownsEffects)
78 {
79 if (Application.isPlaying) Object.Destroy(Effects);
80 else Object.DestroyImmediate(Effects);
81 }
82 else
83 {
84 Effects.Settings = previousSettings;
85 Effects.Resources = previousResources;
86 Effects.enabled = previousEffectsEnabled;
87 }
88 }
89
90 Atmosphere?.Dispose();
91 Renderer?.Dispose();
92 }

100 public static GraphRenderingSession Initialize(Camera camera, GraphResourceConfiguration resources,
101 GraphPresentationProfile profile = null,
102 GraphExplorerOptions options = null)
103 {
104 if (camera == null) throw new ArgumentNullException(nameof(camera));
105 if (resources == null) throw new ArgumentNullException(nameof(resources));
106 // Validate before allocating meshes, materials, effects or atmosphere objects.
107 foreach (var key in new[] { "GraphNodeToon", "GraphEdge", "GraphSky", "GraphFog", "GraphNodePost" })
108 if (resources.Load<Shader>(key) == null)
109 throw new InvalidOperationException("Missing explorer shader: " + key);
110 if (resources.Load<Material>("GraphNodeMaskMaterial") == null)
111 throw new InvalidOperationException("Missing node mask material.");
112 return new GraphRenderingSession(camera, resources, profile ?? new GraphPresentationProfile(),
113 options ?? new GraphExplorerOptions());
114 }
