110 public static void StyleDisclosure(Foldout foldout)
111 {
112 // Remove built-in checkmarks and reserve a consistent slot for the disclosure chevron.
113 foldout.AddToClassList("inspector-disclosure");
114 foreach (var extra in foldout.Query(className: "unity-foldout__checkmark").ToList())
115 extra.style.display = DisplayStyle.None;
116 var toggle = foldout.Q<Toggle>(className: "unity-foldout__toggle");
117 if (toggle == null) return;
118 toggle.style.flexDirection = FlexDirection.Row;
119 toggle.style.alignItems = Align.Center;
120 var input = toggle.Q(className: "unity-toggle__input");
121 if (input != null)
122 {
123 input.style.backgroundColor = Color.clear;
124 input.style.borderTopWidth = 0;
125 input.style.borderBottomWidth = 0;
126 input.style.borderLeftWidth = 0;
127 input.style.borderRightWidth = 0;
128 input.style.width = 16;
129 input.style.minWidth = 16;
130 input.style.height = 16;
131 input.style.marginRight = 6;
132 input.style.alignItems = Align.Center;
133 input.style.justifyContent = Justify.Center;
134 toggle.Insert(0, input);
135 }
136
137 var checkmark = toggle.Q(className: "unity-toggle__checkmark");
138 if (checkmark != null) checkmark.style.display = DisplayStyle.None;
139 var foldoutMark = toggle.Q(className: "unity-foldout__checkmark");
140 if (foldoutMark != null) foldoutMark.style.display = DisplayStyle.None;
141 var glyph = new VisualElement { pickingMode = PickingMode.Ignore };
142 glyph.style.width = 12;
143 glyph.style.height = 12;
144 glyph.style.flexShrink = 0;
145 glyph.generateVisualContent += context =>
146 {
147 // Paint an expanded or collapsed chevron from the foldout's current value.
148 var rect = glyph.contentRect;
149 var painter = context.painter2D;
150 painter.strokeColor = new Color(.96f, .98f, 1f);
151 painter.lineWidth = 2.5f;
152 painter.BeginPath();
153 if (foldout.value)
154 {
155 painter.MoveTo(new Vector2(rect.x + 1, rect.y + 3));
156 painter.LineTo(new Vector2(rect.x + 6, rect.y + 8));
157 painter.LineTo(new Vector2(rect.x + 11, rect.y + 3));
158 }
159 else
160 {
161 painter.MoveTo(new Vector2(rect.x + 3, rect.y + 1));
162 painter.LineTo(new Vector2(rect.x + 8, rect.y + 6));
163 painter.LineTo(new Vector2(rect.x + 3, rect.y + 11));
164 }
165
166 painter.Stroke();
167 };
168 glyph.RegisterCallback<GeometryChangedEvent>(_ => glyph.MarkDirtyRepaint());
169 foldout.RegisterValueChangedCallback(_ => glyph.MarkDirtyRepaint());
170 if (input != null) input.Add(glyph);
171 else toggle.Insert(0, glyph);
172 }

200 public static void Relationships(VisualElement parent, string nodeId, int total, Func<string> query,
201 Action<string> setQuery,
202 Func<IEnumerable<GraphRelationRow>> rows, Action<IGraphEdge> hover,
203 Action<IGraphEdge> focus, GraphInspectorOptions options)
204 {
205 parent.Add(Label("Search relationships", "eyebrow"));
206 var search = new TextField { value = query(), name = "connection-search" };
207 search.AddToClassList("nolabel");
208 HideFieldLabel(search);
209 options.BindTip(search, "Search all loaded relationships by entity, type, or relationship name.");
210 parent.Add(search);
211 var count = Label("", "note");
212 parent.Add(count);
213 var connections = Element("connections-list");
214 parent.Add(connections);
215 IGraphEdge activeHover = null, activeFocus = null;
216
217 void Populate()
218 {
219 // Clear stale hover/focus highlights and tooltip bindings before rebuilding relationship rows.
220 activeHover = null;
221 activeFocus = null;
222 hover(null);
223 focus(null);
224 options.ForgetTips(connections);
225 connections.Clear();
226 var shown = 0;
227 foreach (var row in rows())
228 {
229 // Search entity, type, and relation labels together without changing the graph's filters.
230 var text = row.Label + " " + row.TypeLabel + " " + row.Edge.Relation + " " + row.RelationLabel;
231 if (!string.IsNullOrWhiteSpace(query()) &&
232 text.IndexOf(query().Trim(), StringComparison.OrdinalIgnoreCase) < 0) continue;
233 shown++;
234 var group = Element("connection");
235 connections.Add(group);
236 group.RegisterCallback<PointerEnterEvent>(_ =>
237 {
238 activeHover = row.Edge;
239 hover(row.Edge);
240 });
241 group.RegisterCallback<PointerLeaveEvent>(_ =>
242 {
243 if (ReferenceEquals(activeHover, row.Edge))
244 {
245 activeHover = null;
246 hover(null);
247 }
248 });
249 group.RegisterCallback<FocusInEvent>(_ =>
250 {
251 activeFocus = row.Edge;
252 focus(row.Edge);
253 });
254 group.RegisterCallback<FocusOutEvent>(_ =>
255 {
256 if (ReferenceEquals(activeFocus, row.Edge))
257 {
258 activeFocus = null;
259 focus(null);
260 }
261 });
262 var heading = Element("connection-header");
263 var tint = row.Color;
264 tint.a = .22f;
265 heading.style.backgroundColor = tint;
266 if (row.Icon != null) heading.Add(row.Icon());
267 heading.Add(Label(row.TypeLabel, "connection-kind"));
268 group.Add(heading);
269 var target = Element("connection-target");
270 var relation = Label(row.RelationLabel, "connection-relation");
271 relation.style.color = RelationInk(row.RelationColor);
272 target.Add(relation);
273 options.BindTip(heading, (row.Edge.From == nodeId ? "OUT → " : "IN ← ") + row.RelationLabel);
274 target.Add(Button(row.Label, row.Select, "link"));
275 group.Add(target);
276 options.AppendRelationshipActions?.Invoke(group, row.Edge);
277 }
278
279 count.text = shown + " of " + total +
280 " loaded relationships · hover or focus an entry to highlight its link";
281 }
282
283 search.RegisterValueChangedCallback(e =>
284 {
285 setQuery(e.newValue);
286 Populate();
287 });
288 Populate();
289 }

296 public static void Render(VisualElement parent, IGraphScene scene, IGraphNode node,
297 GraphInspectorOptions options = null)
298 {
299 // Dispose old tooltip bindings before rebuilding the selected node or the empty selection prompt.
300 options ??= new GraphInspectorOptions();
301 options.ForgetTips(parent);
302 parent.Clear();
303 if (scene == null || node == null)
304 {
305 Section(parent, "EXPLORE THE CONNECTIONS");
306 parent.Add(Label("Select an entity", "inspector-title"));
307 Note(parent, "Select a node to inspect its properties, sources, and relationships.");
308 return;
309 }
310
311 var type = scene.Schema.NodeTypes.TryGetValue(node.Record.TypeId, out var definition) ? definition : null;
312 Header(parent, type?.Label ?? node.Record.TypeId, node.Label, type?.Color ?? Color.gray,
313 node.Neighbors.Count + " distinct neighbors · " + node.Edges.Count + " relationships" +
314 (node.Pinned ? " · PINNED" : "") + (node.Favorite ? " · FAVORITE" : ""));
315 if (options.AppendSectionsBeforeProperties) options.AppendSections?.Invoke(parent, node);
316 var properties = parent;
317 if (options.CollapseProperties)
318 {
319 var disclosure = new Foldout { text = "Properties", value = false };
320 StyleDisclosure(disclosure);
321 parent.Add(disclosure);
322 properties = disclosure;
323 }
324 else
325 {
326 Section(parent, "PROPERTIES");
327 }
328
329 foreach (var pair in node.Record.Properties.OrderBy(pair => pair.Key, StringComparer.Ordinal))
330 {
331 // Honor schema inspectability while retaining undeclared properties under their original keys.
332 var field = scene.Schema.Fields.TryGetValue(pair.Key, out var known) ? known : null;
333 if (field != null && !field.Inspectable) continue;
334 GraphContentUi.AddProperty(properties, field?.Label ?? pair.Key, pair.Value.ToString(),
335 options.Decorate, options.BindTip);
336 }
337
338 if (node.Edges.Count > 0)
339 {
340 // Resolve the opposite projected endpoint and schema labels for each loaded relationship.
341 var disclosure = new Foldout { text = "Relationships", value = false };
342 StyleDisclosure(disclosure);
343 parent.Add(disclosure);
344 options.Decorate(disclosure);
345 var query = "";
346 Relationships(disclosure, node.Id, node.Edges.Count, () => query, value => query = value, () => node
347 .Edges.OrderBy(edge => edge.Relation).Select(edge =>
348 {
349 var other = scene.Nodes[edge.From == node.Id ? edge.To : edge.From];
350 var known = scene.Schema.NodeTypes.TryGetValue(other.Record.TypeId, out var kind)
351 ? kind
352 : null;
353 return new GraphRelationRow
354 {
355 Edge = edge, Label = other.Label,
356 TypeLabel = known?.Label ?? other.Record.TypeId,
357 Color = known?.Color ?? Color.gray,
358 RelationLabel =
359 scene.Schema.Relations.TryGetValue(edge.Relation, out var relation)
360 ? relation.Label
361 : edge.Relation,
362 RelationColor = relation?.Color ?? new Color(.62f, .67f, .74f),
363 Select = () => options.Select?.Invoke(other)
364 };
365 }), _ => { }, _ => { }, options);
366 }
367
368 var sources = new Foldout { text = "Sources & provenance", value = false };
369
370 // Show stable provenance even when the host does not provide an action for opening sources.
371 StyleDisclosure(sources);
372 parent.Add(sources);
373 options.Decorate(sources);
374 Note(sources, "Stable ID: " + node.Id);
375 foreach (var source in node.Record.Sources)
376 if (options.OpenSource != null)
377 sources.Add(Button(source, () => options.OpenSource(source), "source-link"));
378 else Note(sources, source);
379 if (!options.AppendSectionsBeforeProperties) options.AppendSections?.Invoke(parent, node);
380 foreach (var action in options.Actions ?? Array.Empty<GraphInspectorAction>())
381 {
382 // Check visibility now and recheck enabled state at click time before rebuilding changed node details.
383 if (action == null || !action.IsVisible(node)) continue;
384 var enabled = action.IsEnabled(node);
385 var button = Button(action.Label, () =>
386 {
387 if (!action.IsEnabled(node)) return;
388 action.Execute(node);
389 var current = scene.Nodes.TryGetValue(node.Id, out var refreshed) ? refreshed : null;
390 Render(parent, scene, current, options);
391 }, "inspector-action");
392 button.name = action.Id;
393 button.SetEnabled(enabled);
394 button.tooltip = enabled ? action.Tooltip : action.DisabledReason?.Invoke(node) ?? action.Tooltip;
395 parent.Add(button);
396 }
397 }
