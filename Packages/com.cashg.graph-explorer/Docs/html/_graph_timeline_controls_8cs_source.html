519 private float ToProgress(DateTimeOffset time)
520 {
521 // Map UTC ticks into the visible window, retaining double precision until the final slider value.
522 var timeline = Document.Timeline;
523 long start = TimelineStartTimeUtc.UtcDateTime.Ticks, end = EndTime.UtcDateTime.Ticks;
524 return Mathf.Clamp01((float)Viewport.ToVisible(
525 end <= start
526 ? 1
527 : (time.ToUniversalTime().UtcDateTime.Ticks - start) / (double)(end - start)));
528 }

530 private DateTimeOffset FromProgress(float progress)
531 {
532 // Undo viewport zoom before mapping a slider fraction back into the full timestamp range.
533 long start = TimelineStartTimeUtc.UtcDateTime.Ticks, end = EndTime.UtcDateTime.Ticks;
534 return new DateTimeOffset(start + (long)((end - start) * Viewport.FromVisible(progress)), TimeSpan.Zero);
535 }

537 private void ZoomTimeline(WheelEvent evt)
538 {
539 if (!evt.ctrlKey || Document == null || !Document.Timeline.HasBaseline || evt.delta.y == 0) return;
540 var activeSlider = Continuous || Viewport.IsZoomed ? (VisualElement)timeSlider : slider;
541 var bounds = SliderTravel(activeSlider);
542 if (!float.IsFinite(bounds.x) || bounds.y <= 0) return;
543 double anchor = Mathf.Clamp01((evt.mousePosition.x - bounds.x) / bounds.y);
544 // Keep high precision in the viewport: a small session can follow years of repository history.
545 Viewport.Zoom(anchor, Math.Pow(1.2, Math.Max(-20, Math.Min(20, evt.delta.y))));
546 if (Viewport.IsZoomed && EndTime > TimelineStartTimeUtc)
547 Workspace.SetView(TimeAt(Viewport.Start), TimeAt(Viewport.End));
548 else Workspace.SetView(null, null);
549 Refresh();
550 evt.StopImmediatePropagation();
551 evt.PreventDefault();
552 }

554 private double ProgressAt(DateTimeOffset time)
555 {
556 // Continuous history uses elapsed time; discrete history spaces semantic entries evenly.
557 long start = TimelineStartTimeUtc.UtcDateTime.Ticks, end = EndTime.UtcDateTime.Ticks;
558 if (Continuous) return end <= start ? 0 : (time.UtcDateTime.Ticks - start) / (double)(end - start);
559 var dates = new[] { Document.Timeline.BaselineTimeUtc }
560 .Concat(Document.Timeline.Entries.Select(entry => entry.TimestampUtc)).ToArray();
561 if (time <= dates[0]) return 0;
562 for (var i = 1; i < dates.Length; i++)
563 if (time <= dates[i])
564 return (i - 1 + (time - dates[i - 1]).Ticks /
565 (double)Math.Max(1, (dates[i] - dates[i - 1]).Ticks)) / Math.Max(1, dates.Length - 1);
566 return 1;
567 }

569 private DateTimeOffset TimeAt(double progress)
570 {
571 // Invert the timeline mapping, interpolating timestamps between equally spaced semantic entries.
572 progress = Math.Max(0, Math.Min(1, progress));
573 if (Continuous)
574 return TimelineStartTimeUtc.AddTicks((long)((EndTime - TimelineStartTimeUtc).Ticks * progress));
575 var timeline = Document.Timeline;
576 var index = progress * timeline.Entries.Count;
577 var lower = (int)Math.Floor(index);
578 var from = lower == 0 ? timeline.BaselineTimeUtc : timeline.Entries[lower - 1].TimestampUtc;
579 var to = lower >= timeline.Entries.Count ? from : timeline.Entries[lower].TimestampUtc;
580 return from.AddTicks((long)((to - from).Ticks * (index - lower)));
581 }

694 private void TickCache()
695 {
696 try
697 {
698 // Drive nested provider coroutines within a small UI budget and pause at ordinary yields.
699 var started = Stopwatch.StartNew();
700 while (cacheRoutines.Count > 0 && started.ElapsedMilliseconds < 5)
701 {
702 var next = cacheRoutines.Peek();
703 if (!next.MoveNext())
704 {
705 (cacheRoutines.Pop() as IDisposable)?.Dispose();
706 continue;
707 }
708
709 if (next.Current is IEnumerator child) cacheRoutines.Push(child);
710 else break;
711 }
712
713 if (cacheRoutines.Count == 0)
714 {
715 cacheSchedule?.Pause();
716 cacheCancellation?.Dispose();
717 cacheCancellation = null;
718 }
719 }
720 catch (Exception error)
721 {
722 CancelCaching();
723 cacheStatus.text = "Cache failed: " + error.Message;
724 Debug.LogException(error);
725 }
726 }

728 private void CancelCaching()
729 {
730 // Dispose the full coroutine stack so iterator finally blocks clear the workspace's busy state.
731 cacheCancellation?.Cancel();
732 cacheSchedule?.Pause();
733 while (cacheRoutines.Count > 0) (cacheRoutines.Pop() as IDisposable)?.Dispose();
734 cacheCancellation?.Dispose();
735 cacheCancellation = null;
736 }
