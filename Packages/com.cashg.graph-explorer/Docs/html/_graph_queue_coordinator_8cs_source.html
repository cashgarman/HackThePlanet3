9 // The host owns durable drafts and pending identities; the provider owns execution.
10 // No submission may leave this boundary before the identity has been persisted.
12 public sealed class GraphQueueCoordinator

18 public GraphQueueCoordinator(IGraphActionProvider provider)

24 public bool Busy { get; private set; }

26 private IEnumerator Perform(Func<IEnumerator> operation, Action<Exception> failed)
27 {
28 // Serialize provider operations and release the busy flag even when a coroutine fails or is disposed.
29 if (Busy)
30 {
31 failed(new InvalidOperationException("A queue operation is already in progress."));
32 yield break;
33 }
34
35 Busy = true;
36 try
37 {
38 yield return operation();
39 }
40 finally
41 {
42 Busy = false;
43 }
44 }

52 public IEnumerator Review(IReadOnlyList<GraphActionRequest> draft, CancellationToken cancellation,
53 Action<GraphActionRun> completed, Action<Exception> failed)
54 {
55 // Snapshot ordering before yielding: edits cannot reorder an in-flight review.
56 var ordered = draft.ToArray();
57 return Perform(() => provider.Review(ordered, cancellation, completed, failed), failed);
58 }

67 public IEnumerator Submit(GraphActionRun review, Func<bool> persistPending, CancellationToken cancellation,
68 Action<GraphActionRun> completed, Action<Exception> failed)
69 {
70 return Perform(() => SubmitCore(review, persistPending, cancellation, completed, failed), failed);

73 private IEnumerator SubmitCore(GraphActionRun review, Func<bool> persistPending, CancellationToken cancellation,
74 Action<GraphActionRun> completed, Action<Exception> failed)
75 {
76 if (cancellation.IsCancellationRequested)
77 {
78 failed(new OperationCanceledException());
79 yield break;
80 }
81
82 if (review == null || string.IsNullOrEmpty(review.Id) || string.IsNullOrEmpty(review.ReviewToken) ||
83 review.Entries.Any(entry => entry.Errors.Count > 0))
84 {
85 // Reject missing review authority or entry errors before saving or sending a request.
86 failed(new InvalidOperationException("A valid reviewed queue is required."));
87 yield break;
88 }
89
90 // Persist the exact request identity before the provider can begin external execution.
91 if (!persistPending())
92 {
93 failed(new InvalidOperationException("Pending submission could not be saved."));
94 yield break;
95 }
96
97 yield return provider.Execute(review.Id, review, cancellation, completed, failed);
98 }
99
106 public IEnumerator Recover(string requestId, CancellationToken cancellation, Action<GraphActionRun> completed,
107 Action<Exception> failed)
108 {
109 return Perform(() => provider.Recover(requestId, cancellation, completed, failed), failed);
110 }
111
120 public IEnumerator Control(string runId, string operation, string entryId, CancellationToken cancellation,

122 {
123 return Perform(() => provider.Control(runId, operation, entryId, cancellation, completed, failed), failed);
124 }
