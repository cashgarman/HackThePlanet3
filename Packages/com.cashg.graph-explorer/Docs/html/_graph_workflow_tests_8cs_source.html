83 [Test]
84 public void ProtocolHandshakeRequestsApprovalsQuestionsAndDisconnectArePumped()
85 {
86 var wire = new FakeTransport();
87 using var client = new CodexAppServerClient(wire);
88 var ready = false;
89 string error = null;
90 client.Start(() => ready = true, e => error = e);
91 wire.Receive(new { id = 1, result = new { } });
92
93 // Transport callbacks queue messages; readiness must change only when the host pumps them.
94 Assert.That(ready, Is.False);
95 client.Pump();
96 Assert.That(ready, Is.True);
97 Assert.That(wire.Sent[1], Does.Contain("initialized"));
98 var result = false;
99 client.Request("thread/start", new { cwd = "fixture" }, r => result = true, e => error = e);
100 wire.Receive(new { id = 2, result = new { thread = new { id = "thread" } } });
101 client.Pump();
102 Assert.That(result, Is.True);
103 var methods = new List<string>();
104
105 // Exercise server requests and notifications, then fail outstanding work when the connection closes.
106 client.Message += (method, p, id) =>
107 {
108 methods.Add(method);
109 if (id.HasValue) client.Respond(id.Value, new { decision = "decline" });
110 };
111 foreach (var method in new[]
112 { "item/commandExecution/requestApproval", "item/tool/requestUserInput", "turn/completed" })
113 wire.Receive(new { method, id = 90, @params = new { threadId = "thread" } });
114 client.Pump();
115 Assert.That(methods.Count, Is.EqualTo(3));
116 Assert.That(wire.Sent.Last(), Does.Contain("decline"));
117 client.Request("thread/resume", new { threadId = "thread" }, _ => Assert.Fail(), e => error = e);
118 wire.Close();
119 client.Pump();
120 Assert.That(error, Is.EqualTo("offline"));
121 Assert.That(client.Ready, Is.False);
122 }

125 [Test]
126 public void ProtocolReportsErrorsAndDoesNotRetryMutatingRequests()
127 {
128 // Keep transport failures visible and require explicit reconciliation before retry.
129 var wire = new FakeTransport();
130 using var client = new CodexAppServerClient(wire);
131 string error = null;
132 client.Start(() => { }, e => error = e);
133 wire.Receive(new { id = 1, result = new { } });
134 client.Pump();
135 client.Request("turn/start", new { threadId = "thread" }, _ => Assert.Fail(), e => error = e);
136 wire.Receive(new { id = 2, error = new { message = "denied" } });
137 client.Pump();
138 Assert.That(error, Does.Contain("denied"));
139
140 // Pumping again must not resend a mutating request after a protocol failure.
141 client.Pump();
142 Assert.That(wire.Sent.Count(s => s.Contains("turn/start")), Is.EqualTo(1));
143 }
