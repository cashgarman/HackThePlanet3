29 [Test]
30 public void CrossingEdgesKeepIdentityDirectionAndMultiplicity()
31 {
32 var source = MakeScene();
33 var before = GraphJson.Serialize(source.Document);
34 var view = new GraphClusterScene(source);
35 var id = view.Collapse(new[] { "a", "b", "c" }, "Neighborhood");
36
37 // Internal edges disappear from projection while crossing IDs, directions, and parallel facts survive.
38 Assert.That(view.Nodes.Keys, Is.EquivalentTo(new[] { id, "d", "outside" }));
39 Assert.That(view.Edges.Keys, Is.EquivalentTo(new[] { "cd", "in", "out", "parallel" }));
40 Assert.That(view.Edges["in"].From, Is.EqualTo("outside"));
41 Assert.That(view.Edges["in"].To, Is.EqualTo(id));
42 Assert.That(view.Edges["out"].From, Is.EqualTo(id));
43 Assert.That(view.Edges["out"].To, Is.EqualTo("outside"));
44 Assert.That(view.Nodes[id].Record.Properties["memberCount"].Number, Is.EqualTo(3));
45 Assert.That(view.LeafIds(id), Is.EquivalentTo(new[] { "a", "b", "c" }));
46 Assert.That(view.MemberRecords(id).Select(record => record.Id), Is.EquivalentTo(new[] { "a", "b", "c" }));
47 Assert.That(view.InternalEdges(id).Select(edge => edge.Id), Is.EquivalentTo(new[] { "ab", "bc", "self" }));
48 Assert.That(GraphJson.Serialize(source.Document), Is.EqualTo(before));
49 Assert.That(view.Expand(id), Is.True);
50 Assert.That(view.Edges.Keys, Is.EquivalentTo(source.Edges.Keys));
51 Assert.That(view.Nodes.Keys, Is.EquivalentTo(source.Nodes.Keys));
52 }

55 [Test]
56 public void NestedClustersRestoreOneLevelAtATimeAndCarryPositions()
57 {
58 // Verify both hierarchy depth and source-node positions across collapse and expand.
59 var source = MakeScene();
60 var view = new GraphClusterScene(source);
61 var inner = view.Collapse(new[] { "a", "b" }, "Inner");
62 var outer = view.Collapse(new[] { inner, "c" }, "Outer");
63 Assert.That(view.LeafIds(outer), Is.EquivalentTo(new[] { "a", "b", "c" }));
64 var original = source.Nodes["a"].Position;
65 view.Nodes[outer].Position += Vector3.right * 10;
66
67 // Expand one level at a time so the outer drag propagates through the still-collapsed inner group.
68 view.Expand(outer);
69 Assert.That(view.Nodes.ContainsKey(inner), Is.True);
70 Assert.That(view.Nodes.ContainsKey("a"), Is.False);
71 view.Expand(inner);
72 Assert.That(view.Nodes.ContainsKey("a"), Is.True);
73 Assert.That(source.Nodes["a"].Position.x, Is.EqualTo(original.x + 10).Within(.001f));
74 Assert.That(view.Edges.Keys, Is.EquivalentTo(source.Edges.Keys));
75 Assert.That(view.Recollapse(inner), Is.True);
76 Assert.That(view.Recollapse(outer), Is.True);
77 }

80 [Test]
81 public void PositionHistoryUsesCanonicalNodesAndTracksCollapsedClusterMotion()
82 {
83 var source = MakeScene();
84 var view = new GraphClusterScene(source);
85 var initial = source.Nodes["a"].Position;
86 var cluster = view.Collapse(new[] { "a", "b", "c" }, "Group");
87 view.Nodes[cluster].Position += Vector3.right * 7;
88
89 // Motion history must contain source identities with group displacement, never temporary cluster IDs.
90 var positions = view.CanonicalPositions();
91 Assert.That(positions.Keys, Is.EquivalentTo(source.Document.Nodes.Keys));
92 Assert.That(positions.ContainsKey(cluster), Is.False);
93 Assert.That(positions["a"].x, Is.EqualTo(initial.x + 7).Within(.001f));
94 view.ApplyCanonicalPositions(positions);
95 var restored = view.CanonicalPositions();
96 Assert.That(restored["a"], Is.EqualTo(positions["a"]));
97 Assert.That(restored["b"], Is.EqualTo(positions["b"]));
98 }

101 [Test]
102 public void DistanceLodOnlyGroupsFarConnectedNeighborsAndReleasesNearCamera()
103 {
104 var source = MakeScene();
105 source.Nodes["a"].Position = new Vector3(100, 0, 0);
106 source.Nodes["b"].Position = new Vector3(101, 0, 0);
107 source.Nodes["c"].Position = new Vector3(102, 0, 0);
108 source.Nodes["d"].Position = new Vector3(200, 0, 0);
109 source.Nodes["outside"].Position = Vector3.zero;
110 var view = new GraphClusterScene(source)
111 { LodEnabled = true, LodEnterDistance = 80, LodExitDistance = 65, LodNeighborSeparation = 4 };
112 Assert.That(view.UpdateLod(Vector3.zero), Is.True);
113 Assert.That(view.Nodes.Count, Is.EqualTo(3));
114 var cluster = view.Nodes.Keys.Single(view.IsCluster);
115 Assert.That(view.LeafIds(cluster), Is.EquivalentTo(new[] { "a", "b", "c" }));
116 Assert.That(view.Edges["in"].To, Is.EqualTo(cluster));
117 Assert.That(view.Expand(cluster), Is.True);
118 Assert.That(view.UpdateLod(Vector3.zero), Is.False, "User expansion overrides automatic regrouping.");
119 view.RemoveCluster(cluster);
120
121 // Clear the manual override to allow regrouping, then approach the group to trigger automatic expansion.
122 view.ClearLodOverrides();
123 Assert.That(view.UpdateLod(Vector3.zero), Is.True);
124 Assert.That(view.UpdateLod(new Vector3(50, 0, 0)), Is.True);
125 Assert.That(view.Nodes.Keys, Is.EquivalentTo(source.Nodes.Keys));
126 }
