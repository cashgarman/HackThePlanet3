86 public bool Reveal(string sourceNodeId)
87 {
88 if (!source.Nodes.ContainsKey(sourceNodeId)) return false;
89
90 // Collect enclosing groups so the outermost collapsed ancestor is expanded first.
91 var ancestors = new List<string>();
92 var id = sourceNodeId;
93 while (parents.TryGetValue(id, out var parent))
94 {
95 ancestors.Add(parent);
96 id = parent;
97 }
98
99 var changed = false;
100 for (var i = ancestors.Count - 1; i >= 0; i--) changed |= Expand(ancestors[i]);
101
102 // Keep automatic LOD from immediately hiding the explicitly revealed node again.
103 lodSuppressed.Add(sourceNodeId);
104 return changed;
105 }

140 public IReadOnlyDictionary<string, Vector3> CanonicalPositions()
141 {
142 var result = new SortedDictionary<string, Vector3>(StringComparer.Ordinal);
143 foreach (var sourceNode in source.Nodes.Values)
144 {
145 // Accumulate dragged offsets from every collapsed ancestor without modifying source facts.
146 var position = sourceNode.Position;
147 var child = sourceNode.Id;
148 while (parents.TryGetValue(child, out var parent))
149 {
150 var cluster = clusters[parent];
151 if (cluster.Collapsed) position += cluster.View.Position - cluster.LastPosition;
152 child = parent;
153 }
154
155 result.Add(sourceNode.Id, position);
156 }
157
158 return new ReadOnlyDictionary<string, Vector3>(result);

164 public void ApplyCanonicalPositions(IReadOnlyDictionary<string, Vector3> positions)
165 {
166 if (positions == null) throw new ArgumentNullException(nameof(positions));
167
168 // Restore source-node motion first, resetting any missing nodes to their stable seed.
169 foreach (var node in source.Nodes.Values)
170 {
171 node.Position = positions.TryGetValue(node.Id, out var position)
172 ? position
173 : GraphViewState.InitialPosition(node.Id);
174 node.Velocity = Vector3.zero;
175 }
176
177 // Recenter inner groups before their parents so nested centroids use restored positions.
178 foreach (var cluster in clusters.Values.OrderByDescending(Depth))
179 {
180 var members = cluster.Children.Select(child => clusters.TryGetValue(child, out var nested)
181 ? nested.View.Position
182 : source.Nodes.TryGetValue(child, out var node)
183 ? node.Position
184 : (Vector3?)null)
185 .Where(position => position.HasValue).Select(position => position.Value).ToArray();
186 if (members.Length == 0) continue;
187 cluster.View.Position =
188 members.Aggregate(Vector3.zero, (sum, position) => sum + position) / members.Length;
189 cluster.View.Velocity = Vector3.zero;
190 cluster.LastPosition = cluster.View.Position;
191 }

229 public string Collapse(IEnumerable<string> memberIds, string label = null)
230 {
231 Synchronize();
232
233 // Only visible siblings can form a group; reject mixed parents before changing membership.
234 var members = (memberIds ?? throw new ArgumentNullException(nameof(memberIds)))
235 .Distinct(StringComparer.Ordinal).ToArray();
236 if (members.Length < 2 || members.Any(id => !nodes.ContainsKey(id)))
237 throw new ArgumentException("Choose at least two currently visible nodes or clusters.",
238 nameof(memberIds));
239 var enclosing = parents.TryGetValue(members[0], out var owner) ? owner : null;
240 if (members.Any(id => (parents.TryGetValue(id, out var parent) ? parent : null) != enclosing))
241 throw new ArgumentException("Members must share the same immediate parent.", nameof(memberIds));
242 var id = "graph-explorer:cluster:" + Guid.NewGuid().ToString("N");
243
244 // Place the new group at its members' centroid and splice it into the parent hierarchy.
245 var cluster = new Cluster
246 {
247 Id = id, Label = string.IsNullOrWhiteSpace(label) ? members.Length + " nodes" : label,
248 Children = members,
249 LastPosition =
250 members.Select(member => nodes[member].Position).Aggregate(Vector3.zero, (a, b) => a + b) /
251 members.Length
252 };
253 cluster.View.Position = cluster.LastPosition;
254 clusters.Add(id, cluster);
255 if (enclosing != null)
256 {
257 var outer = clusters[enclosing];
258 var selected = new HashSet<string>(members, StringComparer.Ordinal);
259 outer.Children = outer.Children.Where(child => !selected.Contains(child)).Concat(new[] { id })
260 .ToArray();
261 parents[id] = enclosing;
262 }
263
264 foreach (var member in members) parents[member] = id;
265 Rebuild();
266 return id;
267 }

277 private bool Expand(string id, bool suppressLod)
278 {
279 if (!clusters.TryGetValue(id, out var cluster) || !cluster.Collapsed) return false;
280
281 // An explicit expansion exempts automatic group members from further distance collapsing.
282 if (suppressLod && cluster.Automatic)
283 foreach (var member in LeafIds(id))
284 lodSuppressed.Add(member);
285 // Move members with a dragged cluster while retaining their relative layout.
286 var delta = cluster.View.Position - cluster.LastPosition;
287 foreach (var child in cluster.Children)
288 if (clusters.TryGetValue(child, out var nested)) nested.View.Position += delta;
289 else if (source.Nodes.TryGetValue(child, out var node)) node.Position += delta;
290 cluster.Collapsed = false;
291 Rebuild();
292 return true;
293 }

298 public bool Recollapse(string id)
299 {
300 if (!clusters.TryGetValue(id, out var cluster) || cluster.Collapsed) return false;
301
302 // Reuse the existing membership but recenter it on its currently visible children.
303 cluster.View.Position = cluster.Children.Where(child => nodes.ContainsKey(child))
304 .Select(child => nodes[child].Position)
305 .DefaultIfEmpty(cluster.View.Position)
306 .Aggregate(Vector3.zero, (a, b) => a + b) /
307 Math.Max(1, cluster.Children.Count(child => nodes.ContainsKey(child)));
308 cluster.LastPosition = cluster.View.Position;
309 cluster.Collapsed = true;
310 Rebuild();
311 return true;
312 }

317 public bool RemoveCluster(string id)
318 {
319 if (!clusters.TryGetValue(id, out var cluster)) return false;
320 if (cluster.Collapsed) Expand(id, false);
321
322 // Splice children into the removed group's parent rather than losing their nesting.
323 if (parents.TryGetValue(id, out var outerId))
324 {
325 var outer = clusters[outerId];
326 outer.Children = outer.Children.Where(child => child != id).Concat(cluster.Children).ToArray();
327 foreach (var child in cluster.Children) parents[child] = outerId;
328 }
329 else
330 {
331 foreach (var child in cluster.Children) parents.Remove(child);
332 }
333
334 parents.Remove(id);
335 clusters.Remove(id);
336 Rebuild();
337 return true;
338 }

341 public void Synchronize()
342 {
343 if (sourceRevision == source.Revision) return;
344 sourceRevision = source.Revision;
345 lodSuppressed.RemoveWhere(id => !source.Nodes.ContainsKey(id));
346 // Remove vanished facts from temporary groups. A group reduced to one
347 // member dissolves, including when it is nested in another group.
348 bool pruned;
349 do
350 {
351 pruned = false;
352 foreach (var cluster in clusters.Values.ToArray())
353 {
354 cluster.Children = cluster.Children
355 .Where(child => source.Nodes.ContainsKey(child) ||
356 clusters.ContainsKey(child)).ToArray();
357 if (cluster.Children.Length >= 2) continue;
358 var outerId = parents.TryGetValue(cluster.Id, out var parent) ? parent : null;
359 if (outerId != null && clusters.TryGetValue(outerId, out var outer))
360 outer.Children = outer.Children.Where(child => child != cluster.Id).Concat(cluster.Children)
361 .ToArray();
362 foreach (var child in cluster.Children)
363 if (outerId != null) parents[child] = outerId;
364 else parents.Remove(child);
365 parents.Remove(cluster.Id);
366 clusters.Remove(cluster.Id);
367 pruned = true;
368 }
369 } while (pruned);
370
371 Rebuild();
372 }

374 private void Rebuild()
375 {
376 nodes.Clear();
377 edges.Clear();
378
379 string Representative(string id)
380 {
381 // The outermost collapsed ancestor is the sole visible representative of a source node.
382 var representative = id;
383 while (parents.TryGetValue(id, out var parent))
384 {
385 if (clusters[parent].Collapsed) representative = parent;
386 id = parent;
387 }
388
389 return representative;
390 }
391
392 // Project each visible representative once, sharing source view state for uncollapsed nodes.
393 foreach (var item in source.Nodes)
394 {
395 var id = Representative(item.Key);
396 if (!nodes.ContainsKey(id))
397 nodes.Add(id, new Node(this, id == item.Key ? item.Value : null,
398 id == item.Key ? null : clusters[id]));
399 }
400
401 // Retarget crossing edges while omitting connections internal to a collapsed group.
402 foreach (var item in source.Edges)
403 {
404 string from = Representative(item.Value.From), to = Representative(item.Value.To);
405 if (from == to && (from != item.Value.From || to != item.Value.To)) continue;
406 var edge = new Edge(item.Value, from, to);
407 edges.Add(edge.Id, edge);
408 ((Node)nodes[from]).Link(to, edge);
409 ((Node)nodes[to]).Link(from, edge);
410 }
411
412 Revision++;
413 }

427 public bool UpdateLod(Vector3 cameraPosition)
428 {
429 Synchronize();
430 var changed = false;
431
432 // Expand automatic groups near the camera using a smaller exit threshold to prevent flicker.
433 var exitDistance = Mathf.Max(0, Mathf.Min(LodExitDistance, LodEnterDistance - .01f));
434 foreach (var cluster in clusters.Values.Where(c => c.Automatic).ToArray())
435 if (Vector3.Distance(cameraPosition, cluster.View.Position) < exitDistance)
436 {
437 RemoveCluster(cluster.Id);
438 changed = true;
439 }
440
441 if (!LodEnabled)
442 {
443 // Disabling LOD dissolves only automatic groups, preserving manually created ones.
444 foreach (var cluster in clusters.Values.Where(c => c.Automatic).ToArray())
445 {
446 RemoveCluster(cluster.Id);
447 changed = true;
448 }
449
450 return changed;
451 }
452
453 // Exclude manually revealed members and nearby nodes from new automatic groups.
454 var candidates = nodes.Values
455 .Where(node => !(IsCluster(node.Id)
456 ? LeafIds(node.Id).Any(lodSuppressed.Contains)
457 : lodSuppressed.Contains(node.Id)) &&
458 Vector3.Distance(cameraPosition, node.Position) >= LodEnterDistance)
459 .ToDictionary(node => node.Id, node => node, StringComparer.Ordinal);
460 var visited = new HashSet<string>(StringComparer.Ordinal);
461 foreach (var candidate in candidates.Values.ToArray())
462 {
463 // Flood connected, closely spaced siblings so unrelated distant nodes remain separate.
464 if (!visited.Add(candidate.Id)) continue;
465 var enclosing = parents.TryGetValue(candidate.Id, out var owner) ? owner : null;
466 var group = new List<string>();
467 var queue = new Queue<string>();
468 queue.Enqueue(candidate.Id);
469 while (queue.Count > 0)
470 {
471 var id = queue.Dequeue();
472 group.Add(id);
473 foreach (var neighbor in candidates[id].Neighbors)
474 if (candidates.TryGetValue(neighbor, out var other) &&
475 (parents.TryGetValue(neighbor, out var parent) ? parent : null) == enclosing &&
476 Vector3.Distance(candidates[id].Position, other.Position) <= LodNeighborSeparation &&
477 visited.Add(neighbor))
478 queue.Enqueue(neighbor);
479 }
480
481 if (group.Count < Math.Max(2, LodMinimumMembers)) continue;
482 var clusterId = Collapse(group);
483 clusters[clusterId].Automatic = true;
484 changed = true;
485 }
486
487 return changed;
488 }
