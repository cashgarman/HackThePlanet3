118 public static GraphRevisionChangeSummary Between(GraphSnapshot before, GraphSnapshot after)
119 {
120 if (after == null) throw new ArgumentNullException(nameof(after));
121
122 // Index both revisions by stable identity before classifying additions, edits, and deletions.
123 var oldNodes =
124 (before?.Nodes ?? Array.Empty<GraphNodeRecord>()).ToDictionary(node => node.Id, StringComparer.Ordinal);
125 var newNodes = after.Nodes.ToDictionary(node => node.Id, StringComparer.Ordinal);
126 var oldEdges =
127 (before?.Edges ?? Array.Empty<GraphEdgeRecord>()).ToDictionary(edge => edge.Id, StringComparer.Ordinal);
128 var newEdges = after.Edges.ToDictionary(edge => edge.Id, StringComparer.Ordinal);
129
130 bool NodeChanged(GraphNodeRecord oldNode, GraphNodeRecord newNode)
131 {
132 // Source movement and line shifts are metadata; content and semantic edges report actual code changes.
133 if (oldNode.Properties.TryGetValue("content_hash", out var prior) &&
134 newNode.Properties.TryGetValue("content_hash", out var next)) return !prior.Equals(next);
135 return oldNode.TypeId != newNode.TypeId || oldNode.Label != newNode.Label;
136 }
137
138 bool EdgeChanged(GraphEdgeRecord a, GraphEdgeRecord b)
139 {
140 return a.From != b.From || a.To != b.To ||
141 a.TypeId != b.TypeId || a.Status != b.Status || a.Confidence != b.Confidence ||
142 a.Properties.Count != b.Properties.Count || a.Properties.Any(pair =>
143 !b.Properties.TryGetValue(pair.Key, out var value) || !pair.Value.Equals(value));
144 }
145
146 var createdNodes = newNodes.Keys.Where(id => !oldNodes.ContainsKey(id))
147 .OrderBy(id => id, StringComparer.Ordinal).ToArray();
148 var modifiedNodes = newNodes.Values
149 .Where(node => oldNodes.TryGetValue(node.Id, out var old) &&
150 NodeChanged(old, node)).Select(node => node.Id)
151 .OrderBy(id => id, StringComparer.Ordinal).ToArray();
152 var deletedNodes = oldNodes.Keys.Where(id => !newNodes.ContainsKey(id))
153 .OrderBy(id => id, StringComparer.Ordinal).ToArray();
154 var createdEdges = newEdges.Keys.Where(id => !oldEdges.ContainsKey(id))
155 .OrderBy(id => id, StringComparer.Ordinal).ToArray();
156 var modifiedEdges = newEdges.Values
157 .Where(edge => oldEdges.TryGetValue(edge.Id, out var old) &&
158 EdgeChanged(old, edge)).Select(edge => edge.Id)
159 .OrderBy(id => id, StringComparer.Ordinal).ToArray();
160 var deletedEdges = oldEdges.Keys.Where(id => !newEdges.ContainsKey(id))
161 .OrderBy(id => id, StringComparer.Ordinal).ToArray();
162 return new GraphRevisionChangeSummary(createdNodes.Length, modifiedNodes.Length, deletedNodes.Length,
163 createdEdges.Length, modifiedEdges.Length, deletedEdges.Length)
164 {
165 CreatedNodeIds = Array.AsReadOnly(createdNodes), ModifiedNodeIds = Array.AsReadOnly(modifiedNodes),
166 DeletedNodeIds = Array.AsReadOnly(deletedNodes),
167 CreatedEdgeIds = Array.AsReadOnly(createdEdges), ModifiedEdgeIds = Array.AsReadOnly(modifiedEdges),
168 DeletedEdgeIds = Array.AsReadOnly(deletedEdges)
169 };
170 }

189 public void Replace(IEnumerable<GraphRevision> ordered)
190 {
191 var incoming = (ordered ?? Array.Empty<GraphRevision>()).ToArray();
192
193 // Validate the complete input before discarding the currently displayed revision sequence.
194 if (incoming.Any(revision => revision == null) ||
195 incoming.Select(revision => revision.Id).Distinct(StringComparer.Ordinal).Count() != incoming.Length)
196 throw new ArgumentException("Duplicate or null revision.", nameof(ordered));
197 revisions.Clear();
198 byId.Clear();
199 Selected = null;
200 foreach (var revision in incoming)
201 {
202 revisions.Add(revision);
203 byId.Add(revision.Id, revision);
204 }
205
206 BuildLanes();
207 Changed?.Invoke();
208 }

249 private void BuildLanes()
250 {
251 // Walk newest to oldest: a commit occupies its child's lane and spawns lanes for merge parents.
252 var active = new List<string>();
253 for (var i = revisions.Count - 1; i >= 0; i--)
254 {
255 var revision = revisions[i];
256 var lane = active.IndexOf(revision.Id);
257 if (lane < 0)
258 {
259 lane = active.Count;
260 active.Add(revision.Id);
261 }
262
263 var before = active.ToArray();
264 revision.Lane = lane;
265 revision.LaneCount = active.Count;
266 revision.LaneGlyphs = string.Concat(Enumerable.Range(0, active.Count)
267 .Select(index => index == lane ? "● " : "│ "));
268 active.RemoveAt(lane);
269
270 // Reserve lanes for merge parents without duplicating parents already tracked by another branch.
271 for (var p = revision.ParentIds.Count - 1; p >= 0; p--)
272 if (!active.Contains(revision.ParentIds[p]))
273 active.Insert(Math.Min(lane, active.Count), revision.ParentIds[p]);
274 var links = new List<GraphRevisionLaneLink>();
275
276 // Connect old lane positions to their new parent or continuing-branch positions.
277 for (var oldLane = 0; oldLane < before.Length; oldLane++)
278 if (oldLane == lane)
279 foreach (var parent in revision.ParentIds)
280 links.Add(new GraphRevisionLaneLink(lane, active.IndexOf(parent)));
281 else links.Add(new GraphRevisionLaneLink(oldLane, active.IndexOf(before[oldLane])));
282 revision.LaneLinks = links.AsReadOnly();
283 }
284 }
