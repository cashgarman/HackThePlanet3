68 public void Focus(IGraphNode node, float radius = 12)
69 {
70 if (node == null) return;
71 if (looking) Release();
72 subject = node;
73 Tracking = true;
74 trackedPosition = node.Position;
75 var orbitDistance = Mathf.Max(8, radius / Mathf.Tan(camera.fieldOfView * Mathf.Deg2Rad * .5f));
76 var currentDistance = Vector3.Distance(camera.transform.position, node.Position);
77 // Nodes already closer than the orbit distance keep the current distance as the new zoom level;
78 // distant nodes still frame at the orbit distance.
79 FrameAtDistance(node.Position,
80 currentDistance < orbitDistance ? Mathf.Max(1f, currentDistance) : orbitDistance);
81 }

109 private void FrameAtDistance(Vector3 center, float distance)
110 {
111 // Cancel residual input motion and ease toward the target along the current viewing direction.
112 panVelocity = Vector3.zero;
113 flightVelocity = Vector3.zero;
114 orbitCenter = center;
115 destination = center - camera.transform.forward * distance;
116 framing = true;
117 }

121 public void Overview(IGraphScene graph)
122 {
123 // Fit only visible nodes, retaining a small default framing volume for an empty scene.
124 StopTracking();
125 panVelocity = Vector3.zero;
126 var bounds = new Bounds(Vector3.zero, Vector3.one * 8);
127 var first = true;
128 foreach (var node in graph.Nodes.Values)
129 if (node.Visible)
130 {
131 if (first)
132 {
133 bounds = new Bounds(node.Position, Vector3.one * 6);
134 first = false;
135 }
136 else
137 {
138 bounds.Encapsulate(node.Position);
139 }
140 }
141
142 Frame(bounds.center, bounds.extents.magnitude * 1.3f);
143 }

147 public void Follow(bool hold)
148 {
149 // While held, update the target without translating the camera; otherwise follow its exact displacement.
150 if (!Tracking || subject == null) return;
151 if (hold)
152 {
153 trackedPosition = subject.Position;
154 orbitCenter = subject.Position;
155 framing = false;
156 return;
157 }
158
159 var delta = subject.Position - trackedPosition;
160 trackedPosition = subject.Position;
161 if (delta.sqrMagnitude <= 1e-12f) return;
162 camera.transform.position += delta;
163 destination += delta;
164 orbitCenter += delta;
165 }

171 public void Tick(bool blocked, IGraphNode selected, bool frameSelected = false)
172 {
173 // Ignore navigation input while UI owns the pointer or keyboard focus.
174 var mouse = Mouse.current;
175 var keyboard = Keyboard.current;
176 if (mouse == null || keyboard == null) return;
177 if (frameSelected) Focus(selected);
178 // Locked nodes orbit under right-drag. Fly mode stays available once the lock is released.
179 var trackOrbit = Tracking && subject != null && mouse.rightButton.isPressed && !blocked;
180 if (!trackOrbit && mouse.rightButton.wasPressedThisFrame && !blocked)
181 {
182 looking = true;
183 framing = false;
184 panVelocity = Vector3.zero;
185 CaptureCursor();
186 SyncAngles();
187 }
188
189 if (looking && !mouse.rightButton.isPressed) Release();
190 var delta = mouse.delta.ReadValue();
191 var dt = Mathf.Min(Time.unscaledDeltaTime, .05f);
192 var coastFlight = true;
193 if (trackOrbit)
194 {
195 // Orbit a tracked subject while clearing incompatible pan and flight momentum.
196 if (!orbiting)
197 {
198 orbiting = true;
199 panVelocity = Vector3.zero;
200 flightVelocity = Vector3.zero;
201 CaptureCursor();
202 SyncAngles();
203 }
204
205 framing = false;
206 orbitCenter = subject.Position;
207 camera.transform.RotateAround(orbitCenter, Vector3.up, delta.x * .25f);
208 camera.transform.RotateAround(orbitCenter, camera.transform.right, -delta.y * .25f);
209 SyncAngles();
210 Dolly(mouse.scroll.ReadValue().y);
211 coastFlight = false;
212 }
213 else if (orbiting)
214 {
215 orbiting = false;
216 Release();
217 }
218
219 if (looking)
220 {
221 // Integrate camera-relative flight with bounded pitch and scroll-adjustable speed.
222 yaw += delta.x * .13f;
223 pitch = Mathf.Clamp(pitch - delta.y * .13f, -88, 88);
224 camera.transform.rotation = Quaternion.Euler(pitch, yaw, 0);
225 var movement = new Vector3((keyboard.dKey.isPressed ? 1 : 0) - (keyboard.aKey.isPressed ? 1 : 0),
226 (keyboard.eKey.isPressed ? 1 : 0) - (keyboard.qKey.isPressed ? 1 : 0),
227 (keyboard.wKey.isPressed ? 1 : 0) - (keyboard.sKey.isPressed ? 1 : 0));
228 var wish = camera.transform.TransformDirection(Vector3.ClampMagnitude(movement, 1)) * Speed *
229 (keyboard.shiftKey.isPressed ? 3 : 1);
230 IntegrateFlight(dt, wish);
231 coastFlight = false;
232 var wheel = mouse.scroll.ReadValue().y;
233 if (wheel != 0) Speed = Mathf.Clamp(Speed * Mathf.Exp(wheel * .001f), 1, 300);
234 orbitCenter = camera.transform.position + camera.transform.forward * 20;
235 }
236 else if (!orbiting && !blocked)
237 {
238 if (mouse.middleButton.wasPressedThisFrame)
239 {
240 panVelocity = Vector3.zero;
241 flightVelocity = Vector3.zero;
242 }
243
244 if (mouse.middleButton.isPressed)
245 {
246 // Translate in the view plane and retain drag velocity for inertial panning after release.
247 coastFlight = false;
248 flightVelocity = Vector3.zero;
249 var move = (-camera.transform.right * delta.x - camera.transform.up * delta.y) * Speed * .0025f;
250 if (dt > 1e-5f) panVelocity = move / dt;
251 camera.transform.position += move;
252 orbitCenter += move;
253 framing = false;
254 }
255 else if (!framing && panVelocity.sqrMagnitude > 1e-4f)
256 {
257 camera.transform.position += panVelocity * dt;
258 orbitCenter += panVelocity * dt;
259 panVelocity *= Mathf.Exp(-PanInertiaDrag * dt);
260 if (panVelocity.sqrMagnitude < 1e-4f) panVelocity = Vector3.zero;
261 }
262
263 if (keyboard.altKey.isPressed && mouse.leftButton.isPressed)
264 {
265 // Orbit the current selection, or the existing orbit center when nothing is selected.
266 coastFlight = false;
267 panVelocity = Vector3.zero;
268 flightVelocity = Vector3.zero;
269 framing = false;
270 if (selected != null) orbitCenter = selected.Position;
271 camera.transform.RotateAround(orbitCenter, Vector3.up, delta.x * .25f);
272 camera.transform.RotateAround(orbitCenter, camera.transform.right, -delta.y * .25f);
273 SyncAngles();
274 }
275
276 var wheel = mouse.scroll.ReadValue().y;
277 if (wheel != 0)
278 {
279 coastFlight = false;
280 Dolly(wheel);
281 }
282 }
283 else if (!orbiting && panVelocity.sqrMagnitude > 1e-4f)
284 {
285 panVelocity = Vector3.zero;
286 }
287
288 if (coastFlight && !framing && !Tracking) IntegrateFlight(dt, Vector3.zero);
289 if (framing)
290 {
291 // Ease toward the framing destination until the remaining offset is negligible.
292 camera.transform.position =
293 Vector3.Lerp(camera.transform.position, destination, 1 - Mathf.Exp(-6 * dt));
294 if ((camera.transform.position - destination).sqrMagnitude < .01f) framing = false;
295 }
296 }

298 private void CaptureCursor()
299 {
300 // Save the host's cursor state only once across repeated look or orbit input frames.
301 if (!ownsCursor)
302 {
303 previousLock = Cursor.lockState;
304 previousCursorVisible = Cursor.visible;
305 ownsCursor = true;
306 }
307
308 Cursor.lockState = CursorLockMode.Locked;
309 Cursor.visible = false;
310 }

313 public void Release()
314 {
315 // Restore the cursor state that was active before this navigation controller captured it.
316 looking = false;
317 orbiting = false;
318 if (!ownsCursor) return;
319 Cursor.lockState = previousLock;
320 Cursor.visible = previousCursorVisible;
321 ownsCursor = false;
322 }

324 private void Dolly(float wheel)
325 {
326 // Scale dolly distance by distance to the orbit center while clearing conflicting inertia.
327 if (wheel == 0) return;
328 panVelocity = Vector3.zero;
329 flightVelocity = Vector3.zero;
330 framing = false;
331 wheel = Mathf.Clamp(wheel, -3f, 3f);
332 var distance = Vector3.Distance(camera.transform.position, orbitCenter);
333 if (distance < 6f) distance = 6f;
334 camera.transform.position += camera.transform.forward * distance * (1f - Mathf.Pow(1f - WheelZoom, wheel));
335 }
336
337 private void IntegrateFlight(float dt, Vector3 wish)
338 {
339 // Accelerate toward active input and use a separate exponential drag when coasting.
340 var rate = wish.sqrMagnitude > 1e-8f ? FlightAcceleration : FlightDrag;
341 flightVelocity = Vector3.Lerp(flightVelocity, wish, 1f - Mathf.Exp(-rate * dt));
342 if (wish.sqrMagnitude <= 1e-8f && flightVelocity.sqrMagnitude < 1e-4f) flightVelocity = Vector3.zero;
343 var step = flightVelocity * dt;
344 camera.transform.position += step;
345 orbitCenter += step;
346 }

351 public void Restore(Vector3 position, Quaternion rotation)
352 {
353 // Restore a saved pose without retaining tracking, framing, captured cursor, or prior velocities.
354 StopTracking();
355 framing = false;
356 panVelocity = Vector3.zero;
357 flightVelocity = Vector3.zero;
358 Release();
359 camera.transform.SetPositionAndRotation(position, rotation);
360 orbitCenter = position + camera.transform.forward * 20;
361 SyncAngles();
