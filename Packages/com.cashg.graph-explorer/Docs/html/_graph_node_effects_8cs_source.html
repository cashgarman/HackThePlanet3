40 private void BeginCamera(ScriptableRenderContext context, Camera camera)
41 {
42 // Enqueue effects only for this graph camera; progress remains available without glow or outlines.
43 if (camera != GetComponent<Camera>() || Renderer == null) return;
44 if (Renderer.VisibleProgressIndicators > 0)
45 {
46 progressPass ??= new ProgressPass();
47 progressPass.Renderer = Renderer;
48 camera.GetUniversalAdditionalCameraData().scriptableRenderer.EnqueuePass(progressPass);
49 }
50
51 if (Settings == null) Settings = Resources.Load<GraphVisualSettings>("GraphVisualSettings");
52 if (Settings == null || (!Settings.Outlines && !Settings.Bloom && !Settings.SelectionGlow)) return;
53 pass ??= new EffectsPass(Resources);
54 pass.Setup(this);
55 camera.GetUniversalAdditionalCameraData().scriptableRenderer.EnqueuePass(pass);
56 }

115 public override void RecordRenderGraph(RenderGraph graph, ContextContainer frameData)
116 {
117 var resources = frameData.Get<UniversalResourceData>();
118 if (resources.isActiveTargetBackBuffer) return;
119
120 // Translate visual settings into mask and composite shader parameters for this camera frame.
121 var settings = owner.Settings;
122 mask.SetVector("_SelectionColor", settings.SelectionColor.linear);
123 mask.SetFloat("_BloomIntensity", settings.Bloom ? Mathf.Max(0, settings.BloomIntensity) : 0);
124 mask.SetFloat("_SelectionEnabled", settings.SelectionGlow ? 1 : 0);
125 composite.SetVector("_OutlineColor", settings.OutlineColor.linear);
126 composite.SetVector("_OutlineParams", new Vector4(
127 settings.Outlines ? Mathf.Clamp(settings.OutlinePixels, .5f, 4) : 0,
128 Mathf.Clamp(settings.OutlineOpacity, 0, 1),
129 Mathf.Cos(Mathf.Clamp(settings.CreaseAngle, 5, 120) * Mathf.Deg2Rad),
130 Mathf.Max(.001f, settings.DepthThreshold)));
131 composite.SetVector("_SelectionColor", settings.SelectionColor.linear);
132 composite.SetFloat("_SelectionGlowIntensity",
133 settings.SelectionGlow ? Mathf.Max(0, settings.SelectionGlowIntensity) : 0);
134
135 var source = resources.activeColorTexture;
136
137 // Allocate matching normal/depth and glow seed targets, retaining the depth buffer's sample count.
138 var desc = graph.GetTextureDesc(source);
139 desc.name = "Graph node normals and depth";
140 desc.colorFormat = GraphicsFormat.R16G16B16A16_SFloat;
141 desc.depthBufferBits = DepthBits.None;
142 desc.msaaSamples = graph.GetTextureDesc(resources.activeDepthTexture).msaaSamples;
143 desc.bindTextureMS = false;
144 desc.clearBuffer = true;
145 desc.clearColor = Color.clear;
146 desc.filterMode = FilterMode.Point;
147 var normals = graph.CreateTexture(desc);
148 desc.name = "Graph node bloom and selection seed";
149 var seed = graph.CreateTexture(desc);
150 using (var builder =
151 graph.AddRasterRenderPass<MaskData>("Graph nodes: normal/depth and glow mask", out var data))
152 {
153 data.Renderer = owner.Renderer;
154 data.Material = mask;
155 builder.SetRenderAttachment(normals, 0);
156 builder.SetRenderAttachment(seed, 1);
157 // Reuse the actual opaque depth buffer: hidden nodes never leak glow.
158 builder.SetRenderAttachmentDepth(resources.activeDepthTexture, AccessFlags.Read);
159 builder.SetRenderFunc(static (MaskData d, RasterGraphContext ctx) =>
160 d.Renderer.DrawEffectsMask(ctx.cmd, d.Material));
161 }
162
163 // RGB is subtle bloom; alpha is the wider selected-node glow.
164 // Prefilter before wide blurs so sparse taps never create visible steps.
165 desc.msaaSamples = MSAASamples.None;
166 desc.width = Mathf.Max(1, desc.width / 2);
167 desc.height = Mathf.Max(1, desc.height / 2);
168 desc.filterMode = FilterMode.Bilinear;
169 desc.clearBuffer = false;
170 var bloom = seed;
171 var glow = seed;
172 desc.name = "Graph glow prefilter half";
173 var halfSeed = graph.CreateTexture(desc);
174 Blur(graph, seed, halfSeed, Vector4.zero, 2);
175 if (settings.Bloom || settings.SelectionGlow)
176 bloom = MakeBlur(graph, halfSeed, desc, Mathf.Clamp(settings.BloomRadius, 1, 16), "Graph bloom");
177 if (settings.SelectionGlow)
178 glow = MakeBlur(graph, halfSeed, desc, Mathf.Clamp(settings.SelectionGlowRadius, 4, 64),
179 "Graph selected glow");
180 var outputDesc = graph.GetTextureDesc(source);
181
182 // Composite outlines and filtered glow into a new color target for downstream post processing.
183 outputDesc.name = "Graph node outlines and bloom composite";
184 outputDesc.msaaSamples = MSAASamples.None;
185 outputDesc.clearBuffer = false;
186 var output = graph.CreateTexture(outputDesc);
187 using (var builder =
188 graph.AddRasterRenderPass<CompositeData>("Graph nodes: outlines and bloom composite",
189 out var data))
190 {
191 data.Source = source;
192 data.Normals = normals;
193 data.Seed = seed;
194 data.Bloom = bloom;
195 data.Glow = glow;
196 data.Material = composite;
197 builder.UseTexture(source);
198 builder.UseTexture(normals);
199 builder.UseTexture(seed);
200 builder.UseTexture(bloom);
201 builder.UseTexture(glow);
202 builder.SetRenderAttachment(output, 0);
203 builder.AllowGlobalStateModification(true);
204 builder.SetRenderFunc(static (CompositeData d, RasterGraphContext ctx) =>
205 {
206 ctx.cmd.SetGlobalTexture(NormalTexture, d.Normals);
207 ctx.cmd.SetGlobalTexture(SeedTexture, d.Seed);
208 ctx.cmd.SetGlobalTexture(BloomTexture, d.Bloom);
209 ctx.cmd.SetGlobalTexture(GlowTexture, d.Glow);
210 Blitter.BlitTexture(ctx.cmd, d.Source, new Vector4(1, 1, 0, 0), d.Material, 1);
211 });
212 }
213
214 resources.cameraColor = output;
215 }

217 private TextureHandle MakeBlur(RenderGraph graph, TextureHandle halfSeed, TextureDesc desc, float radius,
218 string name)
219 {
220 // Keep Gaussian sample spacing below ~1.5 texels even at the largest editor radius.
221 var levels = Mathf.Clamp(Mathf.CeilToInt(Mathf.Log(Mathf.Max(1, radius * .4f), 2)), 1, 5);
222 var filtered = halfSeed;
223 for (var level = 2; level <= levels; level++)
224 {
225 desc.width = Mathf.Max(1, desc.width / 2);
226 desc.height = Mathf.Max(1, desc.height / 2);
227 desc.name = name + " prefilter " + level;
228 var next = graph.CreateTexture(desc);
229 Blur(graph, filtered, next, Vector4.zero, 2);
230 filtered = next;
231 }
232
233 desc.name = name + " horizontal";
234
235 // Apply a separable blur to the chosen prefiltered level.
236 var horizontal = graph.CreateTexture(desc);
237 desc.name = name + " vertical";
238 var vertical = graph.CreateTexture(desc);
239 Blur(graph, filtered, horizontal, new Vector4(radius, 0, 0, 0));
240 Blur(graph, horizontal, vertical, new Vector4(0, radius, 0, 0));
241 return vertical;
242 }

244 private void Blur(RenderGraph graph, TextureHandle source, TextureHandle destination, Vector4 radius,
245 int shaderPass = 0)
246 {
247 // Declare source/destination dependencies before recording the matching blur or prefilter shader pass.
248 using var builder = graph.AddRasterRenderPass<BlurData>(
249 shaderPass == 2 ? "Graph nodes: downsample prefilter" : "Graph nodes: Gaussian blur", out var data);
250 data.Source = source;
251 data.Material = composite;
252 data.Blur = radius;
253 data.Pass = shaderPass;
254 builder.UseTexture(source);
255 builder.SetRenderAttachment(destination, 0);
256 builder.AllowGlobalStateModification(true);
257 builder.SetRenderFunc(static (BlurData d, RasterGraphContext ctx) =>
258 {
259 ctx.cmd.SetGlobalVector("_GraphBlurRadius", d.Blur);
260 Blitter.BlitTexture(ctx.cmd, d.Source, new Vector4(1, 1, 0, 0), d.Material, d.Pass);
261 });
262 }
