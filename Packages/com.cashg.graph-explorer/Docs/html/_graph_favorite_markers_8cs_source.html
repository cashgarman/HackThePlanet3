30 // Project rendered favorites and offset each marker diagonally from the node's visible radius.

41 var anchor =
42 RuntimePanelUtils.ScreenToPanel(root.panel, new Vector2(screen.x, Screen.height - screen.y));
43 var offset = radius(node, anchor) * .7071f;
44 var center = new Vector2(anchor.x + offset, anchor.y - offset);
45 // Markers never sit over the header, rail, or dock. Hidden instead of clamped, so no marker points at the wrong node.
46 if (center.x < area.xMin - half || center.x > area.xMax + half || center.y < area.yMin - half ||
47 center.y > area.yMax + half) continue;
48 var mark = Marker(used++, favoriteMarks, worldLabels, color);
49 mark.style.display = DisplayStyle.Flex;
50 mark.style.left = Mathf.Round(center.x - half);
51 mark.style.top = Mathf.Round(center.y - half);
52 }
53 }
54
55 // Keep unused markers pooled but hidden when fewer favorites are visible this frame.
56 for (var i = used; i < favoriteMarks.Count; i++) favoriteMarks[i].style.display = DisplayStyle.None;
57 }
