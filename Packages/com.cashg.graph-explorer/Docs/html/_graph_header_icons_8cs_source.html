237 public static void Paint(Painter2D painter, string icon, Rect rect, Color color)
238 {
239 if (rect.width < 1f || rect.height < 1f || !sources.TryGetValue(icon, out var spec)) return;
240
241 // Compile each source path once and reuse its normalized drawing operations across icon sizes.
242 if (!cached.TryGetValue(icon, out var ops))
243 {
244 ops = Compile(spec.Path);
245 cached[icon] = ops;
246 }
247
248 var inner = new Rect(rect.x + .5f, rect.y + .5f, Mathf.Max(1f, rect.width - 1f),
249 Mathf.Max(1f, rect.height - 1f));
250
251 // Preserve the source aspect ratio and center the path within a half-pixel inset.
252 var scale = Mathf.Min(inner.width / spec.Width, inner.height / spec.Height);
253 var ox = inner.x + (inner.width - spec.Width * scale) * .5f;
254 var oy = inner.y + (inner.height - spec.Height * scale) * .5f;
255 painter.fillColor = color;
256 painter.BeginPath();
257 for (var i = 0; i < ops.Count; i++)
258 {
259 var op = ops[i];
260 if (op.Kind == Move) painter.MoveTo(Map(op.A, ox, oy, scale));
261 else if (op.Kind == Line) painter.LineTo(Map(op.A, ox, oy, scale));
262 else if (op.Kind == Cubic)
263 painter.BezierCurveTo(Map(op.A, ox, oy, scale), Map(op.B, ox, oy, scale), Map(op.C, ox, oy, scale));
264 else painter.ClosePath();
265 }
266
267 painter.Fill();
268 }

275 private static List<DrawOp> Compile(string path)
276 {
277 // Parse the SVG path once into drawing operations reused by later paints.
278 var tokens = Tokenize(path);
279 var ops = new List<DrawOp>();
280 var index = 0;
281 var command = '\0';
282 float cx = 0f, cy = 0f, sx = 0f, sy = 0f;
283 Vector2? cubic = null, quad = null;
284 while (index < tokens.Count)
285 {
286 if (tokens[index].Command != '\0') command = tokens[index++].Command;
287 if (command == '\0') break;
288 var key = char.ToUpperInvariant(command);
289 if (key == 'Z')
290 {
291 // Closing a subpath resets its current point and clears smooth-curve reflection state.
292 ops.Add(new DrawOp { Kind = Close });
293 cx = sx;
294 cy = sy;
295 cubic = quad = null;
296 continue;
297 }
298
299 if (index >= tokens.Count || tokens[index].Command != '\0') break;
300 var relative = char.IsLower(command);
301 if (key == 'M')
302 {
303 // Additional coordinate pairs after a move are implicit line commands under SVG path rules.
304 float x = Read(tokens, ref index), y = Read(tokens, ref index);
305 if (relative)
306 {
307 x += cx;
308 y += cy;
309 }
310
311 ops.Add(new DrawOp { Kind = Move, A = new Vector2(x, y) });
312 cx = sx = x;
313 cy = sy = y;
314 cubic = quad = null;
315 command = relative ? 'l' : 'L';
316 continue;
317 }
318
319 if (key == 'L')
320 {
321 float x = Read(tokens, ref index), y = Read(tokens, ref index);
322 if (relative)
323 {
324 x += cx;
325 y += cy;
326 }
327
328 ops.Add(new DrawOp { Kind = Line, A = new Vector2(x, y) });
329 cx = x;
330 cy = y;
331 }
332 else if (key == 'H')
333 {
334 var x = Read(tokens, ref index);
335 if (relative) x += cx;
336 ops.Add(new DrawOp { Kind = Line, A = new Vector2(x, cy) });
337 cx = x;
338 }
339 else if (key == 'V')
340 {
341 var y = Read(tokens, ref index);
342 if (relative) y += cy;
343 ops.Add(new DrawOp { Kind = Line, A = new Vector2(cx, y) });
344 cy = y;
345 }
346 else if (key == 'C')
347 {
348 float x1 = Read(tokens, ref index),
349 y1 = Read(tokens, ref index),
350 x2 = Read(tokens, ref index),
351 y2 = Read(tokens, ref index),
352 x = Read(tokens, ref index),
353 y = Read(tokens, ref index);
354 if (relative)
355 {
356 x1 += cx;
357 y1 += cy;
358 x2 += cx;
359 y2 += cy;
360 x += cx;
361 y += cy;
362 }
363
364 ops.Add(new DrawOp
365 {
366 Kind = Cubic, A = new Vector2(x1, y1), B = new Vector2(x2, y2), C = new Vector2(x, y)
367 });
368 cubic = new Vector2(x2, y2);
369 cx = x;
370 cy = y;
371 }
372 else if (key == 'S')
373 {
374 // Reflect the prior cubic control point for a smooth continuation when one is available.
375 float x2 = Read(tokens, ref index),
376 y2 = Read(tokens, ref index),
377 x = Read(tokens, ref index),
378 y = Read(tokens, ref index);
379 if (relative)
380 {
381 x2 += cx;
382 y2 += cy;
383 x += cx;
384 y += cy;
385 }
386
387 var x1 = cubic.HasValue ? 2f * cx - cubic.Value.x : cx;
388 var y1 = cubic.HasValue ? 2f * cy - cubic.Value.y : cy;
389 ops.Add(new DrawOp
390 {
391 Kind = Cubic, A = new Vector2(x1, y1), B = new Vector2(x2, y2), C = new Vector2(x, y)
392 });
393 cubic = new Vector2(x2, y2);
394 cx = x;
395 cy = y;
396 }
397 else if (key == 'Q')
398 {
399 float x1 = Read(tokens, ref index),
400 y1 = Read(tokens, ref index),
401 x = Read(tokens, ref index),
402 y = Read(tokens, ref index);
403 if (relative)
404 {
405 x1 += cx;
406 y1 += cy;
407 x += cx;
408 y += cy;
409 }
410
411 AddQuadratic(ops, new Vector2(cx, cy), new Vector2(x1, y1), new Vector2(x, y));
412 quad = new Vector2(x1, y1);
413 cx = x;
414 cy = y;
415 }
416 else if (key == 'T')
417 {
418 // Reflect the preceding quadratic control point before converting the curve to cubic form.
419 float x = Read(tokens, ref index), y = Read(tokens, ref index);
420 if (relative)
421 {
422 x += cx;
423 y += cy;
424 }
425
426 var x1 = quad.HasValue ? 2f * cx - quad.Value.x : cx;
427 var y1 = quad.HasValue ? 2f * cy - quad.Value.y : cy;
428 AddQuadratic(ops, new Vector2(cx, cy), new Vector2(x1, y1), new Vector2(x, y));
429 quad = new Vector2(x1, y1);
430 cx = x;
431 cy = y;
432 }
433 else if (key == 'A')
434 {
435 float rx = Read(tokens, ref index),
436 ry = Read(tokens, ref index),
437 phi = Read(tokens, ref index),
438 fa = Read(tokens, ref index),
439 fs = Read(tokens, ref index),
440 x = Read(tokens, ref index),
441 y = Read(tokens, ref index);
442 if (relative)
443 {
444 x += cx;
445 y += cy;
446 }
447
448 AddArc(ops, new Vector2(cx, cy), rx, ry, phi, fa != 0f, fs != 0f, new Vector2(x, y));
449 cx = x;
450 cy = y;
451 }
452
453 if (key != 'C' && key != 'S') cubic = null;
454 if (key != 'Q' && key != 'T') quad = null;
455 }
456
457 return ops;
458 }

462 // Convert a quadratic curve to equivalent cubic controls supported by Painter2D.
463 var c1 = start + (control - start) * (2f / 3f);
464 var c2 = end + (control - end) * (2f / 3f);
465 ops.Add(new DrawOp { Kind = Cubic, A = c1, B = c2, C = end });

468 private static void AddArc(List<DrawOp> ops, Vector2 start, float rx, float ry, float phiDegrees, bool large,
469 bool sweep, Vector2 end)
470 {
471 if (Vector2.Distance(start, end) < 1e-4f) return;
472 if (Mathf.Abs(rx) < 1e-4f || Mathf.Abs(ry) < 1e-4f)
473 {
474 ops.Add(new DrawOp { Kind = Line, A = end });
475 return;
476 }
477
478 var phi = phiDegrees * Mathf.Deg2Rad;
479
480 // Convert SVG endpoint coordinates into the rotated ellipse's coordinate frame.
481 float cos = Mathf.Cos(phi), sin = Mathf.Sin(phi);
482 float dx = (start.x - end.x) * .5f, dy = (start.y - end.y) * .5f;
483 var x1 = cos * dx + sin * dy;
484 var y1 = -sin * dx + cos * dy;
485 rx = Mathf.Abs(rx);
486 ry = Mathf.Abs(ry);
487 var lambda = x1 * x1 / (rx * rx) + y1 * y1 / (ry * ry);
488 if (lambda > 1f)
489 {
490 // Expand undersized radii so the ellipse can reach both endpoints.
491 var root = Mathf.Sqrt(lambda);
492 rx *= root;
493 ry *= root;
494 }
495
496 float rxsq = rx * rx, rysq = ry * ry, x1sq = x1 * x1, y1sq = y1 * y1;
497
498 // Choose the ellipse center and signed sweep from the large-arc and sweep flags.
499 var num = rxsq * rysq - rxsq * y1sq - rysq * x1sq;
500 var den = rxsq * y1sq + rysq * x1sq;
501 var coef = den == 0f ? 0f : Mathf.Sqrt(Mathf.Max(0f, num / den));
502 if (large == sweep) coef = -coef;
503 float cx1 = coef * rx * y1 / ry, cy1 = coef * -ry * x1 / rx;
504 var centerX = cos * cx1 - sin * cy1 + (start.x + end.x) * .5f;
505 var centerY = sin * cx1 + cos * cy1 + (start.y + end.y) * .5f;
506 var theta1 = Angle(1f, 0f, (x1 - cx1) / rx, (y1 - cy1) / ry);
507 var delta = Angle((x1 - cx1) / rx, (y1 - cy1) / ry, (-x1 - cx1) / rx, (-y1 - cy1) / ry);
508 if (!sweep && delta > 0f) delta -= Mathf.PI * 2f;
509 else if (sweep && delta < 0f) delta += Mathf.PI * 2f;
510 var segments = Mathf.Max(1, Mathf.CeilToInt(Mathf.Abs(delta) / (Mathf.PI * .5f) - 1e-4f));
511
512 // Approximate the arc with cubic segments spanning at most a quarter turn each.
513 var step = delta / segments;
514 for (var i = 0; i < segments; i++)
515 {
516 var a0 = theta1 + step * i;
517 var a1 = a0 + step;
518 var alpha = Mathf.Tan(step * .5f) * 4f / 3f;
519 var c1 = ArcPoint(centerX, centerY, rx, ry, cos, sin, a0, -alpha * Mathf.Sin(a0),
520 alpha * Mathf.Cos(a0));
521 var c2 = ArcPoint(centerX, centerY, rx, ry, cos, sin, a1, alpha * Mathf.Sin(a1),
522 -alpha * Mathf.Cos(a1));
523 var point = ArcPoint(centerX, centerY, rx, ry, cos, sin, a1, 0f, 0f);
524 ops.Add(new DrawOp { Kind = Cubic, A = c1, B = c2, C = point });
525 }
526 }

536 private static float Angle(float ux, float uy, float vx, float vy)
537 {
538 // Clamp the dot product for numerical stability and recover orientation from the cross product.
539 var length = Mathf.Sqrt(ux * ux + uy * uy) * Mathf.Sqrt(vx * vx + vy * vy);
540 var value = length == 0f ? 1f : Mathf.Clamp((ux * vx + uy * vy) / length, -1f, 1f);
541 var angle = Mathf.Acos(value);
542 if (ux * vy - uy * vx < 0f) angle = -angle;
543 return angle;
544 }

551 private static List<Token> Tokenize(string path)
552 {
553 // Separate SVG command letters from signed decimal numbers, including scientific notation.
554 var tokens = new List<Token>(path.Length / 2);
555 var i = 0;
556 while (i < path.Length)
557 {
558 var c = path[i];
559 if (char.IsWhiteSpace(c) || c == ',')
560 {
561 i++;
562 continue;
563 }
564
565 if (char.IsLetter(c))
566 {
567 tokens.Add(new Token { Command = c });
568 i++;
569 continue;
570 }
571
572 var start = i;
573 if (c == '+' || c == '-') i++;
574 if (i < path.Length && path[i] == '.')
575 {
576 i++;
577 while (i < path.Length && char.IsDigit(path[i])) i++;
578 }
579 else
580 {
581 while (i < path.Length && char.IsDigit(path[i])) i++;
582 if (i < path.Length && path[i] == '.')
583 {
584 i++;
585 while (i < path.Length && char.IsDigit(path[i])) i++;
586 }
587 }
588
589 if (i < path.Length && (path[i] == 'e' || path[i] == 'E'))
590 {
591 var exponent = i + 1;
592 if (exponent < path.Length && (path[exponent] == '+' || path[exponent] == '-')) exponent++;
593 if (exponent < path.Length && char.IsDigit(path[exponent]))
594 {
595 i = exponent + 1;
596 while (i < path.Length && char.IsDigit(path[i])) i++;
597 }
598 }
599
600 tokens.Add(new Token
601 {
602 Number = float.Parse(path.Substring(start, i - start), CultureInfo.InvariantCulture)
603 });
604 }
605
606 return tokens;
607 }
