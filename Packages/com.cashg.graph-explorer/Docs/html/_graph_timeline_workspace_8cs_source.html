174 public void RestoreFrom(GraphTimelineWorkspace source)
175 {
176 if (ReferenceEquals(this, source)) return;
177
178 // Copy portable bounds and snapshots while leaving transient progress and errors local.
179 SelectionStartUtc = source.SelectionStartUtc;
180 SelectionEndUtc = source.SelectionEndUtc;
181 ViewStartUtc = source.ViewStartUtc;
182 ViewEndUtc = source.ViewEndUtc;
183 cached.Clear();
184 cachedIds.Clear();
185 foreach (var item in source.CachedEvents)
186 {
187 cached.Add(Key(item.Context, item.Event.Id), item);
188 cachedIds.Add(item.Event.Id);
189 }

195 public IEnumerator CacheSelection(GraphDocument document, GraphTimelineEventTrack track,
196 IGraphTimelineEventProvider provider, CancellationToken token)
197 {
198 if (IsCaching) throw new InvalidOperationException("A timeline cache operation is already running.");
199
200 // Freeze the selected descriptors so range edits cannot change an in-flight cache operation.
201 var selected = SelectedEvents(track?.Events).ToArray();
202 IsCaching = true;
203 Completed = 0;
204 Total = selected.Length;
205 Error = null;
206 Changed?.Invoke();
207 try
208 {
209 var context = provider?.CacheContext ?? "";
210 foreach (var item in selected)
211 {
212 if (token.IsCancellationRequested) yield break;
213
214 // Reuse completed work in this context; cancellation and later failures keep earlier snapshots.
215 if (!TryGet(context, item.Id, out var snapshot))
216 {
217 if (provider == null)
218 {
219 Error = "No event loader is attached.";
220 yield break;
221 }
222
223 Exception failure = null;
224 yield return provider.LoadEvent(item.Id, token, value => snapshot = value,
225 error => failure = error);
226 if (token.IsCancellationRequested) yield break;
227 if (failure != null || snapshot == null)
228 {
229 Error = failure?.Message ?? "The event loader returned no snapshot.";
230 yield break;
231 }
232
233 if (context != provider.CacheContext)
234 {
235 // Do not mix snapshots from a provider that changed sources while the load yielded.
236 Error = "The event provider changed during caching. Retry the selection.";
237 yield break;
238 }
239
240 if (snapshot.DatasetId != document.DatasetId)
241 {
242 Error = "The event belongs to another graph.";
243 yield break;
244 }
245
246 // Validate before admitting an independently loaded snapshot to the portable cache.
247 var validation = new GraphDocument(document.DatasetId);
248 validation.ReplaceSnapshot(snapshot);
249 Store(item, context, snapshot);
250 }
251
252 Completed++;
253 Changed?.Invoke();
254 yield return null;
255 }
256 // Session events and motion are already part of GraphJson's baseline/change history.
257 }
258 finally
259 {
260 IsCaching = false;
261 Changed?.Invoke();
262 }
263 }

267 public sealed class GraphSnapshotCache
268 {
269 private readonly int capacity;
270
271 private readonly Dictionary<string, LinkedListNode<KeyValuePair<string, GraphSnapshot>>> entries =
272 new(StringComparer.Ordinal);
273
274 private readonly LinkedList<KeyValuePair<string, GraphSnapshot>> order = new();
275
278 public GraphSnapshotCache(int capacity = 8)
279 {
280 if (capacity < 1) throw new ArgumentOutOfRangeException(nameof(capacity));
281 this.capacity = capacity;
282 }
283
288 public bool TryGet(string key, out GraphSnapshot snapshot)
289 {
290 if (!entries.TryGetValue(key, out var node))
291 {
292 snapshot = null;
293 return false;
294 }
295
296 order.Remove(node);
297
298 // A successful lookup promotes the entry to the front of the eviction order.
299 order.AddFirst(node);
300 snapshot = node.Value.Value;
301 return true;
302 }
303
307 public void Store(string key, GraphSnapshot snapshot)
308 {
309 if (entries.TryGetValue(key, out var old))
310 {
311 order.Remove(old);
312 entries.Remove(key);
313 }
314
315 entries.Add(key, order.AddFirst(new KeyValuePair<string, GraphSnapshot>(key, snapshot)));
316
317 // Evict from the least-recently-used end until the configured capacity is respected.
318 while (entries.Count > capacity)
319 {
320 entries.Remove(order.Last.Value.Key);
321 order.RemoveLast();
322 }
323 }
324
326 public void Clear()
327 {
328 entries.Clear();
329 order.Clear();
330 }
331 }
