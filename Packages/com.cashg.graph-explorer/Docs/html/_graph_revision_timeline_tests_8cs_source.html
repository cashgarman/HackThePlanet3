10 [Test] public void BranchLanesKeepMergeParentsAndNonMonotonicDates()
11 {
12 // Preserve ancestry order even when commit timestamps run backward or are absent for the working tree.
13 var now = DateTimeOffset.UtcNow;
14 var root = new GraphRevision("root", null, now, "Root");
15 var left = new GraphRevision("left", new[] { "root" }, now.AddDays(-1), "Left");
16 var right = new GraphRevision("right", new[] { "root" }, now.AddDays(1), "Right");
17 var merge = new GraphRevision("merge", new[] { "left", "right" }, now, "Merge");
18 var working = new GraphRevision("work", new[] { "merge" }, null, "Working Tree");
19 var timeline = new GraphRevisionTimeline(); timeline.Replace(new[] { root, left, right, merge, working });
20 Assert.That(merge.LaneLinks.Count, Is.EqualTo(2));
21 Assert.That(left.Lane, Is.Not.EqualTo(right.Lane));
22 Assert.That(left.TimestampUtc, Is.EqualTo(now.AddDays(-1)));
23 Assert.That(working.TimestampUtc, Is.Null);
24 string requested = null; timeline.SelectionRequested += revision => requested = revision.Id;
25 timeline.RequestSelection("left"); Assert.That(requested, Is.EqualTo("left")); Assert.That(timeline.Selected, Is.Null);
26 timeline.ConfirmSelection("left"); Assert.That(timeline.Selected, Is.SameAs(left));
27 Assert.Throws<ArgumentException>(() => timeline.Replace(new[] { root, root }));
28 Assert.That(timeline.Revisions.Count, Is.EqualTo(5), "Invalid imports must retain the previous history.");
29 }

31 [Test] public void DeltaTracksBodyEditsDeletionsRenamesAndReferenceCounts()
32 {
33 GraphNodeRecord Symbol(string id, string hash, int line) => new GraphNodeRecord(id, "member", id,
34 new Dictionary<string, GraphValue> { ["content_hash"] = GraphValue.From(hash), ["line"] = GraphValue.From(line) });
35 var before = new GraphSnapshot("test", "Test", new[] { Symbol("same", "one", 1), Symbol("edit", "old", 2), Symbol("gone", "x", 3) },
36 new[] { new GraphEdgeRecord("call", "same", "edit", "calls", properties: new Dictionary<string, GraphValue> { ["count"] = GraphValue.From(1) }) });
37 var after = new GraphSnapshot("test", "Test", new[] { Symbol("same", "one", 200), Symbol("edit", "new", 2), Symbol("renamed", "x", 3) },
38 new[] { new GraphEdgeRecord("call", "same", "edit", "calls", properties: new Dictionary<string, GraphValue> { ["count"] = GraphValue.From(2) }) });
39 var changes = GraphRevisionChangeSummary.Between(before, after);
40
41 // Line shifts are metadata; changed content hashes and relationship counts identify semantic edits.
42 Assert.That(changes.CreatedNodeIds, Is.EqualTo(new[] { "renamed" }));
43 Assert.That(changes.ModifiedNodeIds, Is.EqualTo(new[] { "edit" }));
44 Assert.That(changes.DeletedNodeIds, Is.EqualTo(new[] { "gone" }));
45 Assert.That(changes.ModifiedEdgeIds, Is.EqualTo(new[] { "call" }));
46 Assert.That(GraphRevisionChangeSummary.Between(null, after).NodesCreated, Is.EqualTo(3));
47 }
