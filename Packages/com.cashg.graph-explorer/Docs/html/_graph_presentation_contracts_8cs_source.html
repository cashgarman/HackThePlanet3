160 public GraphNodeTypeDefinition Type(IGraphScene scene, IGraphNode node)
161 {
162 var id = node.Record.TypeId;
163 if (scene.Schema.NodeTypes.TryGetValue(id, out var type)) return type;
164
165 // Cache neutral fallback definitions without assigning domain meaning to unknown types.
166 if (!fallbackTypes.TryGetValue(id, out type))
167 fallbackTypes.Add(id, type = new GraphNodeTypeDefinition(id, id));
168 return type;
169 }

177 public bool TryMetric(IGraphScene scene, IGraphNode node, out float value, out string partition)
178 {
179 value = 0;
180 partition = "";
181
182 // Accept only applicable, finite numeric values from the configured metric field.
183 if (MetricApplies != null && !MetricApplies(node)) return false;
184 if (string.IsNullOrEmpty(MetricField) ||
185 !node.Record.Properties.TryGetValue(MetricField, out var property) ||
186 property.Type != GraphValueType.Number) return false;
187 value = (float)property.Number;
188 if (float.IsNaN(value) || float.IsInfinity(value)) return false;
189
190 // Partition normalization by the schema's companion field when one is declared.
191 if (scene.Schema.Fields.TryGetValue(MetricField, out var field) &&
192 !string.IsNullOrEmpty(field.PartitionBy) &&
193 node.Record.Properties.TryGetValue(field.PartitionBy, out var key)) partition = key.ToString();
194 return true;
195 }
196
202 public Color CategoryColor(IGraphScene scene, IGraphNode node, Color fallback)
203 {
204 // Prefer a registered category color, then the host fallback, for applicable nodes.
205 if (CategoryApplies != null && !CategoryApplies(node)) return fallback;
206 return scene.Schema.Fields.TryGetValue(CategoryField, out var field) &&
207 node.Record.Properties.TryGetValue(CategoryField, out var value) &&
208 field.Categories.TryGetValue(value.ToString(), out var category)
209 ? category.Color
210 : CategoryFallback?.Invoke(node) ?? fallback;
211 }
