79 public void Start()
80 {
81 // Launch an isolated stdio process and deliver complete output lines through transport events.
82 var info = new ProcessStartInfo
83 {
84 FileName = executable, Arguments = "app-server --stdio", UseShellExecute = false,
85 RedirectStandardInput = true, RedirectStandardOutput = true, RedirectStandardError = true,
86 CreateNoWindow = true
87 };
88 process = new Process { StartInfo = info, EnableRaisingEvents = true };
89 process.OutputDataReceived += (_, e) =>
90 {
91 if (e.Data != null) Line?.Invoke(e.Data);
92 else if (!disposed) Closed?.Invoke("Codex output closed.");
93 };
94 // Always drain stderr so a full pipe cannot stall the server. Protocol errors arrive on stdout.
95 process.ErrorDataReceived += (_, e) => { };
96 process.Start();
97 process.BeginOutputReadLine();
98 process.BeginErrorReadLine();
99 }
100
103 public void Send(string line)
104 {
105 process.StandardInput.WriteLine(line);
106 process.StandardInput.Flush();
107 }
108
110 public void Dispose()
111 {
112 disposed = true;
113 if (process == null) return;
114
115 // The process may exit between the state check and Kill; either way, release its handles afterward.
116 try
117 {
118 if (!process.HasExited) process.Kill();
119 }
120 catch (InvalidOperationException)
121 {
122 }
123
124 process.Dispose();
125 process = null;
126 }

209 public void Request(string method, object parameters, Action<JsonElement> done, Action<string> failed)
210 {
211 if (disposed || closed)
212 {
213 failed("Codex connection is closed.");
214 return;
215 }
216
217 var id = ++nextId;
218
219 // Register the response callbacks before sending so synchronous transport replies can be matched.
220 pending[id] = new Pending { Done = done, Failed = failed, Deadline = DateTime.UtcNow.AddMinutes(2) };
221 try
222 {
223 transport.Send(JsonSerializer.Serialize(new { jsonrpc = "2.0", id, method, @params = parameters }));
224 }
225 catch (Exception e)
226 {
227 pending.Remove(id);
228 failed(e.Message);
229 }
230 }

261 public void Pump()
262 {
263 if (disposed) return;
264 // Bound each frame's work; remaining messages stay queued.
265 for (var i = 0; i < 200 && lines.TryDequeue(out var line); i++)
266 try
267 {
268 using var doc = JsonDocument.Parse(line);
269 var root = doc.RootElement;
270 var hasId = root.TryGetProperty("id", out var id);
271 if (root.TryGetProperty("method", out var method))
272 {
273 // Clone JSON values that must outlive the document disposed at the end of this iteration.
274 var parameters = root.TryGetProperty("params", out var p) ? p.Clone() : default;
275 Message?.Invoke(method.GetString(), parameters, hasId ? id.Clone() : null);
276 }
277 else if (hasId && id.ValueKind == JsonValueKind.Number && id.TryGetInt64(out var number) &&
278 pending.TryGetValue(number, out var request))
279 {
280 pending.Remove(number);
281 if (root.TryGetProperty("error", out var error)) request.Failed(error.ToString());
282 else if (root.TryGetProperty("result", out var result)) request.Done(result.Clone());
283 else request.Failed("Codex response has no result.");
284 }
285 }
286 catch (JsonException)
287 {
288 closures.Enqueue("Invalid JSON from Codex app-server.");
289 }
290
291 // Expire requests without retrying operations whose external outcome may already have changed.
292 foreach (var pair in pending.Where(p => p.Value.Deadline < DateTime.UtcNow).ToArray())
293 {
294 pending.Remove(pair.Key);
295 pair.Value.Failed("Codex response timed out; reconcile before retrying.");
296 }
297
298 if (closures.TryDequeue(out var reason) && !closed)
299 {
300 // Fail all outstanding callbacks once and publish the disconnected state on the host thread.
301 closed = true;
302 Ready = false;
303 var requests = pending.Values.ToArray();
304 pending.Clear();
305 foreach (var request in requests) request.Failed(reason);
306 Disconnected?.Invoke(reason);
307 }
308 }

315 }
