103 internal static class GraphRecordCopy
104 {
105 public static string Required(string value, string name)
106 {
107 return !string.IsNullOrWhiteSpace(value)
108 ? value
109 : throw new ArgumentException("An opaque nonempty ID is required.", name);
110 }
111
112 public static Dictionary<string, GraphValue> Properties(IEnumerable<KeyValuePair<string, GraphValue>> values)
113 {
114 // Copy caller-owned properties with ordinal keys, rejecting empty or duplicate identities.
115 var result = new Dictionary<string, GraphValue>(StringComparer.Ordinal);
116 if (values != null)
117 foreach (var pair in values)
118 result.Add(Required(pair.Key, "property"), pair.Value);
119 return result;
120 }
121
122 public static IReadOnlyList<string> Sources(IEnumerable<string> values)
123 {
124 // Validate and deduplicate evidence identifiers before exposing an immutable copy.
125 return Array.AsReadOnly(
126 (values ?? Array.Empty<string>()).Select(value => Required(value, "source"))
127 .Distinct(StringComparer.Ordinal).ToArray());
128 }

181 public GraphNodeRecord WithProperty(string key, GraphValue? value)
182 {
183 GraphRecordCopy.Required(key, nameof(key));
184
185 // Copy the property map so setting or removing one entry leaves the original record intact.
186 var values = GraphRecordCopy.Properties(Properties);
187 if (value.HasValue) values[key] = value.Value;
188 else values.Remove(key);
189 return new GraphNodeRecord(Id, TypeId, Label, values, Sources);
