29 public GraphAtmosphere(Camera camera, GraphResourceConfiguration resources = null)
30 {
31 this.camera = camera != null ? camera : throw new ArgumentNullException(nameof(camera));
32 resources ??= new GraphResourceConfiguration();
33 var skyShader = resources.Load<Shader>("GraphSky");
34 var fogShader = resources.Load<Shader>("GraphFog");
35 if (skyShader == null || fogShader == null)
36 throw new InvalidOperationException("Missing Resources/GraphSky or Resources/GraphFog shader.");
37 var skySource = resources.Load<Material>("GraphSkyMaterial");
38
39 // Clone imported materials and create transient scene objects owned by this atmosphere instance.
40 var fogSource = resources.Load<Material>("GraphFogMaterial");
41 skyMaterial = skySource != null ? new Material(skySource) : new Material(skyShader);
42 fogMaterial = fogSource != null ? new Material(fogSource) : new Material(fogShader);
43 skyMaterial.name = "Graph sky";
44 fogMaterial.name = "Graph fog";
45 skyMaterial.renderQueue = (int)RenderQueue.Background;
46 fogMaterial.renderQueue = (int)RenderQueue.Transparent;
47 skyMaterial.hideFlags = HideFlags.HideAndDontSave;
48 fogMaterial.hideFlags = HideFlags.HideAndDontSave;
49
50 skyMesh = UvSphere(28, 18);
51 bankMesh = Quad();
52 skyMesh.hideFlags = HideFlags.HideAndDontSave;
53 bankMesh.hideFlags = HideFlags.HideAndDontSave;
54 sky = new GameObject("Graph Sky") { hideFlags = HideFlags.DontSave };
55 sky.transform.localScale = Vector3.one * 2200f;
56 renderers.Add(AddRenderer(sky, skyMesh, skyMaterial));
57 rig = new GameObject("Graph Fog") { hideFlags = HideFlags.DontSave };
58 banks = CreateBanks();
59 PlaceSky();
60 foreach (var bank in banks) Place(bank);
61 RenderPipelineManager.beginCameraRendering += BeginCamera;
62 }

87 public void Tick(bool reducedMotion)
88 {
89 // Freeze animation time in reduced-motion mode while keeping the sky centered on the camera.
90 if (disposed) throw new ObjectDisposedException(nameof(GraphAtmosphere));
91 var dt = reducedMotion ? 0f : Time.unscaledDeltaTime;
92 clock += dt;
93 skyMaterial.SetFloat("_GraphAtmosphereTime", clock);
94 fogMaterial.SetFloat("_GraphAtmosphereTime", clock);
95 PlaceSky();
96 for (var i = 0; i < banks.Length; i++)
97 {
98 banks[i].Angle += banks[i].Speed * dt;
99 Place(banks[i]);
100 }
101 }

115 private Bank[] CreateBanks()
116 {
117 // Build each atmosphere layer from its own mesh and material configuration.
118 var random = new Random(17);
119 var created = new Bank[12];
120 for (var i = 0; i < created.Length; i++)
121 {
122 // Use a fixed seed for repeatable near/far fog placement and independent per-renderer shader values.
123 var distant = i < 7;
124 var radius = Lerp(random, distant ? 250f : 125f, distant ? 460f : 195f);
125 var size = Lerp(random, distant ? 150f : 78f, distant ? 240f : 130f);
126 var color = Color.Lerp(new Color(0.16f, 0.26f, 0.40f), new Color(0.26f, 0.38f, 0.54f),
127 (float)random.NextDouble());
128 if (i % 4 == 0) color = Color.Lerp(color, new Color(0.36f, 0.22f, 0.12f), 0.42f);
129 var host = new GameObject(distant ? "Distant fog" : "Near fog") { hideFlags = HideFlags.DontSave };
130 host.transform.SetParent(rig.transform, false);
131 host.transform.localScale = Vector3.one * size;
132 var renderer = AddRenderer(host, bankMesh, fogMaterial);
133 renderers.Add(renderer);
134 var block = new MaterialPropertyBlock();
135 block.SetColor("_FogColor", color);
136 block.SetFloat("_Opacity", Lerp(random, distant ? 0.30f : 0.12f, distant ? 0.48f : 0.22f));
137 block.SetFloat("_Seed", i * 2.73f + 0.4f);
138 block.SetFloat("_Frequency", distant ? 0.0065f : 0.013f);
139 renderer.SetPropertyBlock(block);
140 created[i] = new Bank
141 {
142 Transform = host.transform,
143 Angle = (float)(random.NextDouble() * Math.PI * 2),
144 Speed = Lerp(random, distant ? 0.008f : 0.014f, distant ? 0.02f : 0.032f) * (i % 2 == 0 ? 1f : -1f),
145 Radius = radius,
146 Height = Lerp(random, -28f, 46f),
147 Bob = Lerp(random, 8f, distant ? 26f : 14f),
148 BobSpeed = Lerp(random, 0.08f, 0.18f),
149 Phase = (float)(random.NextDouble() * Math.PI * 2)
150 };
151 }
152
153 return created;
154 }

158 // Atmospheric geometry shares meshes and materials without participating in lighting or motion vectors.

189 private static Mesh UvSphere(int longitude, int latitude)
190 {
191 // Generate latitude rings with a repeated seam vertex, then connect neighboring rings into triangles.
192 var vertices = new List<Vector3>();
193 var triangles = new List<int>();
194 for (var y = 0; y <= latitude; y++)
195 {
196 var v = y / (float)latitude;
197 var pitch = (v - 0.5f) * Mathf.PI;
198 var ring = Mathf.Cos(pitch);
199 var py = Mathf.Sin(pitch);
200 for (var x = 0; x <= longitude; x++)
201 {
202 var yaw = x / (float)longitude * Mathf.PI * 2f;
203 vertices.Add(new Vector3(Mathf.Cos(yaw) * ring, py, Mathf.Sin(yaw) * ring));
204 }
205 }
206
207 var stride = longitude + 1;
208 for (var y = 0; y < latitude; y++)
209 for (var x = 0; x < longitude; x++)
210 {
211 var a = y * stride + x;
212 var b = a + stride;
213 triangles.Add(a);
214 triangles.Add(b);
215 triangles.Add(a + 1);
216 triangles.Add(a + 1);
217 triangles.Add(b);
218 triangles.Add(b + 1);
219 }
220
221 var mesh = new Mesh { name = "Graph sky dome" };
222 mesh.SetVertices(vertices);
223 mesh.SetTriangles(triangles, 0);
224 mesh.RecalculateBounds();
225 return mesh;
226 }
