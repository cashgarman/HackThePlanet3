84 public static void BindReorder<T>(VisualElement handle, VisualElement card, ScrollView rows, IList<T> entries,
85 T entry, Func<T, string> identity, Func<bool> locked, Func<string> getDragged,
86 Action<string> setDragged,
87 Action<string> focus, Action changed)
88 {
89 handle.RegisterCallback<PointerDownEvent>(e =>
90 {
91 // Capture the dragged request identity, since list positions may change before pointer release.
92 if (e.button != 0 || locked()) return;
93 setDragged(identity(entry));
94 handle.CapturePointer(e.pointerId);
95 card.Focus();
96 e.StopPropagation();
97 });
98 handle.RegisterCallback<PointerUpEvent>(e =>
99 {
100 if (getDragged() == null || !handle.HasPointerCapture(e.pointerId)) return;
101 handle.ReleasePointer(e.pointerId);
102 var draggedId = getDragged();
103 setDragged(null);
104 if (locked()) return;
105
106 // Resolve the current source and drop target against live entries before moving the request.
107 var source = -1;
108 for (var i = 0; i < entries.Count; i++)
109 if (identity(entries[i]) == draggedId)
110 {
111 source = i;
112 break;
113 }
114
115 var target = rows.contentContainer.Children()
116 .FirstOrDefault(row => row.worldBound.Contains(e.position));
117 if (source < 0 || target == null) return;
118 var destination = -1;
119 for (var i = 0; i < entries.Count; i++)
120 if ("task-" + identity(entries[i]) == target.name)
121 {
122 destination = i;
123 break;
124 }
125
126 if (destination < 0) return;
127 var dragged = entries[source];
128 entries.RemoveAt(source);
129 entries.Insert(destination, dragged);
130 focus(draggedId);
131 changed();
132 });
133 card.RegisterCallback<KeyDownEvent>(e =>
134 {
135 // Alt-arrow reordering follows the same lock and bounds rules as dragging and restores row focus.
136 if (!e.altKey || locked()) return;
137 int source = entries.IndexOf(entry), destination = source;
138 if (e.keyCode == KeyCode.UpArrow) destination--;
139 else if (e.keyCode == KeyCode.DownArrow) destination++;
140 else return;
141 if (source < 0 || destination < 0 || destination >= entries.Count) return;
142 entries.RemoveAt(source);
143 entries.Insert(destination, entry);
144 focus(identity(entry));
145 changed();
146 rows.Q<VisualElement>("task-" + identity(entry))?.Focus();
147 e.StopPropagation();
148 });
149 }

175 public static string ShortStatus(string status)
176 {
177 switch (status)
178 {
179 case "running": return "RUN";
180 case "completed": return "OK";
181 case "failed": return "FAIL";
182 case "blocked": return "BLOCK";
183 case "cancelled": return "STOP";
184 case "skipped": return "SKIP";

186 case "paused": return "HOLD";

188 case "cancelling": return "STOP";
