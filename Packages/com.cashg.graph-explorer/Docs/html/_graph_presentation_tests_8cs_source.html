84 private static GraphScene Scene()
85 {
86 // Use domain-specific type names and metric partitions to expose accidental built-in assumptions.
87 var document = new GraphDocument("fictional:arbitrary", "Arbitrary types");
88 document.UpsertNode(new GraphNodeRecord("opaque:a", "division", "A", new Dictionary<string, GraphValue>
89 {
90 ["headcount"] = GraphValue.From(10d), ["unit"] = GraphValue.From("people"),
91 ["state"] = GraphValue.From("open")
92 }));
93 document.UpsertNode(new GraphNodeRecord("opaque:b", "employee", "B", new Dictionary<string, GraphValue>
94 {
95 ["headcount"] = GraphValue.From(50d), ["unit"] = GraphValue.From("people")
96 }));
97 document.UpsertEdge(new GraphEdgeRecord("opaque:edge", "opaque:a", "opaque:b", "reports-to"));
98 var schema = new GraphSchema();
99 schema.Register(new GraphNodeTypeDefinition("division", "Division", "Building", Color.cyan, 1));
100 schema.Register(new GraphNodeTypeDefinition("employee", "Employee", "Person", Color.magenta));
101 schema.Register(new GraphRelationDefinition("reports-to", "Reports to", Color.yellow, springStrength: .9f));
102 schema.Register(new GraphFieldDefinition("headcount", "Headcount", GraphValueType.Number,
103 partitionBy: "unit"));
104 schema.Register(new GraphFieldDefinition("state", "State",
105 categories: new Dictionary<string, GraphCategoryDefinition>
106 {
107 ["open"] = new("Open", Color.green)
108 }));
109 return new GraphScene(document, schema);
110 }
