85 public float ProximityOpacity(float screenDistance, bool pointerOverGraph, bool selected)
86 {
87 // Handle selection and pointer exit first, then fade smoothly across the configured screen-space radii.
88 if (!MouseProximityReveal || (selected && KeepSelectedLabelVisible)) return 1f;
89 if (!pointerOverGraph) return 0f;
90 var inner = Mathf.Max(0f, RevealInnerRadius);
91 var outer = Mathf.Max(inner, RevealOuterRadius);
92 if (outer <= inner) return screenDistance <= inner ? 1f : 0f;
93 return 1f - Mathf.SmoothStep(0f, 1f, Mathf.InverseLerp(inner, outer, screenDistance));
94 }
