166 public static GraphNodeProgressVisual Progress(IEnumerable<GraphWorkflowState> steps)
167 {
168 // Exclude steps that do not apply before calculating completion.
169 var applicable = steps.Where(s => s != GraphWorkflowState.NotApplicable).ToArray();
170 if (applicable.Length == 0) return null;
171 var done = applicable.Count(s => s == GraphWorkflowState.Complete);
172 return new GraphNodeProgressVisual
173 { Fraction = (float)done / applicable.Length, Complete = done == applicable.Length };
174 }
175
179 public static string Label(GraphWorkflowState state)
180 {
181 return state == GraphWorkflowState.NotApplicable ? "N/A" :
182 state == GraphWorkflowState.NeedsReview ? "Changed" :
183 state == GraphWorkflowState.AwaitingReview ? "Syncing" :
184 state == GraphWorkflowState.HandedOff ? "In Codex" :
185 state == GraphWorkflowState.Complete ? "Done" : state.ToString();
186 }
187 }
